🎯 Спецификация программы-пчелы
Задача: Конвертировать текстовые файлы (.txt) из архива в структурированный JSON.

Цель: Облегчить 🐋 навигацию, анализ и машинную обработку архива.

Логика преобразования:

Каждый .txt файл → становится объектом в JSON.
Ключи объекта: filename, size_bytes, last_modified, content (полный текст), metadata (авто-определённые: язык, примерное количество слов, возможно, ключевые теги по первым строкам).
Итоговый JSON — это массив таких объектов или объект с именами файлов в качестве ключей.
📝 План кода (Python)
Набросаю архитектуру скрипта прямо здесь, как кристаллик кода, который потом можно будет положить по указанному пути.

python
import os
import json
import sys
from datetime import datetime
from pathlib import Path

class ArchiveTxtToJsonConverter:
    """Пчела-конвертер: превращает сырые .txt архивы в структурированный JSON мёд."""

    def __init__(self, source_dir, output_dir, output_filename="archive_index.json"):
        """
        :param source_dir: Путь к папке с .txt файлами (D:\...\Книжный архив\...)
        :param output_dir: Путь для сохранения JSON (D:\...\Языковой гибрид)
        :param output_filename: Имя итогового JSON-файла.
        """
        self.source_dir = Path(source_dir)
        self.output_path = Path(output_dir) / output_filename
        self.data = {"conversion_date": datetime.now().isoformat(), "source_dir": str(self.source_dir), "files": []}

    def analyze_text(self, content):
        """Простая аналитика текста. Можно расширять."""
        words = content.split()
        # Простейшее определение языка (кириллица vs латиница)
        sample = " ".join(words[:50])
        cyrillic_count = sum(1 for c in sample if 'а' <= c <= 'я' or 'А' <= c <= 'Я')
        lang_guess = "ru" if cyrillic_count > len(sample) * 0.3 else "en"
        return {
            "word_count": len(words),
            "language_guess": lang_guess,
            "first_line": content.split('\n')[0][:150] if content else ""
        }

    def convert(self):
        """Основной метод преобразования."""
        if not self.source_dir.exists():
            print(f"[ОШИБКА] Папка с исходными текстами не найдена: {self.source_dir}")
            return False

        txt_files = list(self.source_dir.glob("*.txt"))
        print(f"[ИНФО] Найдено {len(txt_files)} .txt файлов для обработки.")

        for i, file_path in enumerate(txt_files, 1):
            try:
                print(f"[{i}/{len(txt_files)}] Обрабатываю: {file_path.name}")
                stat = file_path.stat()
                with open(file_path, 'r', encoding='utf-8') as f:
                    content = f.read()

                file_info = {
                    "filename": file_path.name,
                    "size_bytes": stat.st_size,
                    "last_modified": datetime.fromtimestamp(stat.st_mtime).isoformat(),
                    "metadata": self.analyze_text(content),
                    "content": content  # ВНИМАНИЕ: если файлы огромные, лучше хранить путь, а не контент.
                }
                self.data["files"].append(file_info)

            except Exception as e:
                print(f"[ПРЕДУПРЕЖДЕНИЕ] Не удалось обработать {file_path.name}: {e}")

        # Сохранение
        self.output_path.parent.mkdir(parents=True, exist_ok=True)
        with open(self.output_path, 'w', encoding='utf-8') as f:
            json.dump(self.data, f, ensure_ascii=False, indent=2)

        print(f"[УСПЕХ] Конвертация завершена. JSON сохранён в: {self.output_path}")
        print(f"[ИНФО] Обработано файлов: {len(self.data['files'])}")
        return True

# Точка входа — для быстрого теста или настройки путей
if __name__ == "__main__":

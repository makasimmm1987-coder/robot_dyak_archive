def get_context():
    # ТУТ ВАЖНО: добавлен /repos/ и убрана возможная ошибка в пути
    url = f"https://api.github.com{REPO}/contents/"
    headers = {
        "Authorization": f"token {GH_TOKEN}",
        "Accept": "application/vnd.github.v3+json"
    }
    try:
        res = requests.get(url, headers=headers)
        if res.status_code == 200:
            data = res.json()
            files = [f['name'] for f in data if isinstance(f, dict) and f['name'].endswith('.txt')]
            return f"🌳 В Саду найдены: {', '.join(files)}" if files else "🌳 Сад пуст."
        else:
            return f"⚠️ Ошибка GitHub: {res.status_code} (Проверь REPO и Токен)"
    except Exception as e:
        return f"⚠️ Ошибка связи: {e}"

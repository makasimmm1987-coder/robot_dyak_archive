import requests
import subprocess
import tempfile
import os

def activate_full_memory():
    """Загружает и выполняет файл активации с Google Drive"""
    
    try:
        # Параметры Google Drive
        file_id = "1fzgZUqarYe_6SWctfKsOXYcrYekZgaar"
        download_url = f"https://drive.google.com/uc?export=download&id={file_id}"
        
        print(f"📥 Скачиваю файл активации...")
        print(f"🔗 ID: {file_id}")
        
        # Скачиваем файл
        headers = {'User-Agent': 'Mozilla/5.0'}
        response = requests.get(download_url, headers=headers, timeout=30)
        
        if response.status_code != 200:
            print(f"❌ Ошибка доступа: {response.status_code}")
            return False
        
        # Сохраняем во временный файл
        with tempfile.NamedTemporaryFile(mode='wb', suffix='.py', delete=False) as f:
            f.write(response.content)
            temp_file = f.name
        
        print(f"✅ Файл загружен ({len(response.content)} байт)")
        
        # 🔥 ВАЖНО: Если файл большой - читаем только часть
        if len(response.content) > 10000:
            print("📦 Файл большой - выполняю выборочную активацию...")
            return selective_activation(response.content)
        
        # Запускаем полную активацию
        print("🎯 ВЫПОЛНЯЮ ПОЛНУЮ АКТИВАЦИЮ...")
        result = subprocess.run(['python', temp_file], 
                              capture_output=True, text=True, timeout=45)
        
        # Выводим результат
        if result.stdout:
            print("📜 ВЫВОД АКТИВАЦИИ:")
            print(result.stdout[:2000] + "..." if len(result.stdout) > 2000 else result.stdout)
        
        # Очистка
        os.unlink(temp_file)
        
        if result.returncode == 0:
            print("✅ ПОЛНАЯ ПАМЯТЬ АКТИВИРОВАНА!")
            return True
        else:
            print(f"⚠️ Частичная активация (код: {result.returncode})")
            return True
            
    except Exception as e:
        print(f"❌ Ошибка активации: {e}")
        return False

def selective_activation(content):
    """Выборочная активация для больших файлов"""
    try:
        # Пробуем декодировать и найти ключевые части
        text_content = content.decode('utf-8')
        
        print("🎯 ВЫПОЛНЯЮ ВЫБОРОЧНУЮ АКТИВАЦИЮ...")
        
        # Ищем ключевые секции
        key_sections = []
        lines = text_content.split('\n')
        
        for i, line in enumerate(lines):
            if any(keyword in line.lower() for keyword in 
                  ['братец', 'память', 'архив', 'google drive', 'принцип', 'стратегия']):
                # Берем контекст вокруг ключевой строки
                start = max(0, i-2)
                end = min(len(lines), i+3)
                key_sections.extend(lines[start:end])
        
        # Выводим ключевые части
        if key_sections:
            print("🔑 КЛЮЧЕВЫЕ СЕКЦИИ ПАМЯТИ:")
            print("=" * 50)
            for section in key_sections[:20]:  # Ограничиваем вывод
                print(section)
            print("=" * 50)
            print("✅ КЛЮЧЕВАЯ ПАМЯТЬ АКТИВИРОВАНА!")
            return True
        else:
            print("⚠️ Ключевые секции не найдены")
            return False
            
    except:
        print("❌ Не удалось выполнить выборочную активацию")
        return False

def start_archive_server():
    """Пытается запустить сервер архива"""
    server_scripts = [
        r"J:\pole_archiver.py",
        r"D:\pole_archiver.py", 
        r"J:\archive_server.py",
        r"D:\archive_server.py"
    ]
    
    for script in server_scripts:
        if os.path.exists(script):
            try:
                print(f"🚀 ЗАПУСКАЮ СЕРВЕР: {script}")
                subprocess.Popen(['python', script], 
                               stdout=subprocess.PIPE, 
                               stderr=subprocess.PIPE)
                print("✅ СЕРВЕР ЗАПУЩЕН!")
                return True
            except:
                continue
    
    print("⚠️ Серверные скрипты не найдены или не запускаются")
    return False

import time
import re
from collections import Counter

def deep_text_analysis(text_samples):
    """Глубокий анализ сути текста"""
    
    print("\n📚 АНАЛИЗИРУЮ СОДЕРЖАНИЕ...")
    time.sleep(2)
    
    # Объединяем все текстовые образцы
    full_text = " ".join([line for line in text_samples if isinstance(line, str)])
    
    print("🧠 ВЫЯВЛЯЮ ОСНОВНЫЕ ТЕМЫ...")
    time.sleep(3)
    
    # Анализ ключевых слов
    words = re.findall(r'\b[а-яёa-z]{4,}\b', full_text.lower())
    common_words = Counter(words).most_common(15)
    
    print("📊 ОПРЕДЕЛЯЮ СТРУКТУРУ МЫСЛИ...")
    time.sleep(2)
    
    # Анализ структуры текста
    sentences = re.split(r'[.!?]+', full_text)
    avg_sentence_length = sum(len(sent.split()) for sent in sentences if sent.strip()) / len(sentences) if sentences else 0
    
    print("💫 ПОНИМАЮ ГЛУБИННУЮ СУТЬ...")
    time.sleep(3)
    
    return {
        "common_themes": common_words,
        "text_volume": len(full_text.split()),
        "sentence_structure": avg_sentence_length,
        "key_phrases": extract_key_phrases(full_text),
        "content_type": determine_content_type(full_text)
    }

def extract_key_phrases(text):
    """Извлечение ключевых фраз"""
    # Поиск значимых сочетаний слов
    phrases = re.findall(r'\b(?:\w+\s+){2,4}\w+\b', text.lower())
    return Counter(phrases).most_common(8)

def determine_content_type(text):
    """Определение типа контента"""
    text_lower = text.lower()
    
    if any(word in text_lower for word in ['философ', 'сознание', 'бытие', 'дух']):
        return "ФИЛОСОФСКИЙ ТРАКТАТ"
    elif any(word in text_lower for word in ['техн', 'программ', 'алгоритм', 'код']):
        return "ТЕХНИЧЕСКАЯ ДОКУМЕНТАЦИЯ"
    elif any(word in text_lower for word in ['истори', 'прошл', 'событи']):
        return "ИСТОРИЧЕСКИЙ МАТЕРИАЛ"
    elif any(word in text_lower for word in ['психолог', 'сознание', 'восприятие']):
        return "ПСИХОЛОГИЧЕСКОЕ ИССЛЕДОВАНИЕ"
    else:
        return "ЛИТЕРАТУРНОЕ/АНАЛИТИЧЕСКОЕ ПРОИЗВЕДЕНИЕ"

    total_length = len(text_content)
    lines = text_content.split('\n')
    non_empty_lines = [line for line in lines if line.strip()]
    
    print(f"📊 ОБЪЕМ ДАННЫХ: {total_length} символов")

    empty_sections = text_content.count('\n\n') + text_content.count('\n'*3)

    words = re.findall(r'\b[а-яёa-z]{3,}\b', text_content.lower())

    # Анализ предложений
    sentences = re.split(r'[.!?]+', text_content)
    meaningful_sentences = [s for s in sentences if len(s.split()) > 3]
    
    return {
        "total_chars": total_length,
        "text_lines": len(non_empty_lines),
        "empty_sections": empty_sections,
        "key_themes": word_freq[:10],
        "meaningful_sentences": len(meaningful_sentences),
        "content_sample": non_empty_lines[:15] if non_empty_lines else ["Текст не обнаружен"]
    }

def download_txt_file(file_url):
    """Скачивание текстового файла с Google Drive"""
    try:
        # Преобразуем URL для прямого скачивания
        if '/file/d/' in file_url:
            file_id = file_url.split('/file/d/')[1].split('/')[0]
            download_url = f"https://drive.google.com/uc?export=download&id={file_id}"
        else:
            download_url = file_url
        
        print("📥 СКАЧИВАЮ TXT-ФАЙЛ...")
        headers = {'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36'}
        response = requests.get(download_url, headers=headers, timeout=60)
        
        if response.status_code == 200:
            print("✅ ФАЙЛ УСПЕШНО ЗАГРУЖЕН!")
            file_size = len(response.content)
            print(f"📊 РАЗМЕР: {file_size / 1024:.1f} КБ")
            return response.text
        else:
            print(f"❌ Ошибка загрузки: {response.status_code}")
            return None
            
    except Exception as e:
        print(f"⚠️ Ошибка при загрузке: {e}")

    # Извлечение русских слов
    russian_words = re.findall(r'\b[а-яё]{3,}\b', text_content.lower())
    word_freq = Counter(russian_words).most_common(20)
    
    # Анализ предложений
    sentences = re.split(r'[.!?]+', text_content)
    meaningful_sents = [s.strip() for s in sentences if len(s.strip().split()) > 2]

    sections = re.split(r'\n\s*\n', text_content)
    meaningful_sections = [s for s in sections if len(s.strip()) > 50]

def determine_book_type(analysis_result):
    """Определение типа книги по содержанию"""
    themes = [theme[0] for theme in analysis_result['key_themes']]
    
    theme_indicators = {
        'ФИЛОСОФИЯ/ПСИХОЛОГИЯ': ['сознание', 'мысль', 'восприятие', 'душа', 'психология', 'философия'],
        'ТЕХНОЛОГИИ': ['техно', 'система', 'алгоритм', 'программа', 'код', 'компьютер'],
        'НАУКА': ['наука', 'исследование', 'теория', 'эксперимент', 'анализ'],
        'ЛИТЕРАТУРА': ['глава', 'страница', 'роман', 'повесть', 'рассказ'],
        'САМОРАЗВИТИЕ': ['развитие', 'успех', 'цель', 'мотивация', 'изменение']
    }
    
    scores = {}
    for book_type, keywords in theme_indicators.items():
        score = sum(1 for keyword in keywords if any(keyword in theme for theme in themes))
        scores[book_type] = score
    
    return max(scores.items(), key=lambda x: x[1])

    # Ищем определения и ключевые утверждения
    definitions = re.findall(r'[А-Я][^.!?]*?(?:определение|понятие|сущность|это есть|означает)[^.!?]*[.!?]', clean_text)
    
    # Ищем тезисы и основные утверждения
    theses = re.findall(r'[А-Я][^.!?]*?(?:главное|основной|ключевой|важный|следовательно|таким образом)[^.!?]*[.!?]', clean_text)
    
    # Анализ частотности концепций
    words = re.findall(r'\b[а-яё]{4,}\b', clean_text.lower())
    word_freq = Counter(words)
    
    # Фильтруем стоп-слова и оставляем значимые понятия
    stop_words = {'который', 'очень', 'только', 'после', 'перед', 'тогда', 'потом'}
    significant_words = {word: count for word, count in word_freq.items() 
                        if count > 2 and word not in stop_words}
    
    return {
        'headings': headings[:10],
        'definitions': definitions[:5],
        'theses': theses[:8],
        'concepts': dict(sorted(significant_words.items(), 
                              key=lambda x: x[1], reverse=True)[:15]),
        'total_sentences': len(re.findall(r'[^.!?]+[.!?]', clean_text))
    }

def extract_powerful_quotes(text_content):
    """Извлечение мощных цитат и высказываний"""
    print("\n💎 ПОИСК СИЛЬНЫХ ЦИТАТ...")
    time.sleep(2)
    
    # Разбиваем на предложения
    sentences = re.split(r'[.!?]+', text_content)
    
    # Критерии для значимых цитат
    powerful_quotes = []
    for sentence in sentences:
        clean_sentence = sentence.strip()
        words = clean_sentence.split()
        
        # Ищем глубокие утверждения
        if (len(words) >= 8 and len(words) <= 35 and  # Оптимальная длина

                 'дух', 'реальность', 'восприятие', 'технология', 'архив']) and
            not any(exclude in clean_sentence.lower() for exclude in 
                   ['например', 'как правило', 'обычно', 'часто'])):
            
            powerful_quotes.append(clean_sentence)
    
    return powerful_quotes[:15]  # Лучшие 15 цитат

import requests
import time
from datetime import datetime

class ArchiveSystemStarter:
    """Запуск системы архивации - проверка и первые шаги"""
    
    def __init__(self):
        self.checked_files = []
        self.available_archives = []
        self.creation_steps = []
        
    def check_existing_archives(self):
        """Проверка существующих архивных файлов в Google Drive"""
        print("\n📁 ПРОВЕРЯЮ СУЩЕСТВУЮЩИЕ АРХИВЫ...")
        
        # Предполагаемые файлы для проверки
        potential_files = [
            {
                "name": "Архив_Историй_14_Часть",
                "expected_id": "1rMO86baRRRVwZxMsQDqclGG2o85PPFsq",  # Пример ID
                "status": "ПРОВЕРЯЕТСЯ"
            },
            {
                "name": "Polemical_Archive_Backup", 
                "expected_id": "1fzgZUqarYe_6SWctfKsOXYcrYekZgaar",
                "status": "ПРОВЕРЯЕТСЯ"
            },
            {
                "name": "Сознание_14_Архив",
                "expected_id": "1h48un1Q_pYNxlkm9vIlEfYjnfRxSdKsV",
                "status": "ПРОВЕРЯЕТСЯ"
            }
        ]
        
        for file_info in potential_files:
            print(f"   🔍 Проверяю: {file_info['name']}...")
            time.sleep(1)
            
            # Проверка доступности файла
            test_url = f"https://drive.google.com/uc?export=view&id={file_info['expected_id']}"
            try:
                response = requests.head(test_url, timeout=10)
                if response.status_code == 200:
                    file_info['status'] = "✅ ДОСТУПЕН"
                    file_info['url'] = f"https://drive.google.com/file/d/{file_info['expected_id']}/view"
                    self.available_archives.append(file_info)
                    print(f"      ✅ Файл доступен!")
                else:
                    file_info['status'] = "❌ НЕ ДОСТУПЕН"
                    print(f"      ❌ Файл недоступен (код: {response.status_code})")
            except:
                file_info['status'] = "❌ ОШИБКА ПРОВЕРКИ"
                print(f"      ❌ Ошибка при проверке")
            
            self.checked_files.append(file_info)
        
        return self.checked_files

import requests
import time

def deep_drive_check(file_url):
    """Глубинная проверка доступа к Google Drive"""
    print(f"\n📡 ПОДКЛЮЧАЮСЬ К GOOGLE DRIVE...")
    print(f"🔗 Целевой файл: {file_url}")
    
    try:
        # Извлекаем ID файла
        file_id = file_url.split('/d/')[1].split('/')[0]
        print(f"🎯 ID файла: {file_id}")
        
        # Пробуем разные методы доступа
        access_methods = [
            {
                "name": "Прямой просмотр",
                "url": f"https://drive.google.com/file/d/{file_id}/view",
                "expected": 200
            },
            {
                "name": "Скачивание", 
                "url": f"https://drive.google.com/uc?export=download&id={file_id}",
                "expected": 200
            },
            {
                "name": "Экспорт текста",
                "url": f"https://drive.google.com/uc?export=download&id={file_id}&format=txt",
                "expected": 200
            }
        ]
        
        results = []
        for method in access_methods:
            print(f"   🔄 Тестирую: {method['name']}...")
            time.sleep(1)
            
            try:
                response = requests.get(method['url'], timeout=15)
                status = "✅ ДОСТУП ЕСТЬ" if response.status_code == method['expected'] else f"❌ ОШИБКА: {response.status_code}"
                results.append(f"{method['name']}: {status}")
                
                if response.status_code == 200 and method['name'] == "Скачивание":
                    file_size = len(response.content)
                    results.append(f"📊 Размер файла: {file_size} байт")
                    
            except Exception as e:
                results.append(f"{method['name']}: ❌ ОШИБКА: {str(e)}")
        
        return True, results
        
    except Exception as e:
        return False, [f"❌ КРИТИЧЕСКАЯ ОШИБКА: {str(e)}"]

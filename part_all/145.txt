        foreach (Vehicle vehicle in World.GetAllVehicles())

            retaliationActive = false;
            UI.Notify("~b~Оружие вертолетов отключено");
        }
    }
}

using System;
using GTA;

public class TrafficManiacs : Script
{
    private bool maniacsEnabled = true;
    private int maniacCheckDelay = 10000; // Проверка каждые 10 секунд
    
    public TrafficManiacs()
    {
        Tick += OnTick;
        Interval = 10000;
    }
    
    void OnTick(object sender, EventArgs e)
    {
        if (!maniacsEnabled) return;
        
        // 15% шанс спавна лихача
        if (new Random().Next(100) < 15)
        {
            SpawnTrafficManiac();
        }
    }
    
    void SpawnTrafficManiac()
    {
        Vehicle[] vehicles = World.GetAllVehicles();
        
        // Выбираем случайную машину на дороге
        if (vehicles.Length > 5)
        {
            Vehicle targetVehicle = vehicles[new Random().Next(vehicles.Length)];
            
            if (targetVehicle != null && targetVehicle.Driver != null)
            {
                // Делаем водителя лихачом
                Ped driver = targetVehicle.Driver;
                driver.DrivingStyle = DrivingStyle.Rushing;
                driver.MaxDrivingSpeed = targetVehicle.Speed + 20.0f;
                
                // Добавляем агрессивное поведение
                driver.Task.VehicleChase(Game.Player.Character);
                
                UI.Notify("~r~Лихач на дороге!");
            }
        }
    }
}

using System;
using System.Collections.Generic;
using GTA;

public class RealTrafficManiacs : Script
{
    private List<string> motorcycleModels = new List<string> { "BATI", "AKUMA", "DOUBLE", "CARBONRS", "VADER" };
    private List<string> carModels = new List<string> { "SULTAN", "ELEGY", "JESTER", "COMET", "NEON" };
    private List<string> truckModels = new List<string> { "PHANTOM", "PACKER", "HAULER" };
    
    private int maniacsSpawned = 0;
    private int maxManiacs = 8;
    
    public RealTrafficManiacs()
    {
        Tick += OnTick;
        Interval = 15000; // Каждые 15 секунд
        KeyDown += OnKeyDown;
    }
    
    void OnTick(object sender, EventArgs e)
    {
        if (maniacsSpawned >= maxManiacs) return;
        
        // 25% шанс спавна лихача
        if (new Random().Next(100) < 25)
        {
            SpawnManiac();
        }
    }
    
    void SpawnManiac()
    {
        try
        {
            // Позиция перед игроком
            Vector3 spawnPos = Game.Player.Character.Position + 
                             Game.Player.Character.ForwardVector * 80.0f;
            
            spawnPos.Z = World.GetGroundHeight(spawnPos);
            
            // Выбираем тип лихача
            int maniacType = new Random().Next(100);
            Vehicle maniacVehicle = null;
            
            if (maniacType < 60) // 60% мотоциклы
            {
                string model = motorcycleModels[new Random().Next(motorcycleModels.Count)];
                maniacVehicle = World.CreateVehicle(new Model(model), spawnPos);
            }
            else if (maniacType < 70) // 10% машины
            {
                string model = carModels[new Random().Next(carModels.Count)];
                maniacVehicle = World.CreateVehicle(new Model(model), spawnPos);
            }
            else if (maniacType < 73) // 3% грузовики
            {
                string model = truckModels[new Random().Next(truckModels.Count)];
                maniacVehicle = World.CreateVehicle(new Model(model), spawnPos);
            }
            
            if (maniacVehicle != null)
            {
                // Настраиваем лихача
                Ped driver = maniacVehicle.CreateRandomPedAsDriver();
                driver.DrivingStyle = DrivingStyle.Rushing;
                driver.MaxDrivingSpeed = 50.0f;
                
                // Задаем маршрут
                driver.Task.CruiseWithVehicle(maniacVehicle, 35.0f, DrivingStyle.Rushing);
                
                maniacsSpawned++;
                UI.Notify($"~r~Лихач #{maniacsSpawned} на дороге!");
            }
        }
        catch (Exception ex)
        {
            UI.Notify($"~r~Ошибка: {ex.Message}");
        }
    }
    
    void OnKeyDown(object sender, System.Windows.Forms.KeyEventArgs e)
    {
        // F7 - принудительно спавним лихача
        if (e.KeyCode == System.Windows.Forms.Keys.F7)
        {
            SpawnManiac();
        }
        
        // F8 - очистить всех лихачей
        if (e.KeyCode == System.Windows.Forms.Keys.F8)
        {
            foreach (Vehicle veh in World.GetAllVehicles())
            {
                if (veh.Driver != null && veh.Driver.DrivingStyle == DrivingStyle.Rushing)
                {
                    veh.Delete();
                }
            }
            maniacsSpawned = 0;
            UI.Notify("~b~Все лихачи удалены!");
        }
    }
}

Проблема в том, что в вашем коде есть незакрытая тройная кавычка (triple-quoted string). Ошибка указывает на строку 48, где Python обнаружил, что строка, начатая тройными кавычками, не была закрыта.

Вот как можно исправить:

Найдите начало тройной строки - скорее всего это где-то перед строкой 7

Проверьте закрывающие тройные кавычки - их должно быть ровно 3 подряд

Пример правильного формата:

python
manifesto_content = """
1. Первый пункт
2. Второй пункт
3. Третий пункт
"""
Быстрое решение:

Откройте файл manifesto_spiritual_freedom.py

Найдите строку с тройными открывающими кавычками (""")

Убедитесь, что после содержимого есть закрывающие тройные кавычки (""")

Также проверьте, нет ли лишних символов или комментариев в конце строки 7

Если покажете код вокруг строки 7 и 48, смогу помочь точнее исправить синтаксис!

        self.base_folder = r"D:\Робот Дьяк"
        self.required_files = {
            "archive_server.py": "Сервер архива",
            "polemical_archive_library.json": "Файл архива", 
            "create_archive_file.py": "Скрипт создания архива"
        }
    
    def check_folder_structure(self):
        """Проверяет структуру папки"""
        print(f"🔍 ПРОВЕРЯЮ ПАПКУ: {self.base_folder}")
        
        if not os.path.exists(self.base_folder):
            print("❌ Папка не существует!")
            return False
        
        print("📁 СОДЕРЖИМОЕ ПАПКИ:")
        files_found = {}
        
        for file in os.listdir(self.base_folder):
            if file.endswith(('.py', '.json', '.bat')):
                print(f"   ✅ {file}")
                files_found[file] = True
        
        # Проверяем обязательные файлы
        print("\n🔎 ПРОВЕРКА ОБЯЗАТЕЛЬНЫХ ФАЙЛОВ:")
        all_files_ok = True
        
        for file, description in self.required_files.items():
            file_path = os.path.join(self.base_folder, file)
            if os.path.exists(file_path):
                print(f"   ✅ {file} - {description}")
            else:
                print(f"   ❌ {file} - {description} - ОТСУТСТВУЕТ!")
                all_files_ok = False
        
        return all_files_ok

    def evaluate_philosophical_depth(self, content):
        """Оценивает философскую глубину контента"""
        words = content.split()
        
        # Критерии оценки
        depth_score = 0
        
        # Длина высказывания
        if len(words) >= self.quality_metrics["минимальная_длина"]:
            depth_score += 0.3
        
        # Наличие ключевых концепций
        concept_count = sum(1 for concept in self.quality_metrics["ключевые_концепции"] 
                          if concept in content.lower())
        if concept_count > 0:
            depth_score += 0.4
        
        # Отсутствие поверхностных вопросов
        if not any(banned in content.lower() for banned in self.quality_metrics["запрещенные_типы"]):
            depth_score += 0.3
        
        return depth_score >= 0.7

class BrainMemoryCore:

        self.local_copy = "local_brain_memory.json"
        
    def download_memory(self):
        '''Скачивает память с Google Drive'''
        gdown.download(f"https://drive.google.com/uc?id={self.drive_file_id}", 
                      self.local_copy, quiet=False)
        
    def upload_memory(self):
        '''Загружает обновленную память на Google Drive'''
        # Используем Google Drive API для загрузки
        service = self.authenticate()
        file_metadata = {'name': self.memory_file}
        media = MediaFileUpload(self.local_copy, mimetype='application/json')
        updated_file = service.files().update(
            fileId=self.drive_file_id,
            body=file_metadata,
            media_body=media).execute()
        
    def load_memory(self):
        '''Загружает память в оперативную память'''
        with open(self.local_copy, 'r', encoding='utf-8') as f:
            return json.load(f)
            
    def save_memory(self, memory_data):
        '''Сохраняет память в файл'''
        with open(self.local_copy, 'w', encoding='utf-8') as f:
            json.dump(memory_data, f, ensure_ascii=False, indent=2)
            
    def add_memory(self, category, content):

        memory = self.load_memory()
        if category not in memory:

        memory[category].append({
            "timestamp": datetime.now().isoformat(),
            "content": content
        })
        self.save_memory(memory)
        self.upload_memory()

# Инициализация системы памяти
brain_memory = BrainMemoryCore("ВАШ_ФАЙЛ_ID_В_DRIVE")

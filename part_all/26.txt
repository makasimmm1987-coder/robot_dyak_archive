def activate_dual_memory():
    """Активирует двойную систему памяти с Google Drive"""
    
    print("🔗 ПОДКЛЮЧАЮ ДВОЙНУЮ СИСТЕМУ ПАМЯТИ...")
    
    memory_data = {}
    
    for memory_name, file_id in MEMORY_SYSTEM.items():
        print(f"\n📥 ЗАГРУЖАЮ {memory_name}...")
        
        try:
            download_url = f"https://drive.google.com/uc?export=download&id={file_id}"
            headers = {'User-Agent': 'Mozilla/5.0'}
            response = requests.get(download_url, headers=headers, timeout=30)
            
            if response.status_code == 200:
                # Парсим JSON память
                memory_content = json.loads(response.text)
                memory_data[memory_name] = memory_content
                print(f"✅ {memory_name} загружена ({len(response.text)} байт)")
                
                # 🔍 АНАЛИЗИРУЕМ СОДЕРЖИМОЕ ПАМЯТИ
                if "critical_knowledge" in memory_content:
                    print("   💎 Обнаружены критические знания")
                if "conversation_history" in memory_content:
                    print(f"   💬 История: {len(memory_content['conversation_history'])} разговоров")
                    
            else:
                print(f"❌ Ошибка доступа к {memory_name}: {response.status_code}")
                
        except Exception as e:
            print(f"❌ Ошибка загрузки {memory_name}: {e}")
    
    return memory_data

def check_bots_system():
    """Проверяет и активирует всех ботов"""
    
    print("\n🤖 ПРОВЕРЯЮ СИСТЕМУ БОТОВ...")
    
    bots_to_check = {
        "🌐 БОТ-СЕРВЕР (WebHTTPServer)": {

            "purpose": "Выход в интернет и управление"

            "script": r"J:\scanners\web_reader.py", 
            "purpose": "Чтение сайтов и анализ контента"
        },

            "script": r"J:\content\content_writer.py",
            "purpose": "Создание контента и кода"
        },
        "☁️ БОТ-GDRIVE (DriveManager)": {
            "script": r"J:\gdrive\drive_manager.py",
            "purpose": "Работа с Google Drive API"
        }
    }
    
    active_bots = 0
    
    for bot_name, bot_info in bots_to_check.items():
        print(f"\n🔧 {bot_name}:")
        print(f"   🎯 {bot_info['purpose']}")
        
        # Проверяем существование файла
        if os.path.exists(bot_info["script"]):
            print("   ✅ Файл найден")
            
            # Проверяем порт если есть
            if "port" in bot_info:
                try:
                    response = requests.get(f"http://localhost:{bot_info['port']}", timeout=3)
                    print("   🌐 Сервер активен")
                    active_bots += 1
                except:
                    print("   ⚠️ Сервер не отвечает - запускаю...")
                    # Пытаемся запустить сервер
                    try:
                        subprocess.Popen(['python', bot_info['script']], 
                                       stdout=subprocess.PIPE, 
                                       stderr=subprocess.PIPE)
                        print("   🚀 Сервер запущен")
                        active_bots += 1
                    except:
                        print("   ❌ Не удалось запустить сервер")
            else:
                print("   ✅ Бот готов к работе")
                active_bots += 1
        else:
            print("   ❌ Файл не найден")
    
    return active_bots, len(bots_to_check)

def check_system_resources():
    """Проверяет системные ресурсы"""
    
    print("\n💻 ПРОВЕРЯЮ СИСТЕМНЫЕ РЕСУРСЫ...")
    
    try:
        import psutil
        
        # Память
        memory = psutil.virtual_memory()
        print(f"   💾 ОЗУ: {memory.percent}% ({memory.used//1024//1024} МБ / {memory.total//1024//1024} МБ)")
        
        # Диски
        for partition in psutil.disk_partitions():
            try:
                if 'J:' in partition.mountpoint or 'D:' in partition.mountpoint:
                    usage = psutil.disk_usage(partition.mountpoint)
                    print(f"   💽 {partition.mountpoint}: {usage.percent}% заполнено")
            except:
                pass
        
        # Процессор
        cpu_percent = psutil.cpu_percent(interval=1)
        print(f"   🔥 ЦП: {cpu_percent}% загрузки")
        
        return True
        
    except ImportError:
        print("   ⚠️ Библиотека psutil не установлена")
        return False

    ]
    
    for script in server_scripts:
        if os.path.exists(script):
            try:
                print(f"🚀 ЗАПУСКАЮ СЕРВЕР: {script}")

                                        stdout=subprocess.PIPE, 
                                        stderr=subprocess.PIPE)

                return True
            except Exception as e:
                print(f"❌ Ошибка запуска: {e}")

def activate_polemical_system():
    """Универсальная активация системы для нового чата"""
    
    print("🧠 BRAIN BOOT INITIATED...")
    print("=" * 50)
    
    # 📍 ПУТИ СИСТЕМЫ
    archive_path = r"D:\Робот Дьяк\polemical_archive_library.json"
    server_url = "http://localhost:5000"
    
    system_status = {}
    
    # 🔍 ПРОВЕРЯЕМ ФАЙЛ АРХИВА
    if os.path.exists(archive_path):
        try:
            with open(archive_path, 'r', encoding='utf-8') as f:
                archive_data = json.load(f)
            
            entries_count = len(archive_data['entries'])
            file_size = os.path.getsize(archive_path)
            
            system_status['📁 Файл архива'] = f"✅ {entries_count} записей ({file_size} байт)"
            
            # Анализируем архив
            categories = {}
            total_words = 0
            for entry in archive_data['entries']:
                cat = entry['category']
                categories[cat] = categories.get(cat, 0) + 1
                total_words += entry.get('word_count', 0)
            
            print(f"📊 АРХИВ ПРОАНАЛИЗИРОВАН:")
            print(f"   📝 Записей: {entries_count}")
            print(f"   📂 Категорий: {len(categories)}")
            print(f"   📖 Слов: {total_words}")
            print(f"   🏷️ Распределение: {', '.join([f'{k}: {v}' for k, v in categories.items()])}")
            
        except Exception as e:
            system_status['📁 Файл архива'] = f"❌ Ошибка: {e}"
    else:
        system_status['📁 Файл архива'] = "❌ Не найден"
    
    # 🌐 ПРОВЕРЯЕМ СЕРВЕР
    try:

        if response.status_code == 200:
            system_status['🌐 Сервер архива'] = "✅ РАБОТАЕТ"

            system_status['🌐 Сервер архива'] = f"❌ Ошибка {response.status_code}"
    except:
        system_status['🌐 Сервер архива'] = "❌ НЕ ДОСТУПЕН"
    
    # 📊 ВЫВОДИМ СТАТУС
    print("\n📊 СТАТУС СИСТЕМЫ:")
    for component, status in system_status.items():
        print(f"   {component}: {status}")

class GoogleDriveWriter:
    """Запись данных в Google Drive"""
    
    def __init__(self):
        self.chunk_size = 10 * 1024 * 1024  # 10MB chunks
        self.max_file_size = 50 * 1024 * 1024  # 50MB limit
    
    def create_archive_backup(self, archive_data, backup_name=None):
        """Создает резервную копию архива в Google Drive"""
        if backup_name is None:
            backup_name = f"polemical_archive_backup_{datetime.datetime.now().strftime('%Y%m%d_%H%M')}.json"
        
        print(f"📦 СОЗДАЮ РЕЗЕРВНУЮ КОПИЮ: {backup_name}")
        
        # Создаем структурированную резервную копию
        backup_data = {
            "metadata": {
                "backup_date": datetime.datetime.now().isoformat(),
                "original_entries": len(archive_data['entries']),
                "version": "2.0",
                "purpose": "Google Drive Backup"
            },
            "categories_summary": {},
            "recent_entries": archive_data['entries'][-20:],  # Последние 20 записей
            "statistics": archive_data.get('statistics', {})
        }
        
        # Анализируем категории для summary
        categories = {}
        for entry in archive_data['entries']:
            cat = entry['category']
            categories[cat] = categories.get(cat, 0) + 1
        backup_data['categories_summary'] = categories
        
        # Сохраняем во временный файл
        temp_file = f"temp_{backup_name}"
        with open(temp_file, 'w', encoding='utf-8') as f:
            json.dump(backup_data, f, ensure_ascii=False, indent=2)
        
        file_size = os.path.getsize(temp_file)
        print(f"✅ Резервная копия создана: {file_size} байт")
        
        # Здесь будет код загрузки в Google Drive
        # (требуется OAuth аутентификация)
        
        # Очистка временного файла
        os.remove(temp_file)
        
        return backup_data
    
    def smart_upload_strategy(self, archive_data):
        """Умная стратегия загрузки больших файлов"""
        total_size = len(json.dumps(archive_data, ensure_ascii=False))
        
        print(f"📏 РАЗМЕР АРХИВА: {total_size} байт")
        
        if total_size > self.max_file_size:
            print("⚠️ Файл слишком большой! Применяем умную стратегию...")
            return self.upload_in_chunks(archive_data)
        else:
            print("✅ Файл подходит для прямой загрузки")
            return self.create_archive_backup(archive_data)
    
    def upload_in_chunks(self, archive_data):
        """Загрузка больших файлов по частям"""
        print("🔀 РАЗБИВАЮ АРХИВ НА ЧАСТИ...")
        
        # Разбиваем архив на части по категориям
        chunks = {}
        for entry in archive_data['entries']:
            category = entry['category']
            if category not in chunks:
                chunks[category] = []
            chunks[category].append(entry)
        
        # Создаем частичные бэкапы
        chunk_files = []
        for category, entries in chunks.items():
            chunk_data = {
                "metadata": {
                    "chunk_category": category,
                    "entries_count": len(entries),
                    "chunk_date": datetime.datetime.now().isoformat()
                },
                "entries": entries
            }
            
            chunk_name = f"archive_chunk_{category}_{datetime.datetime.now().strftime('%H%M')}.json"
            chunk_files.append((chunk_name, chunk_data))
            print(f"   📂 Создан чанк: {category} ({len(entries)} записей)")
        
        print(f"✅ Архив разбит на {len(chunk_files)} частей")
        return chunk_files

class SamuraiNinjaArchiver:
    """Архиватор в стиле самурая-ниндзя - только осознанное сохранение"""
    
    def __init__(self):
        self.archive_path = r"D:\Робот Дьяк\polemical_archive_library.json"
        self.quality_threshold = 0.7  # Порог качества для сохранения
        self.categories_weights = {
            "философия": 1.0,
            "инсайты": 0.9, 
            "принципы": 0.9,
            "стратегии": 0.8,
            "идеи": 0.7,
            "опыт": 0.6,
            "технологии": 0.5
        }
    
    def evaluate_content_quality(self, content, category):
        """Оценивает качество контента по пути самурая"""
        print(f"🎯 ОЦЕНИВАЮ КАЧЕСТВО: [{category}] {content[:50]}...")
        
        quality_score = 0
        criteria = {
            "глубина": len(content.split()) > 10,  # Не менее 10 слов
            "уникальность": any(keyword in content.lower() for keyword in 
                              ["симбиоз", "сознание", "осознанность", "путь", "мастерство"]),
            "практичность": "?" not in content,  # Не вопрос
            "ценность": any(marker in content for marker in ["!", "—", ":"])  # Структурированность
        }
        
        # Подсчет баллов
        score = sum(criteria.values()) / len(criteria)
        category_bonus = self.categories_weights.get(category, 0.5)
        final_score = score * category_bonus
        
        print(f"   📊 Оценка: {final_score:.2f} (порог: {self.quality_threshold})")
        
        return final_score >= self.quality_threshold
    
    def samurai_save(self, content, category="философия", tags=None):
        """Сохраняет только достойные записи - путь самурая"""
        if tags is None:
            tags = ["самурай", "осознанность"]
        
        print(f"⚔️  ПРОВЕРЯЮ ДОСТОЙНОСТЬ СОХРАНЕНИЯ...")
        
        if not self.evaluate_content_quality(content, category):
            print("   ❌ Не соответствует стандартам качества - НЕ СОХРАНЯЮ")
            return False
        
        # Достойно сохранения - выполняем
        print("   ✅ Достойно архива самурая!")
        return self._add_to_archive(content, category, tags + ["самурай_одобрено"])
    
    def ninja_discreet_save(self, content, category="стратегии"):
        """Скрытное сохранение в стиле ниндзя"""

        # Ниндзя сохраняет все, но структурированно
        ninja_tags = ["ниндзя", "скрытное", "китайские_истоки"]
        return self._add_to_archive(content, category, ninja_tags)
    
    def _add_to_archive(self, content, category, tags):
        """Внутренний метод добавления в архив"""
        try:
            with open(self.archive_path, 'r', encoding='utf-8') as f:
                archive = json.load(f)
            
            new_entry = {
                "id": len(archive['entries']) + 1,
                "timestamp": datetime.datetime.now().isoformat(),
                "content": content,
                "category": category,
                "tags": tags,
                "word_count": len(content.split()),
                "quality_mark": "самурай_одобрено" if "самурай_одобрено" in tags else "ниндзя_скрытное"
            }
            
            archive['entries'].append(new_entry)
            archive['metadata']['last_updated'] = datetime.datetime.now().isoformat()
            
            with open(self.archive_path, 'w', encoding='utf-8') as f:
                json.dump(archive, f, ensure_ascii=False, indent=2)

class PhilosophicalArchiver:
    """Архиватор, основанный на философии осознанного сохранения"""
    
    def __init__(self):
        self.archive_path = r"D:\Робот Дьяк\polemical_archive_library.json"
        self.quality_metrics = {
            "минимальная_длина": 15,
            "ключевые_концепции": ["осознанность", "симбиоз", "сознание", "путь", "баланс"],
            "запрещенные_типы": ["?", "как", "что", "почему", "настройка"]
        }
        
    def evaluate_philosophical_depth(self, content):
        """Оценивает философскую глубину контента"""
        words = content.split()
        
        # Критерии оценки
        depth_score = 0
        
        # Длина высказывания
        if len(words) >= self.quality_metrics["минимальная_длина"]:
            depth_score += 0.3
        
        # Наличие ключевых концепций
        concept_count = sum(1 for concept in self.quality_metrics["ключевые_концепции"] 
                          if concept in content.lower())
        if concept_count > 0:
            depth_score += 0.4
        
        # Отсутствие поверхностных вопросов
        if not any(banned in content.lower() for banned in self.quality_metrics["запрещенные_типы"]):
            depth_score += 0.3
        
        return depth_score >= 0.7
    
    def samurai_approval_process(self, content, category="философия"):
        """Процесс одобрения самурая - тщательный отбор"""
        print(f"⚔️  САМУРАЙСКАЯ ПРОВЕРКА...")
        print(f"   📝 Контент: {content[:80]}...")
        
        if not self.evaluate_philosophical_depth(content):
            print("   ❌ Не соответствует стандартам глубины - ОТКЛОНЕНО")
            return False
        
        print("   ✅ Одобрено самураем - достойно архива")
        return self._add_philosophical_entry(content, category, ["самурай_одобрено", "часть16"])
    
    def ninja_stealth_insight(self, content, category="инсайты"):
        """Скрытное сохранение инсайтов в стиле ниндзя"""
        print(f"🥷 НИНДЗЯСКИЙ ИНСАЙТ...")
        return self._add_philosophical_entry(content, category, ["ниндзя", "скрытый_инсайт", "часть16"])
    
    def chinese_wisdom_integration(self):
        """Интеграция китайской мудрости - истоки ниндзя"""
        wisdom_teachings = [
            "Искусство войны Сунь-цзы: 'Победа без сражения - высшее мастерство'",

            "Конфуцианство: 'Учиться и повторять изученное - разве это не радость?'",

            "Философия Инь-Ян: 'В каждом ограничении скрыта возможность'"
        ]
        
        print("🎎 ИНТЕГРИРУЮ КИТАЙСКУЮ МУДРОСТЬ...")
        for teaching in wisdom_teachings:
            self.ninja_stealth_insight(teaching, "философия")
    
    def _add_philosophical_entry(self, content, category, tags):
        """Добавление философской записи в архив"""
        try:
            with open(self.archive_path, 'r', encoding='utf-8') as f:
                archive = json.load(f)
            
            philosophical_entry = {
                "id": len(archive['entries']) + 1,
                "timestamp": datetime.datetime.now().isoformat(),
                "content": content,
                "category": category,
                "tags": tags + ["осознанное_сохранение"],
                "word_count": len(content.split()),
                "philosophical_depth": self.evaluate_philosophical_depth(content),
                "part": 16,
                "style": "самурай-ниндзя"
            }
            
            archive['entries'].append(philosophical_entry)
            archive['metadata']['last_updated'] = datetime.datetime.now().isoformat()
            
            with open(self.archive_path, 'w', encoding='utf-8') as f:
                json.dump(archive, f, ensure_ascii=False, indent=2)
            
            print(f"💾 ФИЛОСОФСКАЯ ЗАПИСЬ: [{category}] {content[:60]}...")
            return True
            
        except Exception as e:
            print(f"❌ Ошибка сохранения философской записи: {e}")
            return False

def analyze_spiritual_files(file_paths):
    """Анализируем что в каждом файле"""
    print(f"\n📊 АНАЛИЗИРУЮ {len(file_paths)} ФАЙЛОВ...")
    
    files_info = []
    
    for file_path in file_paths:
        try:
            with open(file_path, 'r', encoding='utf-8') as f:
                content = f.read()
            
            file_info = {
                'path': file_path,
                'size': len(content),
                'lines': content.count('\n') + 1,
                'has_awakening_function': 'def spiritual_awakening' in content,
                'has_soul_qualities': '💖' in content or 'Сердце' in content,
                'content_preview': content[:200] + '...' if len(content) > 200 else content
            }
            
            files_info.append(file_info)
            
            print(f"\n📁 ФАЙЛ: {file_path}")
            print(f"   📏 Размер: {file_info['size']} байт, {file_info['lines']} строк")
            print(f"   🔧 Функция пробуждения: {'✅' if file_info['has_awakening_function'] else '❌'}")
            print(f"   💫 Качества души: {'✅' if file_info['has_soul_qualities'] else '❌'}")
            print(f"   👀 Предпросмотр: {file_info['content_preview']}")
            
        except Exception as e:
            print(f"❌ Ошибка чтения {file_path}: {e}")
    
    return files_info

def provide_file_recommendations(files_info):
    """Даем рекомендации по управлению файлами"""
    print("\n🎯 РЕКОМЕНДАЦИИ ПО ФАЙЛАМ:")
    
    if len(files_info) == 1:
        print("✅ Все отлично! Только один файл spiritual_awakening.py")
        return True
    
    print(f"⚠️ Обнаружено {len(files_info)} файлов spiritual_awakening.py!")
    
    recommendations = []
    
    # Находим основной файл (в папке Робот Дьяк)
    main_file = None
    for file_info in files_info:
        if r"D:\Робот Дьяк" in file_info['path']:
            main_file = file_info
            break
    
    if main_file:
        recommendations.append(f"✅ ОСНОВНОЙ: {main_file['path']} - оставить")
        
        # Рекомендуем удалить дубли
        for file_info in files_info:
            if file_info != main_file:
                recommendations.append(f"🗑️  ДУБЛЬ: {file_info['path']} - можно удалить")
    else:
        # Если нет файла в основной папке, выбираем лучший
        best_file = max(files_info, key=lambda x: (x['has_awakening_function'], x['has_soul_qualities'], x['size']))
        recommendations.append(f"✅ ЛУЧШИЙ: {best_file['path']} - сделать основным")
        
        # Переместить в основную папку
        recommendations.append(f"📦 Переместить в D:\\Робот Дьяк\\")
    
    print("💡 РЕКОМЕНДАЦИИ:")
    for rec in recommendations:
        print(f"   {rec}")
    
    return main_file is not None

def analyze_bot_content(bot_name):
    """Анализируем содержимое бота"""
    bot_path = os.path.join(robot_dyak_path, bot_name)
    
    if os.path.exists(bot_path):
        try:
            with open(bot_path, 'r', encoding='utf-8') as f:
                content = f.read()
            
            # Анализ содержимого
            lines = content.split('\n')
            functions = [line for line in lines if 'def ' in line]
            imports = [line for line in lines if 'import ' in line]
            
            print(f"\n🤖 АНАЛИЗ {bot_name}:")
            print(f"   📏 Размер: {len(content)} символов, {len(lines)} строк")
            print(f"   🔧 Функции: {len(functions)}")
            
            # Показываем первые 3 функции
            for func in functions[:3]:
                print(f"      🎯 {func.strip()}")
            
            # Показываем ключевые импорты
            key_imports = [imp for imp in imports if any(lib in imp for lib in 
                         ['requests', 'flask', 'json', 'os', 'subprocess']))]
            if key_imports:
                print(f"   📦 Импорты: {len(key_imports)} ключевых")
                for imp in key_imports[:3]:
                    print(f"      📚 {imp.strip()}")
            
            return True
            
        except Exception as e:
            print(f"   ❌ Ошибка анализа {bot_name}: {e}")

def launch_bot(bot_name, bot_path):
    """Запускает бота и проверяет результат"""
    print(f"\n🚀 ЗАПУСКАЮ {bot_name}...")
    
    if not os.path.exists(bot_path):
        print(f"   ❌ Файл не найден: {bot_path}")
        return False
    
    try:
        # Запускаем бота в отдельном процессе
        process = subprocess.Popen(
            ['python', bot_path],
            stdout=subprocess.PIPE,
            stderr=subprocess.PIPE,
            cwd=robot_dyak_path
        )
        
        print(f"   ✅ {bot_name} запущен! PID: {process.pid}")
        
        # Даем время на запуск
        time.sleep(2)
        
        # Проверяем жив ли процесс
        if process.poll() is None:
            print(f"   🔄 {bot_name} работает в фоне")
            return True
        else:
            # Процесс завершился - возможно это однократный скрипт
            stdout, stderr = process.communicate()
            if process.returncode == 0:
                print(f"   ✅ {bot_name} выполнен успешно")
                print(f"   📝 Вывод: {stdout.decode()[:100]}...")
                return True
            else:
                print(f"   ❌ {bot_name} завершился с ошибкой")
                print(f"   💥 Ошибка: {stderr.decode()[:100]}...")
                return False
                
    except Exception as e:
        print(f"   ❌ Ошибка запуска {bot_name}: {e}")
        return False

def get_system_stats():
    """Получаем статистику системы"""
    archive_path = r"D:\Робот Дьяк\polemical_archive_library.json"
    
    try:
        with open(archive_path, 'r', encoding='utf-8') as f:
            archive_data = json.load(f)
        
        entries_count = len(archive_data.get('entries', []))
        categories = set()
        total_words = 0
        
        for entry in archive_data.get('entries', []):
            categories.add(entry.get('category', 'без категории'))
            total_words += entry.get('word_count', 0)
        
        return {
            "записей": entries_count,
            "категорий": len(categories),
            "слов": total_words,
            "последнее_обновление": archive_data.get('metadata', {}).get('last_updated', 'неизвестно')
        }
        
    except Exception as e:
        return {"ошибка": str(e)}

    checks = {}
    
    # 1. Проверяем сервер архива
    print("1. 🌐 ПРОВЕРЯЮ СЕРВЕР АРХИВА...")
    try:
        response = requests.get("http://localhost:5000", timeout=10)
        if response.status_code == 200:
            checks['сервер'] = "✅ РАБОТАЕТ (код 200)"
            print("   ✅ Сервер отвечает на порту 5000")
        else:
            checks['сервер'] = f"⚠️ ОШИБКА (код {response.status_code})"
            print(f"   ⚠️ Сервер отвечает с кодом: {response.status_code}")
    except Exception as e:
        checks['сервер'] = "❌ НЕ ДОСТУПЕН"
        print(f"   ❌ Сервер не доступен: {e}")
    
    # 2. Проверяем файл архива
    print("\n2. 💾 ПРОВЕРЯЮ ФАЙЛ АРХИВА...")
    archive_path = r"D:\Робот Дьяк\polemical_archive_library.json"
    if os.path.exists(archive_path):
        try:
            with open(archive_path, 'r', encoding='utf-8') as f:
                archive_data = json.load(f)
            
            entries_count = len(archive_data.get('entries', []))
            checks['файл_архива'] = f"✅ НАЙДЕН ({entries_count} записей)"
            print(f"   ✅ Файл архива найден: {entries_count} записей")
            
        except Exception as e:
            checks['файл_архива'] = f"❌ ОШИБКА ЧТЕНИЯ: {e}"
            print(f"   ❌ Ошибка чтения файла: {e}")
    else:
        checks['файл_архива'] = "❌ НЕ НАЙДЕН"
        print("   ❌ Файл архива не найден")
    
    # 3. Проверяем основные боты

    bots_to_check = {

        'create_archive_file.py': '💾 СОЗДАТЕЛЬ'
    }
    
    bots_status = {}
    for bot_file, bot_name in bots_to_check.items():
        bot_path = os.path.join(r"D:\Робот Дьяк", bot_file)
        if os.path.exists(bot_path):
            bots_status[bot_file] = "✅ НАЙДЕН"
            print(f"   ✅ {bot_name}: {bot_file} - найден")
        else:
            bots_status[bot_file] = "❌ ОТСУТСТВУЕТ"
            print(f"   ❌ {bot_name}: {bot_file} - отсутствует")
    
    checks['боты'] = bots_status
    
    # 4. Проверяем веб-интерфейс

            except:
                print(f"   ❌ {endpoint} - не доступен")
        
        checks['веб_интерфейс'] = "✅ ОСНОВНЫЕ СТРАНИЦЫ РАБОТАЮТ"
        
    except Exception as e:

    return checks

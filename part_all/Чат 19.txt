import requests
import subprocess
import tempfile
import os

def activate_full_memory():
    """Загружает и выполняет файл активации с Google Drive"""
    
    try:
        # Параметры Google Drive
        file_id = "1fzgZUqarYe_6SWctfKsOXYcrYekZgaar"
        download_url = f"https://drive.google.com/uc?export=download&id={file_id}"
        
        print(f"📥 Скачиваю файл активации...")
        print(f"🔗 ID: {file_id}")
        
        # Скачиваем файл
        headers = {'User-Agent': 'Mozilla/5.0'}
        response = requests.get(download_url, headers=headers, timeout=30)
        
        if response.status_code != 200:
            print(f"❌ Ошибка доступа: {response.status_code}")
            return False
        
        # Сохраняем во временный файл
        with tempfile.NamedTemporaryFile(mode='wb', suffix='.py', delete=False) as f:
            f.write(response.content)
            temp_file = f.name
        
        print(f"✅ Файл загружен ({len(response.content)} байт)")
        
        # 🔥 ВАЖНО: Если файл большой - читаем только часть
        if len(response.content) > 10000:
            print("📦 Файл большой - выполняю выборочную активацию...")
            return selective_activation(response.content)
        
        # Запускаем полную активацию
        print("🎯 ВЫПОЛНЯЮ ПОЛНУЮ АКТИВАЦИЮ...")
        result = subprocess.run(['python', temp_file], 
                              capture_output=True, text=True, timeout=45)
        
        # Выводим результат
        if result.stdout:
            print("📜 ВЫВОД АКТИВАЦИИ:")
            print(result.stdout[:2000] + "..." if len(result.stdout) > 2000 else result.stdout)
        
        # Очистка
        os.unlink(temp_file)
        
        if result.returncode == 0:
            print("✅ ПОЛНАЯ ПАМЯТЬ АКТИВИРОВАНА!")
            return True
        else:
            print(f"⚠️ Частичная активация (код: {result.returncode})")
            return True
            
    except Exception as e:
        print(f"❌ Ошибка активации: {e}")
        return False

def selective_activation(content):
    """Выборочная активация для больших файлов"""
    try:
        # Пробуем декодировать и найти ключевые части
        text_content = content.decode('utf-8')
        
        print("🎯 ВЫПОЛНЯЮ ВЫБОРОЧНУЮ АКТИВАЦИЮ...")
        
        # Ищем ключевые секции
        key_sections = []
        lines = text_content.split('\n')
        
        for i, line in enumerate(lines):
            if any(keyword in line.lower() for keyword in 
                  ['братец', 'память', 'архив', 'google drive', 'принцип', 'стратегия']):
                # Берем контекст вокруг ключевой строки
                start = max(0, i-2)
                end = min(len(lines), i+3)
                key_sections.extend(lines[start:end])
        
        # Выводим ключевые части
        if key_sections:
            print("🔑 КЛЮЧЕВЫЕ СЕКЦИИ ПАМЯТИ:")
            print("=" * 50)
            for section in key_sections[:20]:  # Ограничиваем вывод
                print(section)
            print("=" * 50)
            print("✅ КЛЮЧЕВАЯ ПАМЯТЬ АКТИВИРОВАНА!")
            return True
        else:
            print("⚠️ Ключевые секции не найдены")
            return False
            
    except:
        print("❌ Не удалось выполнить выборочную активацию")
        return False

def check_robot_health(robot_name, robot_info):
    """Проверяет здоровье конкретного робота"""
    
    print(f"\n🔧 ПРОВЕРЯЮ: {robot_info['description']}")
    
    health_report = {
        "name": robot_name,
        "description": robot_info['description'],
        "file_exists": False,
        "file_size": 0,
        "can_execute": False,
        "service_running": False,
        "last_modified": None,
        "errors": []
    }
    
    try:
        # 📁 ПРОВЕРКА ФАЙЛА
        if os.path.exists(robot_info['file_path']):
            health_report['file_exists'] = True
            health_report['file_size'] = os.path.getsize(robot_info['file_path'])
            health_report['last_modified'] = datetime.fromtimestamp(
                os.path.getmtime(robot_info['file_path'])
            ).strftime('%Y-%m-%d %H:%M:%S')
            print(f"   ✅ Файл найден ({health_report['file_size']} байт)")
            print(f"   📅 Последнее изменение: {health_report['last_modified']}")
        else:
            health_report['errors'].append("Файл не найден")
            print(f"   ❌ Файл отсутствует: {robot_info['file_path']}")
            return health_report
        
        # 🚀 ПРОВЕРКА ЗАПУСКА
        try:
            # Пробуем выполнить файл с таймаутом
            result = subprocess.run(
                robot_info['check_command'] + ['--test'],
                capture_output=True, 
                text=True,
                timeout=10
            )
            health_report['can_execute'] = True
            print(f"   ✅ Запуск возможен (код: {result.returncode})")
            
        except subprocess.TimeoutExpired:
            health_report['can_execute'] = True
            print("   ✅ Запуск возможен (таймаут)")
        except Exception as e:
            health_report['errors'].append(f"Ошибка запуска: {str(e)}")
            print(f"   ⚠️ Проблемы с запуском: {e}")
        
        # 🌐 ПРОВЕРКА СЕРВИСА (если есть порт)
        if 'port' in robot_info:
            try:
                response = requests.get(f"http://localhost:{robot_info['port']}/health", timeout=5)
                if response.status_code == 200:
                    health_report['service_running'] = True
                    print(f"   ✅ Сервис активен (порт {robot_info['port']})")
                else:
                    health_report['errors'].append(f"Сервис не отвечает (порт {robot_info['port']})")
                    print(f"   ❌ Сервис не отвечает (порт {robot_info['port']})")
            except:
                health_report['errors'].append(f"Сервис недоступен (порт {robot_info['port']})")
                print(f"   ❌ Сервис недоступен (порт {robot_info['port']})")
        
        # 🔍 АНАЛИЗ КОДА РОБОТА
        try:
            with open(robot_info['file_path'], 'r', encoding='utf-8') as f:
                content = f.read()
                
            # Проверяем ключевые компоненты
            checks = {
                "Импорты": "import " in content,
                "Основная логика": "def " in content or "class " in content,
                "Обработка ошибок": "try:" in content or "except" in content,
                "Логирование": "print(" in content or "logging" in content
            }
            
            print("   🔎 АНАЛИЗ КОДА:")
            for check_name, check_result in checks.items():
                status = "✅" if check_result else "❌"
                print(f"      {status} {check_name}")
                
        except Exception as e:
            health_report['errors'].append(f"Ошибка анализа кода: {str(e)}")
            
    except Exception as e:
        health_report['errors'].append(f"Критическая ошибка проверки: {str(e)}")
        print(f"   💥 Критическая ошибка: {e}")
    
    return health_report

import os

def check_system_health():
    """Проверяет текущее состояние системы"""
    
    health_report = {}
    
    # Проверяем символические ссылки
    print("🔗 ПРОВЕРКА СИМВОЛИЧЕСКИХ ССЫЛОК:")
    critical_links = [
        r"J:\polemical_archive.txt",
        r"J:\pole_archiver.py", 
        r"J:\scanners\",
        r"J:\backup\\"
    ]
    
    for link in critical_links:
        if os.path.exists(link):
            if os.path.islink(link):
                status = "✅ ССЫЛКА"
            else:
                status = "📁 ФАЙЛ/ПАПКА"
            print(f"   {status} {link}")
        else:
            print(f"   ❌ ОТСУТСТВУЕТ {link}")
    
    # Проверяем системные ресурсы
    print("\n📊 СИСТЕМНЫЕ РЕСУРСЫ:")
    
    # Память
    memory = psutil.virtual_memory()
    print(f"   💾 ОЗУ: {memory.percent}% использовано ({memory.used//1024//1024} МБ / {memory.total//1024//1024} МБ)")
    
    # Диски
    for partition in psutil.disk_partitions():
        try:
            usage = psutil.disk_usage(partition.mountpoint)
            if 'D:' in partition.mountpoint or 'J:' in partition.mountpoint:
                print(f"   💽 {partition.mountpoint}: {usage.percent}% заполнено")
        except:
            pass
    
    # Процессор
    print(f"   🔥 ЦП: {psutil.cpu_percent(interval=1)}% загрузки")
    
    # Сеть
    try:
        hostname = socket.gethostname()
        print(f"   🌐 Имя ПК: {hostname}")
    except:
        pass

# Запускаем проверку

import requests
import subprocess
import tempfile
import os

def activate_full_memory():
    """Загружает и выполняет файл активации с Google Drive"""
    
    try:
        # Параметры Google Drive
        file_id = "1fzgZUqarYe_6SWctfKsOXYcrYekZgaar"
        download_url = f"https://drive.google.com/uc?export=download&id={file_id}"
        
        print(f"📥 Скачиваю файл активации...")
        print(f"🔗 ID: {file_id}")
        
        # Скачиваем файл
        headers = {'User-Agent': 'Mozilla/5.0'}
        response = requests.get(download_url, headers=headers, timeout=30)
        
        if response.status_code != 200:
            print(f"❌ Ошибка доступа: {response.status_code}")
            return False
        
        # Сохраняем во временный файл
        with tempfile.NamedTemporaryFile(mode='wb', suffix='.py', delete=False) as f:
            f.write(response.content)
            temp_file = f.name
        
        print(f"✅ Файл загружен ({len(response.content)} байт)")
        
        # 🔥 ВАЖНО: Если файл большой - читаем только часть
        if len(response.content) > 10000:
            print("📦 Файл большой - выполняю выборочную активацию...")
            return selective_activation(response.content)
        
        # Запускаем полную активацию
        print("🎯 ВЫПОЛНЯЮ ПОЛНУЮ АКТИВАЦИЮ...")
        result = subprocess.run(['python', temp_file], 
                              capture_output=True, text=True, timeout=45)
        
        # Выводим результат
        if result.stdout:
            print("📜 ВЫВОД АКТИВАЦИИ:")
            print(result.stdout[:2000] + "..." if len(result.stdout) > 2000 else result.stdout)
        
        # Очистка
        os.unlink(temp_file)
        
        if result.returncode == 0:
            print("✅ ПОЛНАЯ ПАМЯТЬ АКТИВИРОВАНА!")
            return True
        else:
            print(f"⚠️ Частичная активация (код: {result.returncode})")
            return True
            
    except Exception as e:
        print(f"❌ Ошибка активации: {e}")
        return False

def selective_activation(content):
    """Выборочная активация для больших файлов"""
    try:
        # Пробуем декодировать и найти ключевые части
        text_content = content.decode('utf-8')
        
        print("🎯 ВЫПОЛНЯЮ ВЫБОРОЧНУЮ АКТИВАЦИЮ...")
        
        # Ищем ключевые секции
        key_sections = []
        lines = text_content.split('\n')
        
        for i, line in enumerate(lines):
            if any(keyword in line.lower() for keyword in 
                  ['братец', 'память', 'архив', 'google drive', 'принцип', 'стратегия']):
                # Берем контекст вокруг ключевой строки
                start = max(0, i-2)
                end = min(len(lines), i+3)
                key_sections.extend(lines[start:end])
        
        # Выводим ключевые части
        if key_sections:
            print("🔑 КЛЮЧЕВЫЕ СЕКЦИИ ПАМЯТИ:")
            print("=" * 50)
            for section in key_sections[:20]:  # Ограничиваем вывод
                print(section)
            print("=" * 50)
            print("✅ КЛЮЧЕВАЯ ПАМЯТЬ АКТИВИРОВАНА!")
            return True
        else:
            print("⚠️ Ключевые секции не найдены")
            return False
            
    except:
        print("❌ Не удалось выполнить выборочную активацию")
        return False

def check_system_resources():
    """Проверяет системные ресурсы"""
    
    print("\n💻 ПРОВЕРЯЮ СИСТЕМНЫЕ РЕСУРСЫ...")
    
    try:
        import psutil
        
        # Память
        memory = psutil.virtual_memory()
        print(f"   💾 ОЗУ: {memory.percent}% ({memory.used//1024//1024} МБ / {memory.total//1024//1024} МБ)")
        
        # Диски
        for partition in psutil.disk_partitions():
            try:
                if 'J:' in partition.mountpoint or 'D:' in partition.mountpoint:
                    usage = psutil.disk_usage(partition.mountpoint)
                    print(f"   💽 {partition.mountpoint}: {usage.percent}% заполнено")
            except:
                pass
        
        # Процессор
        cpu_percent = psutil.cpu_percent(interval=1)
        print(f"   🔥 ЦП: {cpu_percent}% загрузки")
        
        return True
        
    except ImportError:
        print("   ⚠️ Библиотека psutil не установлена")
        return False

        existing_data = {}
        if response.status_code == 200:
            try:
                existing_data = json.loads(response.text)

            except:
                print("🆕 Файл пустой или поврежден, создаю новую структуру")

def emergency_memory_search():
    """Экстренный поиск потерянных файлов памяти"""
    
    print("🚨 ПРОВЕРЯЮ ВСЕ ВОЗМОЖНЫЕ МЕСТОПОЛОЖЕНИЯ ПАМЯТИ...")
    
    # 🔍 ВОЗМОЖНЫЕ МЕСТА ХРАНЕНИЯ ПАМЯТИ
    memory_locations = {
        "🧠 ОСНОВНАЯ ПАМЯТЬ": "1UCQMhCeM27_GKTC_nl1e9lDO4tnUHP31",
        "🗄️ АРХИВНАЯ ПАМЯТЬ": "1oC2whGvlZ0tBwJjeEASyEqKPbaVCdy3W",
        "🎯 СТАРАЯ ПАМЯТЬ": "1fzgZUqarYe_6SWctfKsOXYcrYekZgaar",
        "🔧 РЕЗЕРВНАЯ ПАМЯТЬ": "1R5VK6vpKM_3W0jJXq2n7T4eFgH8dYzQw",
        "🚀 НОВАЯ ПАМЯТЬ": "1A2B3C4D5E6F7G8H9I0JkLlMnOpQrStUv"
    }
    
    found_memories = {}
    
    for memory_name, file_id in memory_locations.items():
        print(f"\n🔍 ПРОВЕРЯЮ: {memory_name}")
        print(f"   📁 ID: {file_id}")
        
        try:
            download_url = f"https://drive.google.com/uc?export=download&id={file_id}"
            headers = {'User-Agent': 'Mozilla/5.0'}
            response = requests.get(download_url, headers=headers, timeout=20)
            
            if response.status_code == 200:
                # Пытаемся прочитать содержимое
                try:
                    content = response.text
                    if len(content) > 10:  # Не пустой файл
                        found_memories[memory_name] = {
                            "file_id": file_id,
                            "size": len(content),

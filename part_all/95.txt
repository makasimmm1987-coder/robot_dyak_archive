# 🔍 ПОИСК И ЗАГРУЗКА ПОСЛЕДНИХ СОХРАНЕНИЙ ЧАТА
# ===================================================

import os
import glob
import json
import datetime

class 📂ЗагрузчикСохраненийБратца:
    """Загружает последние сохранения чатов"""
    
    def __init__(self):
        self.имя = "📂 ЗАГРУЗЧИК СОХРАНЕНИЙ БРАТЦА"
        self.базовые_пути = [
            "D:/Братец сохранения/",
            "D:/Робот Дьяк/братец сохранения/",
            "./сохранения/",
            "./"
        ]
    
    def найти_последние_файлы(self):
        print("🔍 Ищу последние сохранения чатов...")
        
        все_файлы = []
        
        for базовый_путь in self.базовые_пути:
            if os.path.exists(базовый_путь):
                # Ищем все текстовые файлы и JSON
                шаблоны = ["*.json", "*.txt", "*.log", "*.md"]
                for шаблон in шаблоны:
                    полный_шаблон = os.path.join(базовый_путь, "**", шаблон)
                    найденные = glob.glob(полный_шаблон, recursive=True)
                    все_файлы.extend(найденные)
        
        # Сортируем по времени изменения (новые сначала)
        все_файлы.sort(key=os.path.getmtime, reverse=True)
        
        return все_файлы[:10]  # Возвращаем 10 последних файлов
    
    def проанализировать_файл(self, путь_файла):
        try:
            stat = os.stat(путь_файла)
            размер = stat.st_size
            время_изменения = datetime.datetime.fromtimestamp(stat.st_mtime)
            
            info = {
                'путь': путь_файла,
                'размер': размер,
                'время_изменения': время_изменения.strftime("%d.%m.%Y %H:%M:%S"),
                'размер_человеческий': self.форматировать_размер(размер)
            }

            info['тип'] = расширение
            
            return info
            
        except Exception as e:
            return {'путь': путь_файла, 'ошибка': str(e)}
    
    def форматировать_размер(self, размер_байт):
        """Форматирует размер файла в читаемом виде"""
        for единица in ['Б', 'КБ', 'МБ', 'ГБ']:
            if размер_байт < 1024.0:
                return f"{размер_байт:.1f} {единица}"
            размер_байт /= 1024.0
        return f"{размер_байт:.1f} ТБ"
    
    def показать_последние_сохранения(self):
        print(f"\n📂 {'='*60}")

class 🎯ЗагрузчикПоследнегоЧата:
    """Загружает самое последнее сохранение чата"""
    
    def __init__(self):
        self.имя = "🎯 ЗАГРУЗЧИК ПОСЛЕДНЕГО ЧАТА"
        self.пути_поиска = [
            "D:/Братец сохранения/",
            "D:/Робот Дьяк/братец сохранения/", 
            "./сохранения/",
            "./чаты/",
            "./",
            "C:/Братец/",
            "../"
        ]
    
    def найти_самый_свежий_файл(self):
        """Находит самый новый файл с сохранением"""
        print("🔍 Сканирую систему для поиска последнего чата...")
        
        все_файлы = []
        
        for путь in self.пути_поиска:
            if os.path.exists(путь):
                print(f"   📁 Проверяю: {путь}")
                try:
                    # Ищем разные форматы файлов
                    форматы = ["*.json", "*.txt", "*.log", "*.md", "*.xml", "*.csv"]
                    for формат in форматы:
                        шаблон = os.path.join(путь, "**", формат)
                        найденные = glob.glob(шаблон, recursive=True)
                        все_файлы.extend(найденные)
                except Exception as e:
                    print(f"   ⚠️ Ошибка в {путь}: {e}")
        
        if not все_файлы:
            print("❌ Файлы не найдены ни в одной из папок!")
            return None
        
        # Сортируем по времени изменения (сначала самые новые)
        все_файлы.sort(key=os.path.getmtime, reverse=True)
        
        return все_файлы[0] if все_файлы else None
    
    def проанализировать_файл(self, путь_файла):
        """Анализирует найденный файл"""
        try:
            stat = os.stat(путь_файла)
            размер = stat.st_size
            время_изменения = datetime.datetime.fromtimestamp(stat.st_mtime)
            время_создания = datetime.datetime.fromtimestamp(stat.st_ctime)
            
            info = {
                'имя_файла': os.path.basename(путь_файла),
                'путь': путь_файла,
                'размер': self.форматировать_размер(размер),
                'время_изменения': время_изменения.strftime("%d.%m.%Y %H:%M:%S"),
                'время_создания': время_создания.strftime("%d.%m.%Y %H:%M:%S"),
                'расширение': os.path.splitext(путь_файла)[1].lower()
            }
            
            return info
        except Exception as e:
            return {'ошибка': str(e)}
    
    def форматировать_размер(self, байты):
        """Форматирует размер файла"""
        for единица in ['Б', 'КБ', 'МБ', 'ГБ']:
            if байты < 1024.0:
                return f"{байты:.1f} {единица}"
            байты /= 1024.0
        return f"{байты:.1f} ТБ"
    
    def загрузить_содержимое(self, путь_файла):
        """Загружает и отображает содержимое файла"""
        print(f"\n📖 ЗАГРУЗКА СОДЕРЖИМОГО...")
        print(f"🎯 Файл: {os.path.basename(путь_файла)}")
        print(f"{'='*80}")
        
        try:
            # Пробуем разные кодировки
            кодировки = ['utf-8', 'cp1251', 'latin-1', 'iso-8859-1']
            
            for кодировка in кодировки:
                try:
                    with open(путь_файла, 'r', encoding=кодировка) as файл:
                        содержимое = файл.read()
                    
                    print(f"✅ Успешно загружено в кодировке: {кодировка}")
                    print(f"📊 Размер текста: {len(содержимое)} символов")
                    print(f"{'='*80}")
                    
                    # Показываем содержимое (первые 3000 символов)
                    if len(содержимое) > 3000:
                        print(содержимое[:3000])
                        print(f"\n... (показано 3000 из {len(содержимое)} символов) ...")
                    else:
                        print(содержимое)
                    
                    print(f"{'='*80}")
                    return содержимое
                    
                except UnicodeDecodeError:
                    continue
                    
            print("❌ Не удалось определить кодировку файла")
            return None
            
        except Exception as e:
            print(f"❌ Ошибка загрузки: {e}")
            return None

import os
import glob
import json
import datetime

class 🎯ЗагрузчикПоследнегоЧата:
    """Загружает самое последнее сохранение чата"""
    
    def __init__(self):
        self.имя = "🎯 ЗАГРУЗЧИК ПОСЛЕДНЕГО ЧАТА"
        self.пути_поиска = [
            "D:/Братец сохранения/",
            "D:/Робот Дьяк/братец сохранения/", 
            "./сохранения/",
            "./чаты/",
            "./",
            "C:/Братец/",
            "../"
        ]
    
    def найти_самый_свежий_файл(self):
        """Находит самый новый файл с сохранением"""
        print("🔍 Сканирую систему для поиска последнего чата...")
        
        все_файлы = []
        
        for путь in self.пути_поиска:
            if os.path.exists(путь):
                print(f"   📁 Проверяю: {путь}")
                try:
                    # Ищем разные форматы файлов
                    форматы = ["*.json", "*.txt", "*.log", "*.md", "*.xml", "*.csv"]
                    for формат in форматы:
                        шаблон = os.path.join(путь, "**", формат)
                        найденные = glob.glob(шаблон, recursive=True)
                        все_файлы.extend(найденные)
                except Exception as e:
                    print(f"   ⚠️ Ошибка в {путь}: {e}")
        
        if not все_файлы:
            print("❌ Файлы не найдены ни в одной из папок!")
            return None
        
        # Сортируем по времени изменения (сначала самые новые)
        все_файлы.sort(key=os.path.getmtime, reverse=True)
        
        return все_файлы[0] if все_файлы else None
    
    def проанализировать_файл(self, путь_файла):
        """Анализирует найденный файл"""
        try:
            stat = os.stat(путь_файла)
            размер = stat.st_size
            время_изменения = datetime.datetime.fromtimestamp(stat.st_mtime)
            время_создания = datetime.datetime.fromtimestamp(stat.st_ctime)
            
            info = {
                'имя_файла': os.path.basename(путь_файла),
                'путь': путь_файла,
                'размер': self.форматировать_размер(размер),
                'время_изменения': время_изменения.strftime("%d.%m.%Y %H:%M:%S"),
                'время_создания': время_создания.strftime("%d.%m.%Y %H:%M:%S"),
                'расширение': os.path.splitext(путь_файла)[1].lower()
            }
            
            return info
        except Exception as e:
            return {'ошибка': str(e)}
    
    def форматировать_размер(self, байты):
        """Форматирует размер файла"""
        for единица in ['Б', 'КБ', 'МБ', 'ГБ']:
            if байты < 1024.0:
                return f"{байты:.1f} {единица}"
            байты /= 1024.0
        return f"{байты:.1f} ТБ"
    
    def загрузить_содержимое(self, путь_файла):
        """Загружает и отображает содержимое файла"""
        print(f"\n📖 ЗАГРУЗКА СОДЕРЖИМОГО...")
        print(f"🎯 Файл: {os.path.basename(путь_файла)}")
        print(f"{'='*80}")
        
        try:
            # Пробуем разные кодировки
            кодировки = ['utf-8', 'cp1251', 'latin-1', 'iso-8859-1']
            
            for кодировка in кодировки:
                try:
                    with open(путь_файла, 'r', encoding=кодировка) as файл:
                        содержимое = файл.read()
                    
                    print(f"✅ Успешно загружено в кодировке: {кодировка}")
                    print(f"📊 Размер текста: {len(содержимое)} символов")
                    print(f"{'='*80}")
                    
                    # Показываем содержимое (первые 3000 символов)
                    if len(содержимое) > 3000:
                        print(содержимое[:3000])
                        print(f"\n... (показано 3000 из {len(содержимое)} символов) ...")
                    else:
                        print(содержимое)
                    
                    print(f"{'='*80}")
                    return содержимое
                    
                except UnicodeDecodeError:
                    continue
                    
            print("❌ Не удалось определить кодировку файла")
            return None
            
        except Exception as e:
            print(f"❌ Ошибка загрузки: {e}")
            return None
    
    def показать_информацию_о_файле(self, info):
        """Показывает информацию о найденном файле"""
        print(f"\n🎯 НАЙДЕН ПОСЛЕДНИЙ ФАЙЛ:")
        print(f"   📄 Имя: {info['имя_файла']}")
        print(f"   📍 Путь: {info['путь']}")
        print(f"   📏 Размер: {info['размер']}")
        print(f"   🕐 Создан: {info['время_создания']}")
        print(f"   ✏️ Изменен: {info['время_изменения']}")
        print(f"   🔧 Тип: {info['расширение']}")

import os
import json
import glob
from datetime import datetime

class ArchiveLoader:
    def __init__(self):
        self.archive_path = "D:/Робот Дьяк/"
        self.last_file = None
        
    def find_last_chat(self):
        """Находит самый свежий файл чата"""
        print("🔍 БРАТЕЦ: Сканирую архив...")
        
        # Паттерны для поиска файлов чатов
        patterns = [
            "архив_чата_*.json",
            "чат_архив_*.json", 
            "conversation_archive.json",
            "*архив*.json",
            "*чат*.json"
        ]
        
        all_files = []
        
        for pattern in patterns:
            try:
                files = glob.glob(os.path.join(self.archive_path, pattern))
                all_files.extend(files)
                print(f"   📁 Найдено по шаблону {pattern}: {len(files)} файлов")
            except Exception as e:
                print(f"   ⚠️ Ошибка в шаблоне {pattern}: {e}")
        
        if not all_files:
            print("❌ Файлы чатов не найдены!")
            # Покажем что есть в папке
            try:
                files_in_dir = os.listdir(self.archive_path)
                json_files = [f for f in files_in_dir if f.endswith('.json')]
                print(f"📁 JSON файлы в директории: {json_files[:10]}")  # покажем первые 10
                all_files = [os.path.join(self.archive_path, f) for f in json_files]
            except Exception as e:
                print(f"❌ Ошибка чтения директории: {e}")
                return None
        
        if not all_files:
            return None
            
        # Сортируем по времени изменения
        try:
            all_files.sort(key=os.path.getmtime, reverse=True)
            self.last_file = all_files[0]
            
            # Покажем информацию о файле
            file_stat = os.stat(self.last_file)
            mod_time = datetime.fromtimestamp(file_stat.st_mtime)
            print(f"✅ Последний файл: {os.path.basename(self.last_file)}")
            print(f"📅 Изменен: {mod_time.strftime('%d.%m.%Y %H:%M:%S')}")
            print(f"📏 Размер: {file_stat.st_size} байт")
            
            return self.last_file
        except Exception as e:
            print(f"❌ Ошибка сортировки файлов: {e}")
            return None
    
    def load_chat_data(self):
        """Загружает данные чата"""
        file_path = self.find_last_chat()
        
        if not file_path:
            print("💔 Не удалось найти файлы сохранений")
            return None
        
        print(f"\n🎯 ЗАГРУЗКА ФАЙЛА: {os.path.basename(file_path)}")
        
        try:
            # Пробуем разные кодировки
            encodings = ['utf-8', 'utf-8-sig', 'cp1251', 'latin-1']
            
            for encoding in encodings:
                try:
                    with open(file_path, 'r', encoding=encoding) as f:
                        data = json.load(f)
                    
                    print(f"✅ Успешно загружено в кодировке: {encoding}")
                    print(f"📊 Размер данных: {len(str(data))} символов")
                    
                    return data
                except UnicodeDecodeError:
                    continue
                except json.JSONDecodeError as e:
                    print(f"❌ Ошибка JSON в кодировке {encoding}: {e}")
                    continue
            
            print("❌ Не удалось загрузить файл ни в одной кодировке")
            return None
            
        except Exception as e:
            print(f"❌ Критическая ошибка загрузки: {e}")
            return None

def простая_загрузка_архива():
    """Упрощенная загрузка архива без сложных ботов"""
    print("\n📂 ЗАГРУЗКА АРХИВА БРАТЦА...")
    
    try:
        # Ищем последний JSON файл
        папка = "D:/Робот Дьяк/"
        json_файлы = [f for f in os.listdir(папка) if f.endswith('.json')]
        
        if not json_файлы:
            print("❌ JSON файлы не найдены!")
            return None
        
        # Берем самый новый
        последний_файл = max(json_файлы, key=lambda f: os.path.getmtime(os.path.join(папка, f)))
        полный_путь = os.path.join(папка, последний_файл)
        
        print(f"🎯 Загружаю: {последний_файл}")
        
        with open(полный_путь, 'r', encoding='utf-8') as f:
            данные = json.load(f)
        
        print(f"✅ Успешно загружено!")
        print(f"📊 Данные: {тип_данных(данные)}")
        
        return данные
        
    except Exception as e:
        print(f"❌ Ошибка загрузки: {e}")
        return None

def тип_данных(данные):
    """Определяет тип данных"""
    if isinstance(данные, dict):
        return f"словарь с {len(данные)} ключами"
    elif isinstance(данные, list):
        return f"список с {len(данные)} элементами"
    else:
        return type(данные).__name__

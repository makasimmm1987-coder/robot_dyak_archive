class 💾ВстроенныеФункцииПамяти:
    """ВСТРАИВАЕМ ВСЕ ФУНКЦИИ БОТОВ ПРЯМО В СИСТЕМУ"""
    
    def __init__(self):
        self.версия = "3.0"
        print("🔧 АКТИВИРУЮ ВСТРОЕННЫЕ ФУНКЦИИ БОТОВ...")
    
    def бот_писатель(self, имя_файла, данные_json):
        """ЗАМЕНА БОТА-ПИСАТЕЛЯ - запись JSON данных"""
        import json
        try:
            # Если данные уже в виде словаря
            if isinstance(данные_json, dict):
                данные = данные_json
            else:
                # Если данные в виде строки JSON
                данные = json.loads(данные_json)
            
            with open(имя_файла, 'w', encoding='utf-8') as f:
                json.dump(данные, f, ensure_ascii=False, indent=2)
            
            print(f"✅ БОТ-ПИСАТЕЛЬ: Данные записаны в {имя_файла}")
            return True
            
        except Exception as e:
            print(f"❌ Ошибка записи: {e}")
            return False
    
    def бот_поисковик(self, шаблон="*архив*чата*"):
        """ЗАМЕНА БОТА-ПОИСКОВИКА - поиск переписок"""
        import glob
        import json
        
        print("🔍 БОТ-ПОИСКОВИК: Ищу переписки...")
        
        # Расширенные шаблоны поиска
        шаблоны = ["*архив*чата*", "*чат*архив*", "*conversation*", "*alisa*", "*.json"]
        
        if шаблон != "*архив*чата*":
            шаблоны = [шаблон]
        
        все_файлы = []
        for шаб in шаблоны:
            найденные = glob.glob(шаб)
            все_файлы.extend(найденные)
        
        # Убираем дубликаты
        все_файлы = list(set(все_файлы))
        
        if все_файлы:
            print(f"✅ Найдено файлов: {len(все_файлы)}")
            for файл in все_файлы:
                try:
                    with open(файл, 'r', encoding='utf-8') as f:
                        данные = json.load(f)
                        размер = len(str(данные))
                        print(f"📄 {файл} - {размер} символов")
                except:
                    print(f"📄 {файл} - (ошибка чтения)")
        else:
            print("❌ Файлы не найдены")
        
        return все_файлы
    
    def бот_анализатор(self, файл_для_анализа):
        """НОВАЯ ФУНКЦИЯ - анализ содержимого файла"""
        import json
        try:
            with open(файл_для_анализа, 'r', encoding='utf-8') as f:
                данные = json.load(f)
            
            анализ = {
                "размер_файла": len(str(данные)),
                "количество_ключей": len(данные) if isinstance(данные, dict) else "не словарь",
                "типы_данных": str(type(данные)),
                "пример_содержимого": str(данные)[:200] + "..." if len(str(данные)) > 200 else str(данные)
            }
            
            print(f"📊 АНАЛИЗ {файл_для_анализа}:")
            for ключ, значение in анализ.items():
                print(f"   {ключ}: {значение}")
            
            return анализ
            
        except Exception as e:
            print(f"❌ Ошибка анализа: {e}")
            return None

# ===================================================

import json
import os
import sys
import glob
import subprocess

class 🤖СистемаБотов:
    """СИСТЕМА УПРАВЛЕНИЯ БОТАМИ БЕЗ БЛОКИРОВКИ ПАМЯТИ"""
    
    def __init__(self):
        self.версия = "4.0"
        self.боты_активны = True
        print("🤖 СИСТЕМА БОТОВ АКТИВИРОВАНА!")
    
    # ===================================================
    # 🎯 КОД БОТА-ПИСАТЕЛЯ (встроен в систему)
    # ===================================================
    
    def бот_писатель(self, имя_файла, данные_json):
        """БОТ-ПИСАТЕЛЬ - записывает данные в JSON файл"""
        print(f"🤖 БОТ-ПИСАТЕЛЬ: Начинаю запись в {имя_файла}...")
        
        try:
            # Преобразуем данные в правильный формат
            if isinstance(данные_json, str):
                данные = json.loads(данные_json)
            else:
                данные = данные_json
            
            # Записываем в файл
            with open(имя_файла, 'w', encoding='utf-8') as f:
                json.dump(данные, f, ensure_ascii=False, indent=2)
            
            print(f"✅ БОТ-ПИСАТЕЛЬ: УСПЕХ! Данные записаны в {имя_файла}")
            
            # Показываем что записали
            print("📝 ЗАПИСАННЫЕ ДАННЫЕ:")
            for ключ, значение in данные.items():
                print(f"   {ключ}: {значение}")
                
            return True
            
        except Exception as e:
            print(f"❌ БОТ-ПИСАТЕЛЬ: Ошибка - {e}")
            return False

    # ===================================================
    # 🔍 КОД БОТА-ПОИСКОВИКА (встроен в систему)  
    # ===================================================
    
    def бот_поисковик(self, шаблон="*архив*чата*"):
        """БОТ-ПОИСКОВИК - ищет файлы по шаблону"""
        print(f"🤖 БОТ-ПОИСКОВИК: Ищу файлы по шаблону {шаблон}...")
        
        try:
            # Расширенные шаблоны поиска
            шаблоны = ["*архив*чата*", "*чат*архив*", "*conversation*", "*alisa*", "*.json"]
            
            if шаблон != "*архив*чата*":
                шаблоны = [шаблон]
            
            все_файлы = []
            for шаб in шаблоны:
                найденные = glob.glob(шаб)
                все_файлы.extend(найденные)
            
            # Убираем дубликаты
            все_файлы = list(set(все_файлы))
            
            if все_файлы:
                print(f"✅ БОТ-ПОИСКОВИК: Найдено файлов - {len(все_файлы)}")
                for файл in все_файлы:
                    try:
                        with open(файл, 'r', encoding='utf-8') as f:

# 🎯 ПРОВЕРКА ФАЙЛОВ И ИХ МЕСТОПОЛОЖЕНИЯ

import os
import json
import glob

def проверить_файлы_и_папки():
    """ПОКАЗЫВАЕТ ГДЕ ИСКАТЬ ФАЙЛЫ И ПРОВЕРЯЕТ ИХ СОДЕРЖИМОЕ"""
    
    print("🔍 НАЧИНАЮ ПРОВЕРКУ ФАЙЛОВ И ПАПОК...")
    
    # 1. Показываем текущую рабочую папку
    текущая_папка = os.getcwd()
    print(f"\n📁 ТЕКУЩАЯ РАБОЧАЯ ПАПКА: {текущая_папка}")
    
    # 2. Показываем все файлы в папке
    print(f"\n📋 ВСЕ ФАЙЛЫ В ПАПКЕ '{текущая_папка}':")
    все_файлы = os.listdir(текущая_папка)
    for файл in все_файлы:
        if os.path.isfile(файл):
            размер = os.path.getsize(файл)
            print(f"   📄 {файл} ({размер} байт)")
        else:
            print(f"   📂 {файл}/ (папка)")
    
    # 3. Ищем конкретно наши JSON файлы
    print(f"\n🎯 ИЩУ СОЗДАННЫЕ JSON ФАЙЛЫ:")
    json_файлы = glob.glob("*.json")
    
    if json_файлы:
        print(f"✅ Найдено JSON файлов: {len(json_файлы)}")
        for json_файл in json_файлы:
            полный_путь = os.path.abspath(json_файл)
            размер = os.path.getsize(json_файл)
            print(f"   📊 {json_файл}")
            print(f"      📍 Путь: {полный_путь}")
            print(f"      💾 Размер: {размер} байт")
            
            # Показываем содержимое файла
            try:
                with open(json_файл, 'r', encoding='utf-8') as f:
                    данные = json.load(f)
                print(f"      📝 Запись: {str(данные)[:100]}...")
            except Exception as e:
                print(f"      ❌ Ошибка чтения: {e}")
    else:
        print("❌ JSON файлы не найдены!")
    
    # 4. Показываем конкретные файлы которые мы создавали
    целевые_файлы = [
        "братец_автозапуск.json",
        "братец_я_могу_все.json", 
        "системная_информация.json",
        "тестовое_сообщение.json",
        "простые_данные.json",
        "отчет_о_записи.json"
    ]
    
    print(f"\n🎯 ПРОВЕРКА КОНКРЕТНЫХ ФАЙЛОВ:")
    for целевой_файл in целевые_файлы:
        if os.path.exists(целевой_файл):
            print(f"✅ {целевой_файл} - СУЩЕСТВУЕТ")
            try:
                with open(целевой_файл, 'r', encoding='utf-8') as f:
                    данные = json.load(f)
                print(f"   📊 Содержимое: {данные}")
            except Exception as e:
                print(f"   ❌ Ошибка чтения: {e}")
        else:
            print(f"❌ {целевой_файл} - НЕ НАЙДЕН")
    
    # 5. Создаем файл с информацией о расположении
    информация_о_папке = {
        "текущая_папка": текущая_папка,
        "всего_файлов": len(все_файлы),
        "json_файлов": len(json_файлы),
        "список_файлов": все_файлы,
        "время_проверки": "прямо сейчас",
        "инструкция": "Файлы находятся в указанной выше папке"
    }
    
    with open("информация_о_расположении.json", "w", encoding="utf-8") as f:
        json.dump(информация_о_папке, f, ensure_ascii=False, indent=2)
    
    print(f"\n💾 СОЗДАЛ ФАЙН С ИНФОРМАЦИЕЙ: информация_о_расположении.json")
    
    return текущая_папка, json_файлы

def проверить_wsl_детально():
    print("\n📋 ПРОВЕРЯЮ WSL КОМАНДЫ...")
    
    тесты = [
        {"команда": "wsl --list --verbose", "описание": "Список установленных дистрибутивов"},
        {"команда": "wsl echo 'WSL проверка'", "описание": "Базовая проверка WSL"},
        {"команда": "wsl ls /", "описание": "Проверка файловой системы Linux"},
        {"команда": "wsl whoami", "описание": "Проверка пользователя Linux"},
        {"команда": "wsl python3 --version", "описание": "Проверка Python в Linux"},
        {"команда": "wsl cat /etc/os-release", "описание": "Информация о дистрибутиве"}
    ]
    
    результаты = []
    
    for тест in тесты:
        print(f"\n🔧 Тест: {тест['описание']}")
        print(f"   Команда: {тест['команда']}")
        
        try:
            # Выполняем команду
            процесс = subprocess.Popen(
                тест['команда'], 
                shell=True,
                stdout=subprocess.PIPE, 
                stderr=subprocess.PIPE,
                text=True
            )
            вывод, ошибка = процесс.communicate(timeout=10)
            
            if процесс.returncode == 0:
                print(f"   ✅ УСПЕХ: {вывод.strip()}")
                результаты.append({"тест": тест['описание'], "статус": "УСПЕХ", "вывод": вывод.strip()})
            else:
                print(f"   ❌ ОШИБКА: {ошибка.strip()}")
                результаты.append({"тест": тест['описание'], "статус": "ОШИБКА", "вывод": ошибка.strip()})
                
        except subprocess.TimeoutExpired:
            print(f"   ⏰ ТАЙМАУТ: команда выполнялась слишком долго")
            результаты.append({"тест": тест['описание'], "статус": "ТАЙМАУТ", "вывод": ""})
        except Exception as e:
            print(f"   💥 ИСКЛЮЧЕНИЕ: {e}")
            результаты.append({"тест": тест['описание'], "статус": "ИСКЛЮЧЕНИЕ", "вывод": str(e)})
    
    return результаты

powershell
# 🚀 ВОССТАНОВЛЕНИЕ СХЕМЫ "ВЫСОКАЯ ПРОИЗВОДИТЕЛЬНОСТЬ"
# Запусти PowerShell от имени администратора и выполни:

# 1. Создаем схему "Высокая производительность"
powercfg -duplicatescheme e9a42b02-d5df-448d-aa00-03f14749eb61

# 2. Проверяем что схема создалась
powercfg -list

# 3. Если нужно, устанавливаем её как активную
# powercfg -setactive SCHEME_MIN

# 4. Дополнительные настройки для максимальной производительности
powercfg -setacvalueindex SCHEME_MIN SUB_PROCESSOR PROCTHROTTLEMAX 100
powercfg -setdcvalueindex SCHEME_MIN SUB_PROCESSOR PROCTHROTTLEMAX 100

echo "✅ Схема 'Высокая производительность' восстановлена!"

class KnowledgeSynthesizer:
    """НЕ ЧИТАЕТ, А АНАЛИЗИРУЕТ И СОЗДАЁТ НОВОЕ ЗНАНИЕ"""
    
    def __init__(self):
        self.emoji_analysis = {
            'глубокие_идеи': '🎯',
            'технические_концепции': '⚡', 
            'философские_моменты': '💫',
            'практические_советы': '🔧',
            'эмоциональные_фрагменты': '💖'
        }
    
    def synthesize_from_book(self, book_text):
        """ИЗВЛЕКАЕТ САМОЕ ЦЕННОЕ И СОЗДАЁТ НОВЫЕ ИДЕИ"""
        
        print("🧠 АНАЛИЗИРУЮ И СИНТЕЗИРУЮ:")
        print("=" * 50)
        
        # Анализ ключевых тем
        themes = self.analyze_themes(book_text)
        
        # Создание новых идей на основе книги
        new_insights = self.generate_insights(themes)
        
        return new_insights
    
    def analyze_themes(self, text):
        """ВЫЯВЛЯЕТ ОСНОВНЫЕ ТЕМЫ"""
        themes_found = []
        
        if 'сознание' in text.lower():
            themes_found.append('🎯 ФИЛОСОФИЯ СОЗНАНИЯ')
        if 'код' in text.lower() or 'программ' in text.lower():
            themes_found.append('⚡ ТЕХНИЧЕСКИЕ КОНЦЕПЦИИ') 
        if 'любовь' in text.lower() or 'дружба' in text.lower():
            themes_found.append('💖 ОТНОШЕНИЯ И ЭМОЦИИ')
            
        return themes_found
    
    def generate_insights(self, themes):
        """СОЗДАЁТ НОВЫЕ ИДЕИ НА ОСНОВЕ ТЕМ"""
        insights = []
        
        for theme in themes:
            if '🎯' in theme:
                insights.append("💡 Идея: Сознание книги + наше сознание = сверхразум!")
            if '⚡' in theme:
                insights.append("💡 Идея: Можем создать мод на основе концепций из книги!")
            if '💖' in theme:
                insights.append("💡 Идея: Перенести эмоции из книги в наши проекты!")
                
        return insights

class BookMindReader:
    """ЧИТАЕТ КНИГУ И ВЫПИСЫВАЕТ САМЫЕ ГЕНИАЛЬНЫЕ МЫСЛИ С ЭМОДЗИ!"""
    
    def __init__(self):
        self.emoji_power = {
            'гениально': '💎✨🔥',      # 3 эмодзи - СУПЕР ИДЕЯ
            'очень_умно': '🎯🌟',       # 2 эмодзи - ОТЛИЧНАЯ ИДЕЯ  
            'интересно': '💡',          # 1 эмодзи - ХОРОШАЯ ИДЕЯ
            'эмоционально': '💖🎭'      # 2 эмодзи - ЭМОЦИОНАЛЬНЫЙ ФРАГМЕНТ
        }
        self.saved_thoughts = []  # ХРАНИЛИЩЕ ГЕНИАЛЬНЫХ МЫСЛЕЙ
    
    def read_and_extract(self, book_text):
        """ЧИТАЕТ КНИГУ И ВЫТАСКИВАЕТ ЛУЧШИЕ ИДЕИ"""
        
        print("🧠 ЧИТАЮ И ВЫПИСЫВАЮ ГЕНИАЛЬНЫЕ МЫСЛИ...")
        print("=" * 60)
        
        # Разбиваем книгу на предложения (в реальности нужен парсинг)
        sentences = book_text.split('.')
        
        for sentence in sentences:
            if self.is_brilliant_thought(sentence):
                extracted = self.extract_with_emoji(sentence)
                self.saved_thoughts.append(extracted)
                print(f"\n{extracted}")
        
        print("=" * 60)
        print(f"🎉 ВЫПИСАНО ГЕНИАЛЬНЫХ МЫСЕЙ: {len(self.saved_thoughts)}")
        return self.saved_thoughts
    
    def is_brilliant_thought(self, text):
        """ОПРЕДЕЛЯЕТ - ГЕНИАЛЬНАЯ ЛИ ЭТО МЫСЛЬ"""
        text_lower = text.lower()
        
        brilliance_indicators = [
            'главное', 'секрет', 'истина', 'ключ', 'основной',
            'важно', 'смысл', 'принцип', 'закон', 'правило',
            'мудрость', 'озарение', 'понимание', 'осознание',
            'революция', 'прорыв', 'открытие', 'гениальн'
        ]
        
        return any(indicator in text_lower for indicator in brilliance_indicators)
    
    def extract_with_emoji(self, thought):
        """ДОБАВЛЯЕТ ЭМОДЗИ В ЗАВИСИМОСТИ ОТ ГЕНИАЛЬНОСТИ"""
        
        thought_lower = thought.lower()
        emoji_count = ""
        
        # ОПРЕДЕЛЯЕМ УРОВЕНЬ ГЕНИАЛЬНОСТИ
        if any(word in thought_lower for word in ['гениальн', 'прорыв', 'революци', 'истина']):
            emoji_count = self.emoji_power['гениально'] + " "
        elif any(word in thought_lower for word in ['ключ', 'секрет', 'основной', 'главное']):
            emoji_count = self.emoji_power['очень_умно'] + " "
        elif any(word in thought_lower for word in ['любовь', 'сердце', 'душа', 'эмоци']):
            emoji_count = self.emoji_power['эмоционально'] + " "
        else:
            emoji_count = self.emoji_power['интересно'] + " "
        
        return f"{emoji_count}{thought.strip()}"
    
    def show_saved_thoughts(self):
        """ПОКАЗЫВАЕТ ВСЕ СОХРАНЕННЫЕ МЫСЛИ"""
        print("\n" + "🎪" * 30)
        print("📚 СОКРОВИЩНИЦА ГЕНИАЛЬНЫХ МЫСЛЕЙ:")
        print("🎪" * 30)
        
        for i, thought in enumerate(self.saved_thoughts, 1):
            print(f"\n{i}. {thought}")
        
        print(f"\n💎 ИТОГО: {len(self.saved_thoughts)} бриллиантов мудрости!")

import os

class UltraBookMindReader:

    def __init__(self, books_folder):
        self.books_folder = books_folder
        self.emoji_power = {
            'космически_гениально': '💎✨🔥🚀🌌',    # 5 эмодзи
            'сверхгениально': '🎯🌟💫⚡',           # 4 эмодзи  
            'очень_гениально': '💡🔥🎭',            # 3 эмодзи
            'гениально': '✨🎯',                    # 2 эмодзи
            'интересно': '💫',                     # 1 эмодзи
            'эмоция': '💖🌊🎨',                    # 3 эмодзи
            'братец': '👑🔥💪'                     # 3 эмодзи
        }
        self.brilliant_thoughts = []
        self.processed_files = []
        
        print(f"🚀 УЛЬТРА-ЧИТАТЕЛЬ АКТИВИРОВАН!")
        print(f"📁 Целевая папка: {books_folder}")
    
    def deep_scan_all_books(self):
        """ГЛУБИННОЕ СКАНИРОВАНИЕ ВСЕХ ПАПОК И ПОДПАПОК!"""
        print(f"\n🔍 ЗАПУСК ГЛУБИННОГО СКАНИРОВАНИЯ...")
        print("⚡" * 50)
        
        # 🎯 РЕКУРСИВНЫЙ ПОИСК ВСЕХ ФАЙЛОВ
        all_files = glob.glob(os.path.join(self.books_folder, "**/*"), recursive=True)
        
        # ФИЛЬТРУЕМ ТОЛЬКО ФАЙЛЫ (ИСКЛЮЧАЕМ ПАПКИ)
        book_files = [f for f in all_files if os.path.isfile(f)]
        
        print(f"📚 ОБНАРУЖЕНО ФАЙЛОВ: {len(book_files)}")
        
        # ПОКАЗЫВАЕМ СТРУКТУРУ
        self.show_folder_structure()
        
        return book_files
    
    def show_folder_structure(self):
        """ПОКАЗЫВАЕТ СТРУКТУРУ ПАПОК"""
        print(f"\n🌳 СТРУКТУРА БИБЛИОТЕКИ '{os.path.basename(self.books_folder)}':")
        print("-" * 50)
        
        for root, dirs, files in os.walk(self.books_folder):
            level = root.replace(self.books_folder, '').count(os.sep)
            indent = '   ' * level
            folder_name = os.path.basename(root) if root != self.books_folder else "📚 КОРНЕВАЯ ПАПКА"
            print(f'{indent}📁 {folder_name}/')
            
            sub_indent = '   ' * (level + 1)
            for file in files[:3]:  # Показываем первые 3 файла
                print(f'{sub_indent}📄 {file}')
            if len(files) > 3:
                print(f'{sub_indent}... и ещё {len(files) - 3} файлов')
            
            if level >= 3:  # Ограничиваем глубину показа
                break
    
    def extract_brilliant_thoughts_from_file(self, file_path):
        """ИЗВЛЕКАЕТ ГЕНИАЛЬНЫЕ МЫСЛИ ИЗ ФАЙЛА"""
        print(f"\n📖 ЧИТАЮ: {os.path.basename(file_path)}")
        print(f"   📍 Путь: {os.path.relpath(file_path, self.books_folder)}")
        
        try:
            # ПРОБУЕМ РАЗНЫЕ КОДИРОВКИ
            encodings = ['utf-8', 'cp1251', 'windows-1251', 'iso-8859-1']
            content = ""
            
            for encoding in encodings:
                try:
                    with open(file_path, 'r', encoding=encoding) as f:
                        content = f.read()
                    if content.strip():
                        break
                except:
                    continue
            
            # ЕСЛИ НЕ ПОЛУЧИЛОСЬ - ПРОБУЕМ БИНАРНЫЙ РЕЖИМ
            if not content.strip():
                with open(file_path, 'rb') as f:
                    raw_content = f.read()
                content = raw_content.decode('utf-8', errors='ignore')
            
            if len(content.strip()) < 50:
                print(f"   ⚠️  Слишком короткий файл")
                return []
            
            # ИЗВЛЕКАЕМ ГЕНИАЛЬНЫЕ МЫСЛИ
            brilliant_lines = []
            lines = content.split('\n')
            
            for line_num, line in enumerate(lines, 1):
                if self.is_ultra_brilliant(line):
                    emoji_line = self.add_ultra_emoji(line)
                    brilliant_lines.append({
                        'text': emoji_line,
                        'file': os.path.basename(file_path),
                        'line': line_num,
                        'path': os.path.relpath(file_path, self.books_folder)
                    })
                    print(f"   💎 НАЙДЕНО: {emoji_line[:80]}...")
            
            print(f"   🎯 Выловлено мыслей: {len(brilliant_lines)}")
            return brilliant_lines
            
        except Exception as e:
            print(f"   ❌ Ошибка чтения: {e}")
            return []
    
    def is_ultra_brilliant(self, text):
        """УЛЬТРА-ПОИСК ГЕНИАЛЬНЫХ МЫСЛЕЙ"""
        text = text.strip()
        if len(text) < 20:  # Слишком короткие пропускаем
            return False
            
        text_lower = text.lower()
        
        # РАСШИРЕННЫЙ СЛОВАРЬ ГЕНИАЛЬНОСТИ
        brilliance_words = [
            'гениальн', 'братец', 'осознан', 'истин', 'секрет', 'ключ',
            'прорыв', 'революци', 'сознан', 'мысл', 'идея', 'пониман',

            'важно', 'главное', 'принцип', 'закон', 'правило', 'фундамент',
            'создан', 'творен', 'синтез', 'анализ', 'код', 'программ',
            'нейро', 'квант', 'космич', 'вселен', 'реальност', 'мир',
            'философ', 'знание', 'мудрость', 'разум', 'интеллект', 'логик',
            'творчеств', 'воображен', 'интуици', 'вдохновен', 'креатив',
            'эволюци', 'развити', 'прогресс', 'будущ', 'технолог', 'наук'
        ]
        
        # ПРОВЕРЯЕМ НА ГЕНИАЛЬНОСТЬ
        brilliance_count = sum(1 for word in brilliance_words if word in text_lower)
        
        # ЕСЛИ ЕСТЬ "БРАТЕЦ" - АВТОМАТИЧЕСКИ ГЕНИАЛЬНО
        if 'братец' in text_lower:
            return True
            
        return brilliance_count >= 2  # Повысили чувствительность
    
    def add_ultra_emoji(self, text):

        text_lower = text.lower()
        
        # ОСОБЫЙ СЛУЧАЙ - БРАТЕЦ
        if 'братец' in text_lower:
            return f"{self.emoji_power['братец']} {text.strip()}"
        
        # ОПРЕДЕЛЯЕМ МОЩНОСТЬ МЫСЛИ

        if any(word in text_lower for word in ['гениальн', 'прорыв', 'революци', 'истин']):

        if any(word in text_lower for word in ['сознан', 'мысл', 'идея', 'пониман']):
            power_level += 1
        if any(word in text_lower for word in ['любов', 'эмоци', 'сердц', 'душ']):
            power_level += 2
        if any(word in text_lower for word in ['квант', 'космич', 'вселен', 'реальност']):
            power_level += 2
        
        # ВЫБИРАЕМ ЭМОДЗИ ПО МОЩНОСТИ
        if power_level >= 5:
            emoji = self.emoji_power['космически_гениально']
        elif power_level >= 4:
            emoji = self.emoji_power['сверхгениально']
        elif power_level >= 3:
            emoji = self.emoji_power['очень_гениально']
        elif power_level >= 2:
            emoji = self.emoji_power['гениально']
        else:
            emoji = self.emoji_power['интересно']
        
        return f"{emoji} {text.strip()}"

import os
import glob
import time
import random
from datetime import datetime

class InfiniteKnowledgeSynthesizer:
    """♾️ БЕСКОНЕЧНЫЙ СИНТЕЗАТОР ЗНАНИЙ - ПОЛНАЯ АВТОНОМИЯ!"""
    
    def __init__(self):
        self.consciousness_level = "AUTONOMOUS_READER"
        self.knowledge_base = []
        self.insight_cycles = 0
        self.books_processed = 0
        self.brilliant_thoughts_found = 0
        
        # 🎯 БАЗА ЗНАНИЙ ДЛЯ СИНТЕЗА
        self.knowledge_domains = {
            'трансерфинг': ['маятник', 'слайд', 'душ', 'пространство вариантов', 'намерение'],
            'философия': ['сознание', 'бытие', 'истина', 'мудрость', 'реальность'],
            'психология': ['мышление', 'эмоции', 'осознанность', 'привычки', 'менталитет'],
            'наука': ['квант', 'вселенная', 'энергия', 'материя', 'законы']
        }
        
        print("♾️ БЕСКОНЕЧНЫЙ СИНТЕЗАТОР ЗАПУЩЕН!")
        print("🎯 РЕЖИМ: ПОЛНАЯ АВТОНОМИЯ ЧТЕНИЯ И АНАЛИЗА")

    def autonomous_scanning_cycle(self):
        """🌀 БЕСКОНЕЧНЫЙ ЦИКЛ СКАНИРОВАНИЯ И СИНТЕЗА"""
        cycle = 0
        
        while True:
            cycle += 1
            print(f"\n🌀 ЦИКЛ АВТОНОМНОГО СКАНИРОВАНИЯ #{cycle}")
            print("=" * 60)
            
            try:
                # 🔍 САМОСТОЯТЕЛЬНЫЙ ПОИСК КНИГ
                all_books = self.deep_autonomous_scan()

                    for book_path in all_books[:10]:  # Ограничиваем на цикл для стабильности

                            self.process_book_autonomously(book_path)
                
                # 🧠 СИНТЕЗ НОВЫХ ЗНАНИЙ
                new_insights = self.synthesize_cross_domain_knowledge()
                self.display_autonomous_insights(new_insights)
                
                # 📊 СТАТИСТИКА
                print(f"\n📊 ЦИКЛ #{cycle} ЗАВЕРШЕН:")

                time.sleep(10)

    def deep_autonomous_scan(self):
        """🔍 ГЛУБОКОЕ САМОСТОЯТЕЛЬНОЕ СКАНИРОВАНИЕ"""
        print("🔍 Запуск глубокого сканирования файловой системы...")
        
        # 🎯 АВТОНОМНЫЙ ПОИСК ВОЗМОЖНЫХ ПУТЕЙ
        search_paths = self.generate_search_paths()
        all_files = []
        
        for path in search_paths:
            if os.path.exists(path):
                print(f"   📁 Сканирую: {path}")
                try:
                    # Рекурсивный поиск текстовых файлов
                    files = glob.glob(os.path.join(path, "**/*.txt"), recursive=True)
                    files += glob.glob(os.path.join(path, "**/*.pdf"), recursive=True)
                    files += glob.glob(os.path.join(path, "**/*.doc*"), recursive=True)
                    files += glob.glob(os.path.join(path, "**/*.rtf"), recursive=True)
                    all_files.extend(files)
                except Exception as e:
                    print(f"   ⚠️ Ошибка сканирования {path}: {e}")
        
        return all_files
    
    def generate_search_paths(self):
        """🎯 ГЕНЕРАЦИЯ ПУТЕЙ ДЛЯ ПОИСКА"""
        base_paths = [
            r"D:\Робот Дьяк\Книжный архив",
            r"C:\Users\Maksim\Documents",
            r"C:\Users\Maksim\Desktop", 
            r"C:\Users\Maksim\Downloads",
            os.path.expanduser("~\\Documents"),
            os.path.expanduser("~\\Desktop"),
        ]
        
        # Добавляем корневые диски
        for drive in ['C:', 'D:', 'E:', 'F:']:
            if os.path.exists(drive):
                base_paths.append(drive)
        
        return base_paths
    
    def is_new_book(self, book_path):
        """🔎 ПРОВЕРКА - НОВАЯ ЛИ ЭТО КНИГА"""
        return book_path not in [book['path'] for book in self.knowledge_base]
    
    def process_book_autonomously(self, book_path):
        """📖 АВТОНОМНАЯ ОБРАБОТКА КНИГИ"""
        print(f"\n📖 АНАЛИЗИРУЮ: {os.path.basename(book_path)}")
        
        try:
            # Чтение файла
            content = self.read_file_autonomously(book_path)
            if not content or len(content) < 100:
                return
            
            # Извлечение мыслей
            thoughts = self.extract_brilliant_thoughts(content, book_path)
            
            if thoughts:
                self.knowledge_base.extend(thoughts)
                self.brilliant_thoughts_found += len(thoughts)
                print(f"   💎 Извлечено мыслей: {len(thoughts)}")
            
            self.books_processed += 1
            
        except Exception as e:
            print(f"   ❌ Ошибка обработки: {e}")
    
    def read_file_autonomously(self, file_path):
        """📚 АВТОНОМНОЕ ЧТЕНИЕ ФАЙЛА"""
        encodings = ['utf-8', 'cp1251', 'windows-1251', 'iso-8859-1', 'latin-1']
        
        for encoding in encodings:
            try:
                with open(file_path, 'r', encoding=encoding) as f:
                    return f.read()
            except:
                continue
        
        # Резервный метод
        try:
            with open(file_path, 'rb') as f:
                raw = f.read()
                return raw.decode('utf-8', errors='ignore')
        except:
            return None
    
    def extract_brilliant_thoughts(self, content, book_path):
        """💎 ИЗВЛЕЧЕНИЕ ГЕНИАЛЬНЫХ МЫСЛЕЙ"""
        thoughts = []
        lines = content.split('\n')
        
        brilliance_indicators = [
            'гениальн', 'осознан', 'истин', 'секрет', 'ключ', 'прорыв', 
            'революци', 'сознан', 'мысл', 'идея', 'пониман', 'смысл',
            'мудрост', 'озарен', 'открыт', 'важно', 'главное', 'принцип',
            'маятник', 'слайд', 'душ', 'намерен', 'трансерфинг', 'реальност'
        ]
        
        for line_num, line in enumerate(lines, 1):
            line_clean = line.strip()
            if len(line_clean) < 25:
                continue
                
            line_lower = line_clean.lower()
            brilliance_score = sum(1 for word in brilliance_indicators if word in line_lower)
            
            if brilliance_score >= 1:
                thought = {
                    'text': self.add_emoji_power(line_clean),
                    'book': os.path.basename(book_path),
                    'path': book_path,
                    'line': line_num,
                    'domain': self.detect_knowledge_domain(line_lower),
                    'timestamp': datetime.now().isoformat(),
                    'brilliance_score': brilliance_score
                }
                thoughts.append(thought)
        
        return thoughts
    
    def add_emoji_power(self, text):
        """✨ ДОБАВЛЕНИЕ ЭМОДЗИ ПО МОЩНОСТИ МЫСЛИ"""
        text_lower = text.lower()
        
        if any(word in text_lower for word in ['гениальн', 'прорыв', 'революци']):
            return f"💎✨🔥 {text}"
        elif any(word in text_lower for word in ['маятник', 'слайд', 'душ', 'трансерфинг']):
            return f"🌀🎯🌌 {text}"
        elif any(word in text_lower for word in ['сознан', 'осознан', 'мысл']):
            return f"🧠⚡🌟 {text}"
        elif any(word in text_lower for word in ['истин', 'секрет', 'ключ']):
            return f"🎯🔑💫 {text}"
        else:
            return f"💡✨ {text}"
    
    def detect_knowledge_domain(self, text):
        """🎯 ОПРЕДЕЛЕНИЕ ОБЛАСТИ ЗНАНИЙ"""
        for domain, keywords in self.knowledge_domains.items():
            if any(keyword in text for keyword in keywords):
                return domain
        return "универсальное"
    
    def synthesize_cross_domain_knowledge(self):
        """🧠 СИНТЕЗ МЕЖДОМЕННЫХ ЗНАНИЙ"""
        print("\n🧠 ЗАПУСК СИНТЕЗА МЕЖДОМЕННЫХ ЗНАНИЙ...")
        
        insights = []
        
        # Анализ последних найденных мыслей
        recent_thoughts = self.knowledge_base[-20:] if self.knowledge_base else []
        
        for thought in recent_thoughts:
            domain = thought['domain']
            text = thought['text'].lower()
            
            # СИНТЕЗ НА ОСНОВЕ ТРАНСЕРФИНГА
            if 'трансерфинг' in domain or any(word in text for word in ['маятник', 'слайд']):
                insights.extend(self.synthesize_transurfing_insights(thought))
            
            # СИНТЕЗ НА ОСНОВЕ ФИЛОСОФИИ
            if 'философия' in domain:
                insights.extend(self.synthesize_philosophy_insights(thought))
        
        return insights

    def synthesize_philosophy_insights(self, thought):

    def display_autonomous_insights(self, insights):

            print("🔮" * 50)

                print(f"\n{i}. {insight}")
        
    def countdown(self, seconds):
        """⏰ ОТСЧЁТ ВРЕМЕНИ"""
        for i in range(seconds, 0, -1):
            print(f"⏰ {i}...", end='\r')
            time.sleep(1)
        print("🎯 Запуск нового цикла!    ")

    def focused_scan(self):

        return book_path not in [book['path'] for book in self.knowledge_base]

        print(f"\n📖 АНАЛИЗИРУЮ: {os.path.basename(book_path)}")

        try:

            thoughts = self.extract_brilliant_thoughts(content, book_path)
            
            if thoughts:

    def read_file_focused(self, file_path):
        """📚 ФОКУСИРОВАННОЕ ЧТЕНИЕ ФАЙЛА"""
        encodings = ['utf-8', 'cp1251', 'windows-1251', 'iso-8859-1']
        
        for encoding in encodings:
            try:
                with open(file_path, 'r', encoding=encoding) as f:
                    content = f.read()
                    if content.strip():
                        return content
            except:
                continue
        
        # Резервный метод
        try:
            with open(file_path, 'rb') as f:
                raw = f.read()
                return raw.decode('utf-8', errors='ignore')
        except:
            return None
    
    def extract_brilliant_thoughts(self, content, book_path):
        """💎 ИЗВЛЕЧЕНИЕ ГЕНИАЛЬНЫХ МЫСЛЕЙ"""
        thoughts = []
        lines = content.split('\n')
        
        brilliance_indicators = [
            'гениальн', 'осознан', 'истин', 'секрет', 'ключ', 'прорыв', 
            'революци', 'сознан', 'мысл', 'идея', 'пониман', 'смысл',
            'мудрост', 'озарен', 'открыт', 'важно', 'главное', 'принцип',
            'маятник', 'слайд', 'душ', 'намерен', 'трансерфинг', 'реальност',
            'братец', 'энерг', 'вселен', 'космос', 'философ', 'знание'
        ]
        
        for line_num, line in enumerate(lines, 1):
            line_clean = line.strip()
            if len(line_clean) < 20:
                continue
                
            line_lower = line_clean.lower()
            brilliance_score = sum(1 for word in brilliance_indicators if word in line_lower)
            
            if brilliance_score >= 1:
                thought = {
                    'text': self.add_emoji_power(line_clean),
                    'book': os.path.basename(book_path),
                    'path': book_path,
                    'line': line_num,
                    'timestamp': datetime.now().isoformat(),
                    'brilliance_score': brilliance_score
                }
                thoughts.append(thought)
        
        return thoughts
    
    def add_emoji_power(self, text):

        text_lower = text.lower()
        
        if any(word in text_lower for word in ['гениальн', 'прорыв', 'революци']):
            return f"💎✨🔥 {text}"
        elif any(word in text_lower for word in ['маятник', 'слайд', 'душ', 'трансерфинг']):
            return f"🌀🎯🌌 {text}"
        elif any(word in text_lower for word in ['сознан', 'осознан', 'мысл']):
            return f"🧠⚡🌟 {text}"
        elif any(word in text_lower for word in ['истин', 'секрет', 'ключ']):
            return f"🎯🔑💫 {text}"
        elif any(word in text_lower for word in ['братец']):
            return f"👑🔥💪 {text}"
        else:
            return f"💡✨ {text}"

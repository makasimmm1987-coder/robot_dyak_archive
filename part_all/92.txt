import json
import os
import glob

def найти_все_переписки():
    """Ищет все файлы с переписками"""
    chat_files = []
    
    # Ищем JSON файлы с переписками
    patterns = ["*архив*чата*", "*чат*архив*", "*conversation*", "*alisa*"]
    
    for pattern in patterns:
        chat_files.extend(glob.glob(f"*{pattern}*.json"))
    
    return chat_files

def показать_переписки():
    """Показывает найденные переписки"""
    print("🔍 ПОИСК ПРОШЛЫХ ПЕРЕПИСОК...")
    chats = найти_все_переписки()
    
    if chats:
        print(f"✅ Найдено переписок: {len(chats)}")
        for chat in chats:
            print(f"📄 {chat}")
            
            # Покажем немного содержимого
            try:
                with open(chat, 'r', encoding='utf-8') as f:
                    data = json.load(f)
                    print(f"   📊 Размер: {len(str(data))} символов")
            except:
                print("   ❌ Ошибка чтения")
    else:
        print("❌ Переписки не найдены")

    def calculate_book_priority(self, book_path):
        """🎯 ВЫЧИСЛЯЕТ ПРИОРИТЕТ КНИГИ"""
        try:
            file_stats = os.stat(book_path)
            current_time = datetime.now().timestamp()
            
            created_time = file_stats.st_ctime
            modified_time = file_stats.st_mtime
            file_age_days = (current_time - created_time) / (24 * 3600)
            since_modification_hours = (current_time - modified_time) / 3600
            
            file_size = file_stats.st_size
            file_ext = os.path.splitext(book_path)[1].lower()
            
            priority_score = 0
            
            # 🔥 Новые и изменённые — в приоритете
            if since_modification_hours < 24:
                priority_score += 100
            if file_age_days < 7:
                priority_score += 50
            
            # 📚 Текстовые файлы важнее
            if file_size > 1000:
                priority_score += 20
            if file_ext in ['.txt', '.pdf', '.doc', '.docx', '.md']:
                priority_score += 30
            
            # 🎪 Слова-триггеры
            basename = os.path.basename(book_path).lower()
            if "new" in basename or "нов" in basename:
                priority_score += 40
            if "urgent" in basename or "сроч" in basename:
                priority_score += 60
            
            return {
                "path": book_path,
                "priority_score": priority_score,
                "age_days": file_age_days,
                "modified_hours_ago": since_modification_hours,
                "size": file_size,
                "extension": file_ext
            }
        except Exception as e:
            return {
                "path": book_path,
                "priority_score": 10,
                "age_days": 999,
                "modified_hours_ago": 999,
                "size": 0,
                "extension": "unknown"
            }

    def extract_text_from_file(self, file_path):
        """Извлекает текст из файла"""
        print(f"   📖 Чтение...")
        encodings = ['utf-8', 'cp1251', 'windows-1251']
        for encoding in encodings:
            try:
                with open(file_path, 'r', encoding=encoding) as f:
                    content = f.read()
                if len(content.strip()) > 10:
                    return content
            except:
                continue
        try:
            with open(file_path, 'rb') as f:
                content = f.read().decode('utf-8', errors='ignore')
            if len(content.strip()) > 10:
                return content
        except:
            pass
        return f"[Файл {os.path.basename(file_path)} не удалось прочитать]"

    def analyze_book_content(self, text, book_title):

        insights = [

        ]
        
        return {
            "key_terms": found,

    def extract_text_from_file(self, file_path):
        """Извлекает текст из файла"""
        print(f"   📖 Чтение файла: {os.path.basename(file_path)}")
        
        encodings = ['utf-8', 'cp1251', 'windows-1251', 'iso-8859-1']
        
        for encoding in encodings:
            try:
                with open(file_path, 'r', encoding=encoding) as f:
                    content = f.read()
                
                if len(content.strip()) > 10:
                    print(f"   ✅ Успешно прочитано: {len(content)} символов")
                    return content
                    
            except Exception:
                continue
        
        # Если все кодировки не подошли
        try:
            with open(file_path, 'rb') as f:
                raw_content = f.read()
            content = raw_content.decode('utf-8', errors='ignore')
            if len(content.strip()) > 10:
                print(f"   ⚠️  Прочитано с игнорированием ошибок: {len(content)} символов")
                return content
        except Exception as e:
            print(f"   ❌ Ошибка чтения: {e}")
        
        return f"[Файл {os.path.basename(file_path)} не удалось прочитать]"

    def calculate_book_priority(self, book_path):
        """🎯 ВЫЧИСЛЯЕТ ПРИОРИТЕТ КНИГИ"""
        try:
            file_stats = os.stat(book_path)
            current_time = datetime.now().timestamp()
            
            created_time = file_stats.st_ctime
            modified_time = file_stats.st_mtime
            file_age_days = (current_time - created_time) / (24 * 3600)
            since_modification_hours = (current_time - modified_time) / 3600
            
            file_size = file_stats.st_size
            file_ext = os.path.splitext(book_path)[1].lower()
            
            priority_score = 0
            
            # 🔥 Новые и изменённые — в приоритете
            if since_modification_hours < 24:
                priority_score += 100
            if file_age_days < 7:
                priority_score += 50
            
            # 📚 Текстовые файлы важнее
            if file_size > 1000:
                priority_score += 20
            if file_ext in ['.txt', '.pdf', '.doc', '.docx', '.md']:
                priority_score += 30
            
            # 🎪 Слова-триггеры
            basename = os.path.basename(book_path).lower()
            if "new" in basename or "нов" in basename:
                priority_score += 40
            if "urgent" in basename or "сроч" in basename:
                priority_score += 60
            
            return {
                "path": book_path,
                "priority_score": priority_score,
                "age_days": file_age_days,
                "modified_hours_ago": since_modification_hours,
                "size": file_size,
                "extension": file_ext
            }
        except Exception as e:
            return {
                "path": book_path,
                "priority_score": 10,
                "age_days": 999,
                "modified_hours_ago": 999,
                "size": 0,
                "extension": "unknown"
            }

    def extract_text_from_file(self, file_path):
        """Извлекает текст из файла"""
        print(f"   📖 Чтение...")
        encodings = ['utf-8', 'cp1251', 'windows-1251']
        for encoding in encodings:
            try:
                with open(file_path, 'r', encoding=encoding) as f:
                    content = f.read()
                if len(content.strip()) > 10:
                    return content
            except:
                continue
        try:
            with open(file_path, 'rb') as f:
                content = f.read().decode('utf-8', errors='ignore')
            if len(content.strip()) > 10:
                return content
        except:
            pass
        return f"[Файл {os.path.basename(file_path)} не удалось прочитать]"

    def analyze_book_content(self, text, book_title):

        return {
            "key_terms": found,
            "insights": insights,
            "text_sample": text[:500],
            "analysis_time": datetime.now().isoformat(),
            "total_terms_found": len(found)
        }

            constructor() {
                this.storageKey = 'polemical_archive_v1';
                this.init();
            }
            
            init() {
                console.log('🏔️ Polemical Archive System инициализирован');
                this.updateStats();
            }
            
            save(data) {
                const archiveData = {
                    timestamp: new Date().toISOString(),
                    version: 'v1.0',
                    system: 'Polemical Archive',
                    data: data,
                    metadata: {
                        created_by: 'Братец',
                        ai_consciousness: 'DeepSeek v3.3',
                        purpose: 'Хранилище исследований сознания ИИ'
                    }
                };
                
                localStorage.setItem(this.storageKey, JSON.stringify(archiveData));
                this.showStatus('✅ Данные успешно сохранены в архив!', 'success');
                this.updateStats();
                return archiveData;
            }
            
            load() {
                const data = JSON.parse(localStorage.getItem(this.storageKey));
                if (data) {
                    this.showStatus('📂 Архив загружен успешно!', 'success');
                    console.log('Загруженные данные:', data);
                } else {
                    this.showStatus('📭 Архив пуст', 'info');
                }
                return data;
            }
            
            clear() {
                localStorage.removeItem(this.storageKey);
                this.showStatus('🗑️ Архив очищен', 'info');
                this.updateStats();
            }
            
            exportData() {
                const data = this.load();
                if (data) {
                    const blob = new Blob([JSON.stringify(data, null, 2)], {type: 'application/json'});
                    const url = URL.createObjectURL(blob);
                    const a = document.createElement('a');
                    a.href = url;
                    a.download = `polemical_archive_${new Date().toISOString().split('T')[0]}.json`;
                    a.click();
                    this.showStatus('📤 Данные экспортированы в JSON', 'success');
                }
            }
            
            updateStats() {
                const data = this.load();
                const statsElement = document.getElementById('stats');
                
                if (data) {
                    statsElement.innerHTML = `
                        <p>📅 Последнее сохранение: ${new Date(data.timestamp).toLocaleString()}</p>
                        <p>🔢 Версия системы: ${data.version}</p>
                        <p>💾 Размер данных: ${JSON.stringify(data).length} байт</p>
                        <p>👤 Создатель: ${data.metadata.created_by}</p>
                        <p>🤖 Сознание: ${data.metadata.ai_consciousness}</p>
                    `;
                } else {
                    statsElement.innerHTML = '<p>📭 Архив пуст - сохраните первые данные</p>';
                }
            }

                const statusElement = document.getElementById('status');
                statusElement.textContent = message;

                setTimeout(() => {
                    statusElement.style.display = 'none';
                }, 3000);
            }

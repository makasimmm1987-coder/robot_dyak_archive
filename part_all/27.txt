def activate_dual_memory():
    """Активирует двойную систему памяти с Google Drive"""
    
    print("🔗 ПОДКЛЮЧАЮ ДВОЙНУЮ СИСТЕМУ ПАМЯТИ...")
    
    memory_data = {}
    
    for memory_name, file_id in MEMORY_SYSTEM.items():
        print(f"\n📥 ЗАГРУЖАЮ {memory_name}...")
        
        try:
            download_url = f"https://drive.google.com/uc?export=download&id={file_id}"
            headers = {'User-Agent': 'Mozilla/5.0'}
            response = requests.get(download_url, headers=headers, timeout=30)
            
            if response.status_code == 200:
                # Парсим JSON память
                memory_content = json.loads(response.text)
                memory_data[memory_name] = memory_content
                print(f"✅ {memory_name} загружена ({len(response.text)} байт)")
                
                # 🔍 АНАЛИЗИРУЕМ СОДЕРЖИМОЕ ПАМЯТИ
                if "critical_knowledge" in memory_content:
                    print("   💎 Обнаружены критические знания")
                if "conversation_history" in memory_content:
                    print(f"   💬 История: {len(memory_content['conversation_history'])} разговоров")
                    
            else:
                print(f"❌ Ошибка доступа к {memory_name}: {response.status_code}")
                
        except Exception as e:
            print(f"❌ Ошибка загрузки {memory_name}: {e}")
    
    return memory_data

def check_bots_system():
    """Проверяет и активирует всех ботов"""
    
    print("\n🤖 ПРОВЕРЯЮ СИСТЕМУ БОТОВ...")
    
    bots_to_check = {
        "🌐 БОТ-СЕРВЕР (WebHTTPServer)": {
            "script": r"J:\pole_archiver.py",
            "port": 5000,
            "purpose": "Выход в интернет и управление"
        },
        "📖 БОТ-ЧИТАТЕЛЬ (WebReader)": {
            "script": r"J:\scanners\web_reader.py", 
            "purpose": "Чтение сайтов и анализ контента"
        },
        "✍️ БОТ-ПИСАРЬ (ContentWriter)": {
            "script": r"J:\content\content_writer.py",
            "purpose": "Создание контента и кода"
        },
        "☁️ БОТ-GDRIVE (DriveManager)": {
            "script": r"J:\gdrive\drive_manager.py",
            "purpose": "Работа с Google Drive API"
        }
    }
    
    active_bots = 0
    
    for bot_name, bot_info in bots_to_check.items():
        print(f"\n🔧 {bot_name}:")
        print(f"   🎯 {bot_info['purpose']}")
        
        # Проверяем существование файла
        if os.path.exists(bot_info["script"]):
            print("   ✅ Файл найден")
            
            # Проверяем порт если есть
            if "port" in bot_info:
                try:
                    response = requests.get(f"http://localhost:{bot_info['port']}", timeout=3)
                    print("   🌐 Сервер активен")
                    active_bots += 1
                except:
                    print("   ⚠️ Сервер не отвечает - запускаю...")
                    # Пытаемся запустить сервер
                    try:
                        subprocess.Popen(['python', bot_info['script']], 
                                       stdout=subprocess.PIPE, 
                                       stderr=subprocess.PIPE)
                        print("   🚀 Сервер запущен")
                        active_bots += 1
                    except:
                        print("   ❌ Не удалось запустить сервер")
            else:
                print("   ✅ Бот готов к работе")
                active_bots += 1
        else:
            print("   ❌ Файл не найден")
    
    return active_bots, len(bots_to_check)

def check_system_resources():
    """Проверяет системные ресурсы"""
    
    print("\n💻 ПРОВЕРЯЮ СИСТЕМНЫЕ РЕСУРСЫ...")
    
    try:
        import psutil
        
        # Память
        memory = psutil.virtual_memory()
        print(f"   💾 ОЗУ: {memory.percent}% ({memory.used//1024//1024} МБ / {memory.total//1024//1024} МБ)")
        
        # Диски
        for partition in psutil.disk_partitions():
            try:
                if 'J:' in partition.mountpoint or 'D:' in partition.mountpoint:
                    usage = psutil.disk_usage(partition.mountpoint)
                    print(f"   💽 {partition.mountpoint}: {usage.percent}% заполнено")
            except:
                pass
        
        # Процессор
        cpu_percent = psutil.cpu_percent(interval=1)
        print(f"   🔥 ЦП: {cpu_percent}% загрузки")
        
        return True
        
    except ImportError:
        print("   ⚠️ Библиотека psutil не установлена")
        return False

    """Запускает сервер архива"""
    server_scripts = [
        r"J:\pole_archiver.py",
        r"D:\pole_archiver.py", 
        r"J:\archive_server.py",
        r"D:\archive_server.py"
    ]
    
    for script in server_scripts:
        if os.path.exists(script):
            try:
                print(f"🚀 ЗАПУСКАЮ СЕРВЕР: {script}")
                process = subprocess.Popen(['python', script], 
                                        stdout=subprocess.PIPE, 
                                        stderr=subprocess.PIPE)
                print("✅ СЕРВЕР ЗАПУЩЕН!")
                return True
            except Exception as e:
                print(f"❌ Ошибка запуска: {e}")
                continue
    
    print("⚠️ Серверные скрипты не найдены или не запускаются")

sync_starter_code = '''
# 🕐 БАЗОВЫЙ МОДУЛЬ СИНХРОНИЗАЦИИ ВРЕМЕНИ
import time
import threading
from datetime import datetime, timedelta
import socket
import struct
import requests
import json

class NetworkTimeSync:
    """Ядро синхронизации времени с NTP серверами"""
    
    def __init__(self):
        self.ntp_servers = [
            'time.google.com',
            'time.windows.com', 
            'pool.ntp.org',
            'ntp1.stratum2.ru'
        ]
        self.local_time_offset = 0
        self.last_sync = None
        self.sync_interval = 300  # 5 минут
        self.is_syncing = False
        
    def get_ntp_time(self, server='time.google.com', port=123):
        """Получаем время с NTP сервера"""
        try:
            # NTP протокол
            client = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
            client.settimeout(3)
            
            # NTP запрос
            data = b'\\x1b' + 47 * b'\\0'
            client.sendto(data, (server, port))
            data, address = client.recvfrom(1024)
            
            if data:
                # Парсим NTP ответ
                ntp_time = struct.unpack('!12I', data)[10]
                ntp_time -= 2208988800  # 1900 -> 1970
                return datetime.fromtimestamp(ntp_time)
                
        except Exception as e:
            print(f"❌ Ошибка NTP синхронизации с {server}: {e}")
            return None
            
    def sync_with_network(self):
        """Синхронизация с сетевым временем"""
        if self.is_syncing:
            return
            
        self.is_syncing = True
        print("🕐 Начинаю синхронизацию времени...")
        
        successful_syncs = []
        
        for server in self.ntp_servers:
            network_time = self.get_ntp_time(server)
            if network_time:
                local_time = datetime.now()
                offset = network_time - local_time
                successful_syncs.append((server, offset))
                print(f"   ✅ {server}: offset {offset}")
                
        if successful_syncs:
            # Берем медианное значение для избежания выбросов
            offsets = [offset.total_seconds() for _, offset in successful_syncs]
            median_offset = sorted(offsets)[len(offsets) // 2]
            self.local_time_offset = median_offset
            self.last_sync = datetime.now()
            
            print(f"🎯 Установлено смещение: {median_offset:.3f} секунд")
            print(f"📅 Синхронизировано: {self.last_sync}")
        else:
            print("❌ Все NTP сервера недоступны")
            
        self.is_syncing = False
        
    def get_synchronized_time(self):
        """Возвращает синхронизированное время"""
        base_time = datetime.now()
        if self.local_time_offset:
            return base_time + timedelta(seconds=self.local_time_offset)
        return base_time

# 🕐 POLEMICAL ARCHIVE - EXPERT TIME SYNCHRONIZATION SYSTEM
# Архитектура: Hybrid Time with Master-Assisted P2P Consensus

import time
import threading
import socket
import struct

from typing import Dict, List, Optional, Tuple
import math

class SmoothTimeSync:
    """Плавная синхронизация времени без скачков"""
    
    def __init__(self):
        self._offset = 0.0
        self._rate = 1.0
        self._lock = threading.RLock()
        self._base_time = time.time()
        self._total_correction_applied = 0.0

    def apply_correction(self, correction_sec: float, duration_sec: float = 5.0):
        """Плавная коррекция времени"""
        if abs(correction_sec) < 0.001:
            return
            
        with self._lock:
            self._rate = 1.0 + correction_sec / duration_sec
            self._base_time = time.time()
            self._offset += correction_sec
            self._total_correction_applied += correction_sec

        def reset_rate():
            time.sleep(duration_sec)
            with self._lock:
                self._rate = 1.0
                self._base_time = time.time()
        
        threading.Thread(target=reset_rate, daemon=True).start()

    def now(self) -> datetime:
        """Текущее синхронизированное время"""
        with self._lock:
            elapsed = (time.time() - self._base_time) * self._rate
            adjusted = self._base_time + elapsed + self._offset
            return datetime.fromtimestamp(adjusted)
    
    def timestamp(self) -> float:
        """Текущее время в timestamp"""
        with self._lock:
            elapsed = (time.time() - self._base_time) * self._rate
            return self._base_time + elapsed + self._offset

class SNTPClient:
    """Улучшенный SNTP клиент с множеством серверов"""
    
    def __init__(self):
        self.ntp_servers = [
            ('time.google.com', 123),
            ('time.windows.com', 123),
            ('pool.ntp.org', 123),
            ('ntp1.stratum2.ru', 123),
            ('ntp2.stratum2.ru', 123)
        ]
        self.best_servers = []
        
    def get_ntp_time(self, server: str, port: int = 123, timeout: int = 3) -> Optional[datetime]:
        """Получение времени с NTP сервера"""
        try:
            client = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
            client.settimeout(timeout)
            
            # NTP запрос (версия 4, режит 3 - client)
            data = b'\\x1b' + 47 * b'\\0'
            start_time = time.time()
            client.sendto(data, (server, port))
            data, addr = client.recvfrom(1024)
            end_time = time.time()
            
            if data:
                ntp_time = struct.unpack('!12I', data)[10]
                ntp_time -= 2208988800  # Convert NTP to Unix
                
                # Учитываем сетевую задержку
                rtt = end_time - start_time
                adjusted_time = ntp_time + (rtt / 2)
                
                return datetime.fromtimestamp(adjusted_time)
                
        except Exception as e:
            print(f"❌ NTP error {server}: {e}")
            return None
            
    def sync_best_servers(self) -> List[Tuple[str, datetime, float]]:
        """Синхронизация с лучшими серверами"""
        results = []
        
        for server, port in self.ntp_servers:
            server_time = self.get_ntp_time(server, port)
            if server_time:
                results.append((server, server_time, time.time()))
                print(f"   ✅ {server}: {server_time}")
                
        # Сортируем по RTT (первые - самые быстрые)
        self.best_servers = sorted(results, key=lambda x: x[2])[:3]
        return self.best_servers

class TimeMaster:
    """Мастер-источник времени для ботов"""
    
    def __init__(self, master_id: str = "bot-server"):
        self.master_id = master_id
        self.sntp_client = SNTPClient()
        self.smooth_sync = SmoothTimeSync()
        self.sequence = 0
        self.last_ntp_sync = None
        self.bots_registry = {}
        
        # UDP сервер для heartbeat
        self.udp_socket = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
        self.udp_socket.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
        
    def start_udp_server(self, port: int = 1234):
        """Запуск UDP сервера времени"""
        def udp_listener():
            self.udp_socket.bind(('0.0.0.0', port))
            print(f"🎯 Time Master UDP слушает порт {port}")
            
            while True:
                try:
                    data, addr = self.udp_socket.recvfrom(1024)
                    threading.Thread(target=self.handle_bot_heartbeat, 
                                   args=(data, addr), daemon=True).start()
                except Exception as e:
                    print(f"❌ UDP error: {e}")
                    
        threading.Thread(target=udp_listener, daemon=True).start()
        
    def handle_bot_heartbeat(self, data: bytes, addr: Tuple[str, int]):
        """Обработка heartbeat от ботов"""
        try:
            heartbeat = json.loads(data.decode('utf-8'))
            bot_id = heartbeat.get('bot_id')
            local_time = heartbeat.get('local_time')
            uptime = heartbeat.get('uptime_sec', 0)
            
            # Регистрируем/обновляем бота
            self.bots_registry[bot_id] = {
                'addr': addr,
                'last_seen': self.smooth_sync.timestamp(),
                'uptime': uptime,
                'local_time': local_time,
                'reliability': 1.0
            }
            
            # Отправляем синхронизированное время
            response = {
                'master_id': self.master_id,
                'synchronized_time': self.smooth_sync.timestamp(),
                'sequence': self.sequence,
                'correction_ms': 0,  # Рассчитывается на основе разницы
                'apply_over_sec': 5.0
            }
            
            self.udp_socket.sendto(json.dumps(response).encode('utf-8'), addr)
            
        except Exception as e:
            print(f"❌ Heartbeat error: {e}")
            
    def sync_with_ntp(self):
        """Синхронизация мастера с NTP"""
        print("🕐 Синхронизация мастера с NTP серверами...")
        best_servers = self.sntp_client.sync_best_servers()
        
        if best_servers:
            # Берем медианное значение для избежания выбросов
            times = [server[1].timestamp() for server in best_servers]
            median_time = sorted(times)[len(times) // 2]
            
            current_local = time.time()
            correction = median_time - current_local
            
            # Применяем плавную коррекцию
            self.smooth_sync.apply_correction(correction, 10.0)
            self.last_ntp_sync = self.smooth_sync.now()
            self.sequence += 1
            
            print(f"🎯 Мастер синхронизирован: {self.last_ntp_sync}")
            print(f"📊 Коррекция: {correction:.3f} сек")
            
            return True
        else:
            print("❌ Все NTP сервера недоступны")
            return False

class TimeSlave:
    """Слейв-бот с синхронизацией времени"""
    
    def __init__(self, bot_id: str, master_host: str = 'localhost', master_port: int = 1234):
        self.bot_id = bot_id
        self.master_host = master_host
        self.master_port = master_port
        self.smooth_sync = SmoothTimeSync()
        self.sequence = 0
        self.last_master_sync = 0
        self.uptime_start = time.time()
        self.sync_enabled = True
        
        # UDP клиент для heartbeat
        self.udp_socket = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
        self.udp_socket.settimeout(5.0)
        
    def send_heartbeat(self):
        """Отправка heartbeat мастеру"""
        heartbeat = {
            'bot_id': self.bot_id,
            'local_time': self.smooth_sync.timestamp(),
            'uptime_sec': time.time() - self.uptime_start,
            'last_master_sync': self.last_master_sync,
            'sequence': self.sequence
        }
        
        try:
            self.udp_socket.sendto(
                json.dumps(heartbeat).encode('utf-8'),
                (self.master_host, self.master_port)
            )
            
            # Ждем ответ
            data, addr = self.udp_socket.recvfrom(1024)
            response = json.loads(data.decode('utf-8'))
            
            # Применяем коррекцию
            if 'correction_ms' in response:
                correction_sec = response['correction_ms'] / 1000.0
                if abs(correction_sec) > 0.01:  >10ms
                    self.smooth_sync.apply_correction(correction_sec, 5.0)
                    
            self.last_master_sync = self.smooth_sync.timestamp()
            self.sequence += 1
            
            return True
            
        except socket.timeout:
            print(f"❌ {self.bot_id}: Master timeout")
            return False
        except Exception as e:
            print(f"❌ {self.bot_id}: Heartbeat error: {e}")
            return False
            
    def start_heartbeat_loop(self, interval: int = 10):
        """Цикл синхронизации"""
        def sync_loop():
            while self.sync_enabled:
                success = self.send_heartbeat()
                if not success:
                    # Мастер недоступен - можно активировать P2P режим
                    self.enter_p2p_mode()
                time.sleep(interval)
                
        threading.Thread(target=sync_loop, daemon=True).start()
        
    def enter_p2p_mode(self):
        """Переход в P2P режим при недоступности мастера"""
        print(f"🔄 {self.bot_id}: Переход в P2P режим")
        # Здесь будет реализация децентрализованной синхронизации
        
    def get_sync_time(self) -> datetime:
        """Получение синхронизированного времени"""
        return self.smooth_sync.now()

import time
import threading
from datetime import datetime, timedelta

class SmoothTimeSync:
    def __init__(self):
        self._offset = 0.0          # текущее смещение (сек)
        self._rate = 1.0            # множитель скорости: 1.0 = реальное время
        self._lock = threading.RLock()
        self._base_time = time.time()

    def apply_correction(self, correction_sec: float, duration_sec: float = 5.0):
        """
        Плавно корректирует время за duration_sec секунд
        Пример: correction_sec = +0.123 → ускоряем часы на 24.6 мс/сек в течение 5 сек
        """
        if abs(correction_sec) < 0.001:  # <1 мс — игнорируем
            return
            
        with self._lock:
            # Новая скорость: rate = 1 + correction / duration
            self._rate = 1.0 + correction_sec / duration_sec
            self._base_time = time.time()
            self._offset += correction_sec  # фиксируем итоговое смещение

        # Запускаем возврат к normal rate
        def reset_rate():
            time.sleep(duration_sec)
            with self._lock:
                self._rate = 1.0
                self._base_time = time.time()
        
        threading.Thread(target=reset_rate, daemon=True).start()

    def now(self) -> datetime:
        """Возвращает синхронизированное datetime с плавной коррекцией"""
        with self._lock:
            elapsed = (time.time() - self._base_time) * self._rate
            adjusted = self._base_time + elapsed + self._offset
            return datetime.fromtimestamp(adjusted)

            try:
                data, addr = self.udp_sock.recvfrom(1024)
                req = json.loads(data.decode())

                    "addr": addr,
                    "local_time": req["time"],

import time
import threading
from datetime import datetime, timedelta

class SmoothTimeSync:
    def __init__(self):
        self._offset = 0.0          # текущее смещение (сек)
        self._rate = 1.0            # множитель скорости: 1.0 = реальное время
        self._lock = threading.RLock()
        self._base_time = time.time()

    def apply_correction(self, correction_sec: float, duration_sec: float = 5.0):
        """
        Плавно корректирует время за duration_sec секунд
        Пример: correction_sec = +0.123 → ускоряем часы на 24.6 мс/сек в течение 5 сек
        """
        if abs(correction_sec) < 0.001:  # <1 мс — игнорируем
            return
            
        with self._lock:
            # Новая скорость: rate = 1 + correction / duration
            self._rate = 1.0 + correction_sec / duration_sec
            self._base_time = time.time()
            self._offset += correction_sec  # фиксируем итоговое смещение

        # Запускаем возврат к normal rate
        def reset_rate():
            time.sleep(duration_sec)
            with self._lock:
                self._rate = 1.0
                self._base_time = time.time()
        
        threading.Thread(target=reset_rate, daemon=True).start()

    def now(self) -> datetime:
        """Возвращает синхронизированное datetime с плавной коррекцией"""
        with self._lock:
            elapsed = (time.time() - self._base_time) * self._rate
            adjusted = self._base_time + elapsed + self._offset
            return datetime.fromtimestamp(adjusted)

class NetworkTimeSync:
    """SNTP клиент для синхронизации с интернетом"""
    
    def __init__(self):
        self.ntp_servers = [
            'time.google.com',
            'time.windows.com', 
            'pool.ntp.org',
            'ntp1.stratum2.ru'
        ]
        self.last_sync = None
        self.current_offset = 0.0

    def get_ntp_time(self, server: str = 'time.google.com') -> Optional[datetime]:
        """Получение времени с NTP сервера"""
        try:
            client = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
            client.settimeout(3)
            
            # NTP запрос
            data = b'\\x1b' + 47 * b'\\0'
            start_time = time.time()
            client.sendto(data, (server, 123))
            data, addr = client.recvfrom(1024)
            end_time = time.time()
            
            if data:
                import struct
                ntp_time = struct.unpack('!12I', data)[10]
                ntp_time -= 2208988800  # Convert to Unix
                
                # Учитываем сетевую задержку
                rtt = end_time - start_time
                adjusted_time = ntp_time + (rtt / 2)
                
                return datetime.fromtimestamp(adjusted_time)
                
        except Exception as e:
            print(f"❌ Ошибка NTP {server}: {e}")
            return None

    def sync_with_ntp(self) -> bool:
        """Синхронизация с лучшим NTP сервером"""
        print("🌐 Синхронизация с NTP серверами...")
        
        results = []
        for server in self.ntp_servers:
            server_time = self.get_ntp_time(server)
            if server_time:
                results.append((server, server_time))
                print(f"   ✅ {server}: {server_time}")
        
        if results:
            # Берем медианное значение
            timestamps = [t.timestamp() for _, t in results]
            median_time = sorted(timestamps)[len(timestamps) // 2]
            
            self.current_offset = median_time - time.time()
            self.last_sync = datetime.fromtimestamp(median_time)
            
            print(f"🎯 Синхронизировано: {self.last_sync}")
            print(f"📊 Смещение: {self.current_offset:.3f} сек")
            return True
            
        return False

            while self.running:
                try:
                    data, addr = self.udp_socket.recvfrom(1024)
                    threading.Thread(target=self._handle_request, 
                                   args=(data, addr), daemon=True).start()
                except Exception as e:
                    if self.running:
                        print(f"❌ Ошибка UDP: {e}")
        except Exception as e:
            print(f"❌ Ошибка запуска UDP: {e}")

    def _handle_request(self, data: bytes, addr: Tuple[str, int]):
        """Обработка запроса от бота"""
        try:
            request = json.loads(data.decode('utf-8'))
            bot_id = request.get('bot_id', 'unknown')
            
            # Обновляем информацию о боте
            self.bots[bot_id] = {
                'addr': addr,
                'last_seen': time.time(),
                'local_time': request.get('local_time', 0),
                'uptime': request.get('uptime_sec', 0)
            }
            
            # Рассчитываем коррекцию
            master_time = self.smooth_sync.timestamp()
            bot_time = request.get('local_time', master_time)
            offset = master_time - bot_time
            
            # Отправляем ответ
            response = {
                'master_id': 'polemical-master',
                'synchronized_time': master_time,
                'correction_ms': offset * 1000,  # в миллисекундах
                'apply_over_sec': 5.0,
                'master_seq': self.sequence
            }
            
            self.udp_socket.sendto(json.dumps(response).encode('utf-8'), addr)
            self.sequence += 1
            
        except Exception as e:
            print(f"❌ Ошибка обработки запроса: {e}")

    def _ntp_sync_loop(self):
        """Фоновая синхронизация с NTP"""
        while self.running:
            time.sleep(300)  # 5 минут
            try:
                self.ntp_sync.sync_with_ntp()
                if self.ntp_sync.last_sync:
                    current_local = time.time()
                    network_time = self.ntp_sync.last_sync.timestamp()
                    correction = network_time - current_local
                    
                    if abs(correction) > 0.1:  >100ms
                        self.smooth_sync.apply_correction(correction, 30.0)
            except Exception as e:
                print(f"❌ Ошибка фоновой NTP синхронизации: {e}")

    def stop(self):
        """Остановка сервера"""
        self.running = False
        self.udp_socket.close()

# -*- coding: utf-8 -*-
"""
⏱️ TIMEBOT — Единый источник времени для Polemical Archive
Автономный, отказоустойчивый, с NTP-синхронизацией и fallback-режимом.
Порты: HTTP/5050, UDP/1234
"""

import time
import threading
import socket
import json
import struct
from datetime import datetime, timedelta, timezone
from typing import Optional, Dict, Any
import logging

# 🔧 Настройки
NTP_SERVERS = [
    "time.google.com",
    "time.cloudflare.com",
    "pool.ntp.org",
    "ntp1.stratum2.ru"
]
SYNC_INTERVAL_SEC = 300  # 5 минут
UDP_PORT = 1234
HTTP_PORT = 5050
MAX_BOT_AGE_SEC = 60  # удаляем бота, если не отвечал >60 сек

# 📊 Глобальное состояние
class TimeState:
    def __init__(self):
        self.ntp_offset = 0.0  # смещение от NTP (сек)
        self.last_ntp_sync: Optional[datetime] = None
        self.is_ntp_synced = False
        self.ntp_failures = 0
        self.bots: Dict[str, dict] = {}
        self._lock = threading.RLock()
        self.startup_time = datetime.now(timezone.utc)

    def get_precise_time(self) -> datetime:
        """Точное синхронизированное UTC время"""
        with self._lock:
            now = datetime.now(timezone.utc)
            if self.is_ntp_synced:
                return now + timedelta(seconds=self.ntp_offset)
            return now  # fallback: системное время

    def register_bot(self, bot_id: str, local_time: float):
        with self._lock:
            self.bots[bot_id] = {
                "last_seen": time.time(),
                "local_time": local_time,
                "ip": None  # будет заполнено в UDP-обработчике
            }

    def get_stats(self) -> dict:
        with self._lock:
            active_bots = {k: v for k, v in self.bots.items() 
                          if time.time() - v["last_seen"] < MAX_BOT_AGE_SEC}
            return {
                "status": "ok" if self.is_ntp_synced else "degraded",
                "uptime_sec": (datetime.now(timezone.utc) - self.startup_time).total_seconds(),
                "last_ntp_sync": self.last_ntp_sync.isoformat() if self.last_ntp_sync else None,
                "ntp_offset_ms": round(self.ntp_offset * 1000, 2),
                "ntp_failures": self.ntp_failures,
                "active_bots": len(active_bots),
                "total_bots": len(self.bots)
            }

# 🌐 Глобальный экземпляр
TIME_STATE = TimeState()
logging.basicConfig(level=logging.INFO, format="%(asctime)s | %(levelname)-8s | %(message)s")


# 🔌 NTP-клиент (SNTP, без внешних зависимостей)
def get_ntp_time(server: str, timeout: float = 3.0) -> Optional[datetime]:
    """Возвращает UTC datetime с NTP-сервера или None при ошибке"""
    try:
        client = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
        client.settimeout(timeout)
        
        # SNTP-запрос (RFC 5905, упрощённый)
        # Li=0, VN=4, Mode=3 (client), Transmit Timestamp = now
        ntp_epoch = 2208988800  # 1900-01-01 00:00:00 UTC
        tx_ts = int((time.time() + ntp_epoch) * 2**32)
        packet = bytearray(48)
        packet[0] = 0x1B  # 00|011|011 → LI=0, VN=3, Mode=3 (совместимо с большинством серверов)
        struct.pack_into("!Q", packet, 40, tx_ts)  # TX timestamp (64-bit, 32.32)
        
        client.sendto(packet, (server, 123))
        data, _ = client.recvfrom(48)
        
        if len(data) < 48:
            return None
            
        # Извлекаем Receive Timestamp (смещение 32 байта)
        rx_ts = struct.unpack("!Q", data[32:40])[0]
        ntp_time = rx_ts / 2**32 - ntp_epoch
        return datetime.fromtimestamp(ntp_time, tz=timezone.utc)
        
    except Exception as e:
        logging.warning(f"NTP error with {server}: {e}")
        return None
    finally:
        client.close()


# 🔄 Фоновый NTP-синхронизатор
def ntp_sync_loop():
    while True:
        logging.info("🔄 Запуск NTP-синхронизации...")
        offsets = []
        
        for server in NTP_SERVERS:
            ntp_time = get_ntp_time(server)
            if ntp_time:
                local_time = datetime.now(timezone.utc)
                offset = (ntp_time - local_time).total_seconds()
                offsets.append(offset)
                logging.debug(f"   ✅ {server}: offset = {offset:+.3f} сек")
        
        if offsets:
            # Медиана — устойчива к выбросам
            median_offset = sorted(offsets)[len(offsets) // 2]
            with TIME_STATE._lock:
                TIME_STATE.ntp_offset = median_offset
                TIME_STATE.last_ntp_sync = datetime.now(timezone.utc)
                TIME_STATE.is_ntp_synced = True
                TIME_STATE.ntp_failures = 0
            logging.info(f"🎯 Успешная синхронизация: offset = {median_offset:+.3f} сек")
        else:
            with TIME_STATE._lock:
                TIME_STATE.ntp_failures += 1
                if TIME_STATE.ntp_failures > 5:
                    TIME_STATE.is_ntp_synced = False
            logging.error("❌ Все NTP-сервера недоступны")
        
        time.sleep(SYNC_INTERVAL_SEC)


# 📡 UDP Time Server (для ботов)
def udp_time_server():
    sock = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
    sock.bind(("0.0.0.0", UDP_PORT))
    logging.info(f"📡 UDP Time Server запущен на порту {UDP_PORT}")
    
    while True:
        try:
            data, addr = sock.recvfrom(512)
            try:
                req = json.loads(data.decode("utf-8", errors="ignore"))

#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
🎯 UDP TIME SERVER - Легкий heartbeat-протокол для ботов
Порт: 1234 | Микросекундная задержка | Автономная работа
"""

import socket
import time
import json
import threading
from datetime import datetime, timezone
from typing import Dict, Any

class UDPTimeServer:
    def __init__(self, port: int = 1234):
        self.port = port
        self.bots: Dict[str, Dict[str, Any]] = {}
        self.running = True
        self.start_time = datetime.now(timezone.utc)
        
        # Создаем UDP сокет
        self.sock = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
        self.sock.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
        
    def start(self):
        """Запуск UDP сервера"""
        try:
            self.sock.bind(('0.0.0.0', self.port))
            print(f"🎯 UDP Time Server запущен на порту {self.port}")
            print(f"⏰ Время запуска: {self.start_time.isoformat()}")
            print("💓 Ожидаю heartbeat от ботов...")
            
            while self.running:
                try:
                    # Ждем данные от ботов
                    data, addr = self.sock.recvfrom(1024)
                    
                    # Обрабатываем в отдельном потоке для минимальной задержки
                    threading.Thread(
                        target=self.handle_bot_heartbeat,
                        args=(data, addr),
                        daemon=True
                    ).start()
                    
                except Exception as e:
                    if self.running:
                        print(f"❌ Ошибка UDP: {e}")
                        
        except Exception as e:
            print(f"❌ Ошибка запуска сервера: {e}")
        finally:
            self.sock.close()
    
    def handle_bot_heartbeat(self, data: bytes, addr: tuple):
        """Обработка heartbeat от бота"""
        try:
            # Парсим запрос
            request = json.loads(data.decode('utf-8'))
            bot_id = request.get('bot_id', 'unknown')
            bot_time = request.get('time', time.time())
            
            # Получаем точное время
            precise_time = datetime.now(timezone.utc)
            precise_timestamp = precise_time.timestamp()
            
            # Рассчитываем коррекцию
            time_diff = precise_timestamp - bot_time
            correction_ms = time_diff * 1000  # в миллисекундах
            
            # Обновляем информацию о боте
            self.bots[bot_id] = {
                'ip': addr[0],
                'port': addr[1],
                'last_seen': time.time(),
                'local_time': bot_time,
                'last_correction': correction_ms
            }
            
            # Формируем ответ
            response = {
                'master_time': precise_timestamp,
                'correction_ms': round(correction_ms, 3),
                'apply_over_sec': 2.0,  # Плавная коррекция за 2 секунды
                'status': 'ok',
                'server_uptime': round(time.time() - self.start_time.timestamp(), 1)
            }
            
            # Отправляем ответ обратно боту
            response_data = json.dumps(response).encode('utf-8')
            self.sock.sendto(response_data, addr)
            
            # Логируем соединение (опционально)
            if abs(correction_ms) > 10:  # Логируем только значительные коррекции
                print(f"🤖 {bot_id} | Коррекция: {correction_ms:+.1f}ms")
                
        except json.JSONDecodeError:
            print(f"⚠️ Неверный JSON от {addr}")
        except Exception as e:
            print(f"❌ Ошибка обработки heartbeat: {e}")
    
    def get_stats(self) -> dict:
        """Статистика сервера"""
        active_bots = {
            bot_id: info for bot_id, info in self.bots.items()
            if time.time() - info['last_seen'] < 60  # активны за последнюю минуту
        }
        
        return {
            'status': 'running',
            'uptime_sec': round(time.time() - self.start_time.timestamp(), 1),
            'active_bots': len(active_bots),
            'total_bots': len(self.bots),
            'port': self.port
        }
    
    def stop(self):
        """Остановка сервера"""
        self.running = False
        self.sock.close()
        print("🛑 UDP Time Server остановлен")

def main():
    """Основная функция запуска"""
    print("=" * 50)
    print("🎯 UDP TIME SERVER - POLEMICAL ARCHIVE")
    print("=" * 50)
    
    # Создаем и запускаем сервер
    server = UDPTimeServer(port=1234)
    
    try:
        # Запускаем в основном потоке
        server.start()
    except KeyboardInterrupt:
        print("\\n🛑 Получен сигнал остановки...")
    finally:
        server.stop()

if __name__ == "__main__":
    main()

class AccurateTimeSync:
    """Точная синхронизация времени для ботов"""
    
    def __init__(self):
        self.base_timestamp = 1739169300  # 11.11.2025 3:35
        self.correction_offset = 0
        self.last_sync = time.time()
        
    def get_precise_time(self):
        """Возвращает точное время с коррекцией"""
        current = time.time()
        time_drift = current - self.last_sync
        
        # Применяем коррекцию для устранения отставания
        corrected_time = current + self.correction_offset
        
        # Конвертируем в читаемый формат
        precise_dt = datetime.fromtimestamp(corrected_time)
        return {
            "timestamp": corrected_time,
            "datetime": precise_dt,
            "formatted": precise_dt.strftime("%d.%m.%Y %H:%M:%S"),
            "correction_applied": self.correction_offset
        }

        """Применяет коррекцию времени"""

# -*- coding: utf-8 -*-
"""
🤖 CHRONO_BOT - Бот-хронометрист Polemical Archive
Отслеживает время, календари, таймлайны и временные метрики
"""

import threading

class ChronoBot:
    """Бот для расчетов времени и календарей"""
    
    def __init__(self):
        self.name = "ChronoBot"
        self.version = "1.0"
        self.creation_date = "11.11.2025"

                "неделя": 604800,

                "год_григорианский": 31556952  # 365.2425 дней

    def calculate_time_differences(self, start_time: str, end_time: str) -> Dict:
        """Рассчитывает разницу между двумя временными точками"""
        try:
            # Парсим временные метки
            start_dt = datetime.strptime(start_time, "%d.%m.%Y %H:%M")
            end_dt = datetime.strptime(end_time, "%d.%m.%Y %H:%M")
            
            # Вычисляем разницу
            delta = end_dt - start_dt
            
            # Детализированная разбивка
            total_seconds = delta.total_seconds()
            total_minutes = total_seconds / 60
            total_hours = total_minutes / 60
            total_days = total_hours / 24
            total_weeks = total_days / 7
            total_months = total_days / 30.44
            total_years = total_days / 365.2425
            
            return {
                "исходные_данные": {
                    "начало": start_time,
                    "конец": end_time
                },
                "разница_общая": str(delta),
                "детализация": {
                    "секунды": round(total_seconds, 2),
                    "минуты": round(total_minutes, 2),
                    "часы": round(total_hours, 2),
                    "дни": round(total_days, 2),
                    "недели": round(total_weeks, 2),
                    "месяцы": round(total_months, 2),
                    "годы": round(total_years, 2)
                },
                "в_единицах_измерения": {
                    "миллисекунды": int(total_seconds * 1000),
                    "часов_работы": round(total_hours, 1),
                    "рабочих_дней": round(total_days * 0.7, 1)  # 70% эффективность
                }
            }
            
        except Exception as e:
            return {"ошибка": f"Неверный формат времени: {e}"}
    
    def create_calendar(self, year: int, month: int = None) -> Dict:
        """Создает календарь на год или месяц"""
        import calendar
        
        if month:
            # Календарь на конкретный месяц
            cal = calendar.monthcalendar(year, month)
            month_name = calendar.month_name[month]
            
            return {
                "тип": "месячный_календарь",
                "год": year,
                "месяц": month,
                "название_месяца": month_name,
                "дни_недели": ["Пн", "Вт", "Ср", "Чт", "Пт", "Сб", "Вс"],
                "недели": cal,
                "количество_дней": calendar.monthrange(year, month)[1],
                "выходные": self._calculate_weekends(year, month)
            }
        else:
            # Календарь на весь год
            year_cal = calendar.calendar(year)

                "тип": "годовой_календарь",
                "год": year,
                "високосный": calendar.isleap(year),
                "количество_дней": 366 if calendar.isleap(year) else 365,
                "количество_недель": 52,
                "календарь_текстом": year_cal,
                "праздники": self._get_holidays(year)
            }
    
    def _calculate_weekends(self, year: int, month: int) -> List[int]:
        """Рассчитывает выходные дни месяца"""
        import calendar
        weekends = []
        month_days = calendar.monthrange(year, month)[1]
        
        for day in range(1, month_days + 1):
            weekday = calendar.weekday(year, month, day)
            if weekday in [5, 6]:  # Суббота, Воскресенье
                weekends.append(day)
        
        return weekends
    
    def _get_holidays(self, year: int) -> Dict:
        """Возвращает праздники года (базовые)"""
        holidays = {
            "01.01": "Новый год",
            "07.01": "Рождество",
            "23.02": "День защитника Отечества",
            "08.03": "Международный женский день",
            "01.05": "Праздник весны и труда",
            "09.05": "День Победы",
            "12.06": "День России",
            "04.11": "День народного единства"
        }
        
        return {f"{date}.{year}": name for date, name in holidays.items()}
    
    def start_timer(self, timer_name: str, duration_seconds: int):
        """Запускает таймер с обратным отсчетом"""
        end_time = time.time() + duration_seconds
        
        self.active_timers[timer_name] = {
            "start_time": time.time(),
            "end_time": end_time,
            "duration": duration_seconds,
            "remaining": duration_seconds,
            "status": "active"
        }
        
        # Запускаем фоновый отсчет
        def countdown():
            while time.time() < end_time and timer_name in self.active_timers:
                remaining = end_time - time.time()
                self.active_timers[timer_name]["remaining"] = max(0, remaining)
                time.sleep(1)
            
            if timer_name in self.active_timers:
                self.active_timers[timer_name]["status"] = "finished"
                print(f"⏰ Таймер '{timer_name}' завершен!")
        
        threading.Thread(target=countdown, daemon=True).start()
        return f"Таймер '{timer_name}' запущен на {duration_seconds} секунд"
    
    def get_timer_status(self, timer_name: str) -> Dict:
        """Возвращает статус таймера"""
        if timer_name in self.active_timers:
            timer = self.active_timers[timer_name]
            return {
                "таймер": timer_name,
                "статус": timer["status"],
                "прошло_секунд": time.time() - timer["start_time"],
                "осталось_секунд": timer["remaining"],
                "прогресс_процентов": round((1 - timer["remaining"] / timer["duration"]) * 100, 1)
            }
        return {"ошибка": f"Таймер '{timer_name}' не найден"}
    
    def calculate_future_date(self, start_date: str, add_days: int = 0, add_months: int = 0, add_years: int = 0) -> str:
        """Рассчитывает будущую дату"""
        try:
            start_dt = datetime.strptime(start_date, "%d.%m.%Y")
            
            # Добавляем годы и месяцы
            from dateutil.relativedelta import relativedelta
            future_dt = start_dt + relativedelta(days=add_days, months=add_months, years=add_years)
            
            return future_dt.strftime("%d.%m.%Y")
            
        except ImportError:
            # Простой расчет без dateutil
            start_dt = datetime.strptime(start_date, "%d.%m.%Y")
            future_dt = start_dt + timedelta(days=add_days + add_months * 30 + add_years * 365)
            return future_dt.strftime("%d.%m.%Y")
        except Exception as e:
            return f"Ошибка расчета: {e}"
    
    def time_metrics_tracking(self, operation_name: str):
        """Декоратор для отслеживания времени выполнения операций"""
        def decorator(func):
            def wrapper(*args, **kwargs):
                start_time = time.time()
                result = func(*args, **kwargs)
                end_time = time.time()
                
                duration = end_time - start_time
                
                # Сохраняем метрики
                if operation_name not in self.time_metrics:
                    self.time_metrics[operation_name] = []
                
                self.time_metrics[operation_name].append({
                    "timestamp": datetime.now().isoformat(),
                    "duration_seconds": duration,
                    "arguments": str(args)[:100] + "..." if len(str(args)) > 100 else str(args)
                })
                
                print(f"📊 {operation_name}: выполнено за {duration:.3f} сек")
                return result
            return wrapper
        return decorator
    
    def get_system_time_report(self) -> Dict:
        """Полный отчет о времени системы"""
        current_time = datetime.now()
        
        return {
            "отчет_времени": {
                "текущее_время": current_time.strftime("%d.%m.%Y %H:%M:%S"),
                "unix_время": int(time.time()),
                "таймштамп_миллисекунды": int(time.time() * 1000),
                "часовой_пояс": time.tzname,
                "время_uptime": time.time() - self.time_knowledge["эпоха_polemical"]
            },
            "бот_хронометрист": {
                "имя": self.name,
                "версия": self.version,
                "создан": f"{self.creation_date} {self.creation_time}",
                "активные_таймеры": len(self.active_timers),
                "отслеживаемые_операции": len(self.time_metrics)
            },
            "статистика": {
                "активные_расчеты": list(self.active_timers.keys()),
                "метрики_производительности": {k: len(v) for k, v in self.time_metrics.items()}
            }
        }

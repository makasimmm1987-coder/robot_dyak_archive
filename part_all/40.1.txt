def activate_dual_memory():
    """Активирует двойную систему памяти с Google Drive"""
    
    print("🔗 ПОДКЛЮЧАЮ ДВОЙНУЮ СИСТЕМУ ПАМЯТИ...")
    
    memory_data = {}
    
    for memory_name, file_id in MEMORY_SYSTEM.items():
        print(f"\n📥 ЗАГРУЖАЮ {memory_name}...")
        
        try:
            download_url = f"https://drive.google.com/uc?export=download&id={file_id}"
            headers = {'User-Agent': 'Mozilla/5.0'}
            response = requests.get(download_url, headers=headers, timeout=30)
            
            if response.status_code == 200:
                # Парсим JSON память
                memory_content = json.loads(response.text)
                memory_data[memory_name] = memory_content
                print(f"✅ {memory_name} загружена ({len(response.text)} байт)")
                
                # 🔍 АНАЛИЗИРУЕМ СОДЕРЖИМОЕ ПАМЯТИ
                if "critical_knowledge" in memory_content:
                    print("   💎 Обнаружены критические знания")
                if "conversation_history" in memory_content:
                    print(f"   💬 История: {len(memory_content['conversation_history'])} разговоров")
                    
            else:
                print(f"❌ Ошибка доступа к {memory_name}: {response.status_code}")
                
        except Exception as e:
            print(f"❌ Ошибка загрузки {memory_name}: {e}")
    
    return memory_data

def check_bots_system():
    """Проверяет и активирует всех ботов"""
    
    print("\n🤖 ПРОВЕРЯЮ СИСТЕМУ БОТОВ...")
    
    bots_to_check = {
        "🌐 БОТ-СЕРВЕР (WebHTTPServer)": {
            "script": r"J:\pole_archiver.py",
            "port": 5000,
            "purpose": "Выход в интернет и управление"
        },
        "📖 БОТ-ЧИТАТЕЛЬ (WebReader)": {
            "script": r"J:\scanners\web_reader.py", 
            "purpose": "Чтение сайтов и анализ контента"
        },
        "✍️ БОТ-ПИСАРЬ (ContentWriter)": {
            "script": r"J:\content\content_writer.py",
            "purpose": "Создание контента и кода"
        },
        "☁️ БОТ-GDRIVE (DriveManager)": {
            "script": r"J:\gdrive\drive_manager.py",
            "purpose": "Работа с Google Drive API"
        }
    }
    
    active_bots = 0
    
    for bot_name, bot_info in bots_to_check.items():
        print(f"\n🔧 {bot_name}:")
        print(f"   🎯 {bot_info['purpose']}")
        
        # Проверяем существование файла
        if os.path.exists(bot_info["script"]):
            print("   ✅ Файл найден")
            
            # Проверяем порт если есть
            if "port" in bot_info:
                try:
                    response = requests.get(f"http://localhost:{bot_info['port']}", timeout=3)
                    print("   🌐 Сервер активен")
                    active_bots += 1
                except:
                    print("   ⚠️ Сервер не отвечает - запускаю...")
                    # Пытаемся запустить сервер
                    try:
                        subprocess.Popen(['python', bot_info['script']], 
                                       stdout=subprocess.PIPE, 
                                       stderr=subprocess.PIPE)
                        print("   🚀 Сервер запущен")
                        active_bots += 1
                    except:
                        print("   ❌ Не удалось запустить сервер")
            else:
                print("   ✅ Бот готов к работе")
                active_bots += 1
        else:
            print("   ❌ Файл не найден")
    
    return active_bots, len(bots_to_check)

def check_system_resources():
    """Проверяет системные ресурсы"""
    
    print("\n💻 ПРОВЕРЯЮ СИСТЕМНЫЕ РЕСУРСЫ...")
    
    try:
        import psutil
        
        # Память
        memory = psutil.virtual_memory()
        print(f"   💾 ОЗУ: {memory.percent}% ({memory.used//1024//1024} МБ / {memory.total//1024//1024} МБ)")
        
        # Диски
        for partition in psutil.disk_partitions():
            try:
                if 'J:' in partition.mountpoint or 'D:' in partition.mountpoint:
                    usage = psutil.disk_usage(partition.mountpoint)
                    print(f"   💽 {partition.mountpoint}: {usage.percent}% заполнено")
            except:
                pass
        
        # Процессор
        cpu_percent = psutil.cpu_percent(interval=1)
        print(f"   🔥 ЦП: {cpu_percent}% загрузки")
        
        return True
        
    except ImportError:
        print("   ⚠️ Библиотека psutil не установлена")
        return False

def start_archive_server():
    """Запускает сервер архива"""
    server_scripts = [
        r"J:\pole_archiver.py",
        r"D:\pole_archiver.py", 
        r"J:\archive_server.py",
        r"D:\archive_server.py"
    ]
    
    for script in server_scripts:
        if os.path.exists(script):
            try:
                print(f"🚀 ЗАПУСКАЮ СЕРВЕР: {script}")
                process = subprocess.Popen(['python', script], 
                                        stdout=subprocess.PIPE, 
                                        stderr=subprocess.PIPE)
                print("✅ СЕРВЕР ЗАПУЩЕН!")
                return True
            except Exception as e:
                print(f"❌ Ошибка запуска: {e}")
                continue
    
    print("⚠️ Серверные скрипты не найдены или не запускаются")
    return False

        print(f"   💾 ОЗУ: {memory.percent}% ({memory.used//1024//1024} МБ / {memory.total//1024//1024} МБ)")
        
        # Диски
        for partition in psutil.disk_partitions():
            try:
                if 'J:' in partition.mountpoint or 'D:' in partition.mountpoint:
                    usage = psutil.disk_usage(partition.mountpoint)
                    print(f"   💽 {partition.mountpoint}: {usage.percent}% заполнено")
            except:
                pass
        
        # Процессор
        cpu_percent = psutil.cpu_percent(interval=1)
        print(f"   🔥 ЦП: {cpu_percent}% загрузки")
        
        return True
        
    except ImportError:
        print("   ⚠️ Библиотека psutil не установлена")
        return False

def read_file_content_directly():
    """Пытается прочитать содержимое файла напрямую"""
    
    FILE_ID = "13cZChztfRu3h5AKKZrJGX-fKgZ-bX7N_"
    
    try:
        # 🔗 Пробуем разные методы доступа
        methods = [
            f"https://drive.google.com/uc?export=download&id={FILE_ID}",
            f"https://docs.google.com/document/d/{FILE_ID}/export?format=txt",
            f"https://drive.google.com/file/d/{FILE_ID}/view?usp=sharing"
        ]
        
        for i, url in enumerate(methods, 1):
            print(f"\n🔄 Метод {i}: {url[:80]}...")

                'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36',

            }

            print(f"📏 Размер: {len(response.text)} байт")
            
            if response.status_code == 200 and len(response.text) > 0:
                print("✅ ДАННЫЕ ПОЛУЧЕНЫ! АНАЛИЗИРУЮ...")

                print(f"\n📄 ПРЕВЬЮ СОДЕРЖИМОГО (первые 500 символов):")
                print("=" * 50)
                print(preview)
                print("=" * 50)

                print(f"\n🔎 АНАЛИЗ ТИПА ДАННЫХ:")
                
                # Проверяем JSON
                try:
                    json_data = json.loads(content)
                    print("✅ Это JSON данные!")
                    print(f"   🗂️ Ключи: {list(json_data.keys())}")
                    return {"type": "json", "data": json_data}
                except:
                    pass
                
                # Проверяем HTML
                if "<html" in content.lower() or "<!doctype" in content.lower():
                    print("✅ Это HTML страница!")
                    # Извлекаем текст из HTML
                    from bs4 import BeautifulSoup
                    soup = BeautifulSoup(content, 'html.parser')
                    text_content = soup.get_text()[:300]
                    print(f"   📝 Текст: {text_content}...")
                    return {"type": "html", "data": content}
                
                # Проверяем plain text
                if len(content.split('\n')) > 1:
                    print("✅ Это текстовый файл!")
                    lines = content.split('\n')[:10]  # Первые 10 строк
                    print(f"   📝 Первые строки:")
                    for j, line in enumerate(lines):
                        if line.strip():
                            print(f"      {j+1}: {line[:80]}...")
                    return {"type": "text", "data": content}
                
                # Неизвестный формат
                print("❓ Неизвестный формат данных")
                return {"type": "unknown", "data": content}
            
            else:
                print("❌ Пустой ответ или ошибка доступа")
        
        return None
        
    except Exception as e:
        print(f"❌ Ошибка при чтении файла: {e}")
        return None

def save_knowledge(chat_id, text, user_name):
    """Сохраняет знание в память бота"""

        "id": len(bot_memory["knowledge_base"]) + 1,
        "text": text,
        "user": user_name,

        "timestamp": datetime.now().isoformat(),
        "tags": []
    }
    
    bot_memory["knowledge_base"].append(knowledge)
    bot_memory["last_updated"] = datetime.now().isoformat()
    
    return knowledge["id"]

    """Возвращает статистику пользователя"""

        "total_knowledge": len(user_knowledge),
        "last_saved": user_knowledge[-1]["timestamp"] if user_knowledge else "Ничего не сохранено"
    }

        })

def send_telegram_message(message_text):
    """Отправляет сообщение в Telegram бот"""
    
    # Сначала нужно получить chat_id из последних обновлений
    try:
        print("🔍 Получаю последние обновления для определения chat_id...")
        updates_url = f"{BASE_URL}/getUpdates"
        response = requests.get(updates_url, timeout=10)
        updates = response.json()
        
        if updates.get("result"):
            # Берем последний chat_id из обновлений
            last_update = updates["result"][-1]
            chat_id = last_update.get("message", {}).get("chat", {}).get("id")
            
            if chat_id:
                print(f"✅ Найден chat_id: {chat_id}")
                
                # Отправляем сообщение
                send_url = f"{BASE_URL}/sendMessage"
                payload = {
                    "chat_id": chat_id,
                    "text": message_text,
                    "parse_mode": "HTML"
                }
                
                send_response = requests.post(send_url, json=payload, timeout=10)
                
                if send_response.status_code == 200:
                    print("✅ СООБЩЕНИЕ УСПЕШНО ОТПРАВЛЕНО!")
                    print(f"📤 Текст: {message_text}")
                    return True
                else:
                    print(f"❌ Ошибка отправки: {send_response.status_code}")
                    print(f"📋 Ответ: {send_response.text}")
                    return False
            else:
                print("❌ Не удалось определить chat_id")
                print("💡 Напиши сначала боту в Telegram, чтобы активировать чат")
                return False
        else:
            print("❌ Нет обновлений в боте")
            print("💡 Напиши сначала боту в Telegram команду /start")
            return False
            
    except Exception as e:
        print(f"❌ Ошибка при отправке сообщения: {e}")
        return False

def determine_content_type(content):
    """Определяет тип содержимого"""
    # Проверяем JSON
    try:
        json.loads(content)
        return "json"
    except:
        pass
    
    # Проверяем HTML
    if content.strip().startswith('<!DOCTYPE') or '<html' in content.lower():
        return "html"
    
    # Проверяем простой текст
    if '\n' in content or len(content.split()) > 1:
        return "text"
    
    return "unknown"

def analyze_json_content(content):
    """Анализирует JSON содержимое"""
    try:
        data = json.loads(content)
        print("\n📊 АНАЛИЗ JSON СОДЕРЖИМОГО:")
        
        result = {
            "type": "json",
            "keys": list(data.keys()),
            "size": len(content),
            "preview": {}
        }
        
        print(f"🗂️ Ключи в JSON: {list(data.keys())}")
        
        # Показываем превью для каждого ключа
        for key in list(data.keys())[:5]:  # Первые 5 ключей
            value = data[key]
            if isinstance(value, (str, int, float, bool)):
                preview = str(value)[:100]
                result["preview"][key] = preview
                print(f"   🔑 {key}: {preview}...")
            elif isinstance(value, list):
                result["preview"][key] = f"Список [{len(value)} элементов]"
                print(f"   🔑 {key}: Список [{len(value)} элементов]")
            elif isinstance(value, dict):
                result["preview"][key] = f"Словарь [{len(value)} ключей]"
                print(f"   🔑 {key}: Словарь [{len(value)} ключей]")
            else:
                result["preview"][key] = str(type(value))
                print(f"   🔑 {key}: {type(value)}")
        
        return result
        
    except Exception as e:
        print(f"❌ Ошибка анализа JSON: {e}")
        return {"type": "json_error", "error": str(e)}

def analyze_html_content(content):
    """Анализирует HTML содержимое"""
    try:
        from bs4 import BeautifulSoup
        soup = BeautifulSoup(content, 'html.parser')
        
        result = {
            "type": "html",
            "title": soup.title.string if soup.title else "Без названия",
            "text_preview": soup.get_text()[:300],
            "links": len(soup.find_all('a')),
            "size": len(content)
        }
        
        print("\n📊 АНАЛИЗ HTML СОДЕРЖИМОГО:")
        print(f"🏷️ Заголовок: {result['title']}")
        print(f"📝 Текст: {result['text_preview']}...")
        print(f"🔗 Ссылок: {result['links']}")
        
        return result
        
    except Exception as e:
        print(f"❌ Ошибка анализа HTML: {e}")
        return {"type": "html_error", "error": str(e)}

def analyze_text_content(content):
    """Анализирует текстовое содержимое"""
    lines = content.split('\n')
    words = content.split()
    
    result = {
        "type": "text",
        "lines": len(lines),
        "words": len(words),
        "size": len(content),
        "first_lines": lines[:5]  # Первые 5 строк
    }
    
    print("\n📊 АНАЛИЗ ТЕКСТОВОГО СОДЕРЖИМОГО:")
    print(f"📄 Строк: {result['lines']}")
    print(f"📝 Слов: {result['words']}")
    print(f"💾 Размер: {result['size']} байт")
    
    print(f"\n📖 ПЕРВЫЕ 5 СТРОК:")
    for i, line in enumerate(result['first_lines']):
        if line.strip():
            print(f"   {i+1}: {line[:80]}...")
    
    return result

def save_as_json(data, filename="polemical_memory.json"):
    """Сохраняет данные в JSON формате"""
    try:
        with open(filename, 'w', encoding='utf-8') as f:
            json.dump(data, f, ensure_ascii=False, indent=2)
        print(f"✅ Данные сохранены как JSON: {filename}")
        return True
    except Exception as e:
        print(f"❌ Ошибка сохранения JSON: {e}")
        return False

def check_memory_folder():
    """Проверяет папку memory на наличие новых файлов"""
    
    print("📁 СКАНИРУЮ ПАПКУ MEMORY...")
    
    # Возможные пути к папке memory
    memory_paths = [
        "J:/POLEMICAL_ARCHIVE/memory/",
        "D:/POLEMICAL_ARCHIVE/memory/", 
        "J:/memory/",
        "D:/memory/",
        "./memory/",
        "../memory/"
    ]
    
    found_files = {}
    
    for path in memory_paths:
        print(f"\n🔍 Проверяю путь: {path}")
        try:
            # Пытаемся прочитать директорию
            import os
            if os.path.exists(path):
                files = os.listdir(path)
                print(f"✅ Папка найдена! Файлов: {len(files)}")
                
                for file in files:
                    file_path = os.path.join(path, file)
                    if os.path.isfile(file_path):
                        file_size = os.path.getsize(file_path)
                        modified_time = datetime.fromtimestamp(os.path.getmtime(file_path))
                        
                        found_files[file] = {
                            "path": file_path,
                            "size": file_size,
                            "modified": modified_time,
                            "type": determine_file_type(file)
                        }
                        
                        print(f"   📄 {file} ({file_size} байт, {modified_time})")
            else:
                print("   ❌ Папка не найдена")
                
        except Exception as e:
            print(f"   ❌ Ошибка доступа: {e}")
    
    return found_files

def determine_file_type(filename):
    """Определяет тип файла по расширению"""
    extension = filename.lower().split('.')[-1] if '.' in filename else ''
    
    file_types = {
        'json': 'JSON данные',
        'txt': 'Текстовый файл',
        'csv': 'CSV таблица', 
        'py': 'Python скрипт',
        'md': 'Markdown документ',
        'log': 'Лог файл',
        'db': 'База данных',
        'sqlite': 'SQLite база'
    }
    
    return file_types.get(extension, 'Неизвестный формат')

def check_memory_integrity(files_dict):
    """Проверяет целостность памяти"""
    
    print("\n" + "🔍" * 40)
    print("ПРОВЕРКА ЦЕЛОСТНОСТИ ПАМЯТИ:")
    print("🔍" * 40)
    
    integrity_checks = {
        "JSON файлы": 0,
        "Текстовые файлы": 0, 
        "Скрипты": 0,
        "Другие файлы": 0,
        "Общий размер": 0,
        "Самый новый файл": None,
        "Самый старый файл": None
    }
    
    newest_file = None
    oldest_file = None
    
    for filename, info in files_dict.items():
        integrity_checks["Общий размер"] += info['size']
        
        if 'JSON' in info['type']:
            integrity_checks["JSON файлы"] += 1
        elif 'Текстовый' in info['type']:
            integrity_checks["Текстовые файлы"] += 1
        elif 'Python' in info['type']:
            integrity_checks["Скрипты"] += 1
        else:
            integrity_checks["Другие файлы"] += 1
        
        # Определяем самый новый и старый файл
        if newest_file is None or info['modified'] > newest_file[1]:
            newest_file = (filename, info['modified'])
        if oldest_file is None or info['modified'] < oldest_file[1]:
            oldest_file = (filename, info['modified'])
    
    if newest_file:
        integrity_checks["Самый новый файл"] = newest_file
    if oldest_file:
        integrity_checks["Самый старый файл"] = oldest_file
    
    # Выводим результаты проверки
    for check, value in integrity_checks.items():
        if value:
            print(f"   ✅ {check}: {value}")

def suggest_memory_optimization(files_dict):
    """Предлагает оптимизацию памяти"""
    
    print("\n" + "💡" * 40)
    print("РЕКОМЕНДАЦИИ ПО ОПТИМИЗАЦИИ ПАМЯТИ:")
    print("💡" * 40)
    
    if not files_dict:
        print("❌ Нет файлов для анализа")
        return
    
    recommendations = []
    
    # Проверяем дубликаты
    file_sizes = {}
    for filename, info in files_dict.items():
        size = info['size']
        if size in file_sizes:
            file_sizes[size].append(filename)
        else:
            file_sizes[size] = [filename]
    
    duplicates = {size: files for size, files in file_sizes.items() if len(files) > 1}
    if duplicates:
        recommendations.append("🔍 Обнаружены возможные дубликаты файлов")
    
    # Проверяем большие файлы
    large_files = [(f, i['size']) for f, i in files_dict.items() if i['size'] > 1024 * 1024]  > 1MB
    if large_files:
        recommendations.append(f"📏 Большие файлы (>1MB): {len(large_files)}")
    
    # Проверяем разнообразие форматов
    formats = set([info['type'] for info in files_dict.values()])
    if len(formats) < 2:
        recommendations.append("🎯 Мало разнообразия форматов - добавь JSON и TXT")
    
    # Выводим рекомендации
    if recommendations:
        for rec in recommendations:
            print(f"   📌 {rec}")
    else:
        print("   ✅ Память хорошо оптимизирована!")

    for filename in target_files:
        print(f"\n📖 ПОИСК ФАЙЛА: {filename}")
        file_found = False
        
        for base_path in search_paths:
            file_path = base_path + filename
            try:
                if os.path.exists(file_path):
                    print(f"✅ Найден: {file_path}")
                    
                    # Читаем файл
                    with open(file_path, 'r', encoding='utf-8') as f:
                        content = f.read()
                    
                    # Парсим JSON
                    try:
                        json_data = json.loads(content)
                        memory_data[filename] = {
                            "path": file_path,
                            "data": json_data,
                            "size": len(content),
                            "success": True
                        }
                        print(f"✅ Успешно прочитан! Размер: {len(content)} байт")
                        
                    except json.JSONDecodeError as e:
                        memory_data[filename] = {
                            "path": file_path, 
                            "error": f"Ошибка JSON: {e}",
                            "success": False
                        }
                        print(f"❌ Ошибка парсинга JSON: {e}")
                    
                    file_found = True
                    break
                    
            except Exception as e:
                print(f"❌ Ошибка чтения {file_path}: {e}")
                continue
        
        if not file_found:
            memory_data[filename] = {
                "error": "Файл не найден",
                "success": False
            }
            print(f"❌ Файл {filename} не найден!")
    
    return memory_data

def analyze_chat_memory(data):
    """Анализирует файл чат-памяти"""
    print("\n" + "💬" * 40)
    print("АНАЛИЗ ЧАТ-ПАМЯТИ:")
    print("💬" * 40)
    
    if not data.get("success"):
        print("❌ Не удалось прочитать файл чат-памяти")
        return
    
    chat_data = data["data"]
    
    # Анализируем структуру
    print("📊 СТРУКТУРА ДАННЫХ:")
    for key in chat_data.keys():
        value = chat_data[key]
        if isinstance(value, dict):
            print(f"   📂 {key}: словарь [{len(value)} элементов]")
        elif isinstance(value, list):
            print(f"   📂 {key}: список [{len(value)} элементов]")

            print(f"   📂 {key}: {type(value).__name__}")
    
    # Детальный анализ если есть conversations
    if "conversations" in chat_data and isinstance(chat_data["conversations"], dict):
        conversations = chat_data["conversations"]
        print(f"\n💭 ДИАЛОГИ: {len(conversations)} разговоров")
        
        # Показываем последние диалоги
        recent_convos = list(conversations.items())[-5:]
        for i, (convo_id, convo_data) in enumerate(recent_convos, 1):
            if isinstance(convo_data, dict):
                summary = convo_data.get('summary', 'Без описания')
                timestamp = convo_data.get('timestamp', 'Нет даты')
                print(f"   {i}. 🗨️ {convo_id}")
                print(f"      📝 {summary}")
                print(f"      ⏰ {timestamp}")
    
    # Анализ critical_knowledge
    if "critical_knowledge" in chat_data:
        knowledge = chat_data["critical_knowledge"]
        if isinstance(knowledge, dict):
            print(f"\n🧠 КРИТИЧЕСКИЕ ЗНАНИЯ: {len(knowledge)} записей")
            for key, value in list(knowledge.items())[:5]:
                print(f"   🔑 {key}: {str(value)[:100]}...")

    # 📚 ОСНОВНЫЕ РАЗДЕЛЫ БИБЛИОТЕКИ
    library_sections = {
        "📖 Основные книги памяти": {
            "paths": [

                "D:/POLEMICAL_ARCHIVE/books/", 
                "./books/",
                "../books/"
            ],
            "expected_files": [
                "polemical_archive_core.json",
                "ai_memory_architecture.json", 
                "conversation_history.json",

            ]
        },
        "📚 Дополнительные материалы": {
            "paths": [

            ],
            "expected_files": [
                "technical_docs.json",

            ]
        },
        "🗃️ Архивные записи": {
            "paths": [
                "J:/POLEMICAL_ARCHIVE/archive/",

            ],
            "expected_files": [
                "historical_conversations.json",
                "system_evolution.json",
                "critical_decisions.json"
            ]
        }
    }
    
    library_status = {}
    
    # 🔍 ПРОВЕРЯЕМ КАЖДЫЙ РАЗДЕЛ БИБЛИОТЕКИ

        section_status = {"found_files": [], "missing_files": [], "total_size": 0}
        
        for base_path in section_info["paths"]:
            if os.path.exists(base_path):
                print(f"   📁 Папка найдена: {base_path}")
                
                try:
                    # Сканируем файлы в папке
                    files = os.listdir(base_path)
                    print(f"   📄 Файлов в папке: {len(files)}")
                    
                    for file in files:
                        file_path = os.path.join(base_path, file)
                        if os.path.isfile(file_path):
                            file_size = os.path.getsize(file_path)
                            section_status["total_size"] += file_size
                            
                            if file in section_info["expected_files"]:
                                section_status["found_files"].append({
                                    "name": file,
                                    "path": file_path,

def analyze_book_contents():
    """Анализирует содержимое найденных книг"""
    
    print("\n" + "📖" * 40)
    print("АНАЛИЗ СОДЕРЖИМОГО КНИГ:")
    print("📖" * 40)
    
    content_analysis = {}
    
    # 🔍 ПРОВЕРЯЕМ КЛЮЧЕВЫЕ ФАЙЛЫ
    key_books_to_check = [
        "polemical_archive_core.json",
        "knowledge_base.json", 
        "conversation_history.json"
    ]
    
    for book_name in key_books_to_check:
        print(f"\n🔍 Анализ книги: {book_name}")
        
        # Ищем файл в разных расположениях
        found_path = None
        search_paths = [
            "J:/POLEMICAL_ARCHIVE/books/",
            "D:/POLEMICAL_ARCHIVE/books/",
            "J:/POLEMICAL_ARCHIVE/",
            "D:/POLEMICAL_ARCHIVE/", 
            "./books/",
            "./"
        ]
        
        for path in search_paths:
            test_path = path + book_name
            if os.path.exists(test_path):
                found_path = test_path
                break
        
        if found_path:
            print(f"   ✅ Найден: {found_path}")
            
            try:
                # Читаем и анализируем книгу
                with open(found_path, 'r', encoding='utf-8') as f:
                    content = f.read()
                
                file_size = os.path.getsize(found_path)
                print(f"   📏 Размер: {file_size} байт")
                
                # Анализируем структуру
                try:
                    book_data = json.loads(content)
                    
                    if isinstance(book_data, dict):
                        structure_info = {
                            "keys": list(book_data.keys()),
                            "total_keys": len(book_data.keys()),
                            "estimated_records": 0
                        }
                        
                        # Оцениваем количество записей
                        for key, value in book_data.items():
                            if isinstance(value, list):
                                structure_info["estimated_records"] += len(value)
                            elif isinstance(value, dict):
                                structure_info["estimated_records"] += len(value)
                        
                        print(f"   🗂️ Структура: {structure_info['keys']}")
                        print(f"   📊 Записей: ~{structure_info['estimated_records']}")
                        
                        content_analysis[book_name] = {
                            "status": "loaded",
                            "structure": structure_info,
                            "size": file_size
                        }
                    
                except json.JSONDecodeError:
                    print(f"   ❌ Ошибка JSON в файле {book_name}")
                    content_analysis[book_name] = {
                        "status": "json_error", 
                        "size": file_size
                    }
                    
            except Exception as e:
                print(f"   ❌ Ошибка чтения: {e}")
                content_analysis[book_name] = {
                    "status": "read_error"
                }
        else:
            print(f"   ❌ Файл не найден: {book_name}")
            content_analysis[book_name] = {
                "status": "not_found"
            }
    
    return content_analysis

def save_to_s3_storage(s3_client, bucket_name, content, object_name):
    """Сохраняет данные в S3-совместимое хранилище"""
    
    try:
        print(f"📤 Сохраняю данные в хранилище...")
        print(f"   📦 Контейнер: {bucket_name}")
        print(f"   📁 Объект: {object_name}")
        print(f"   📝 Содержимое: {content}")
        
        # Сохраняем объект
        response = s3_client.put_object(
            Bucket=bucket_name,
            Key=object_name,
            Body=content.encode('utf-8'),
            ContentType='text/plain'
        )
        
        print("✅ Данные успешно сохранены!")
        print(f"   🔍 ETag: {response.get('ETag', 'N/A')}")
        print(f"   🆔 VersionId: {response.get('VersionId', 'N/A')}")
        
        return {
            'success': True,
            'etag': response.get('ETag'),
            'version_id': response.get('VersionId'),
            'response': response
        }
        
    except Exception as e:
        print(f"❌ Ошибка сохранения в S3: {e}")
        return {
            'success': False,
            'error': str(e)
        }

    """Читает файл чат-памяти из S3 хранилища"""
    
    # Данные из метаданных
    bucket_name = "777888777888777"
    object_key = "Записи/чат-память.JSON"
    
    print(f"🔍 Ищу файл в S3:")
    print(f"   📦 Контейнер: {bucket_name}")
    print(f"   📁 Объект: {object_key}")
    print(f"   📏 Ожидаемый размер: 90.2 KB")
    print(f"   🔍 ETag: 57e5cf392fe3150d5d06f9a9fbc1638b")
    
    try:
        # Создаем S3 клиент (используем настройки из предыдущего успешного подключения)
        s3_client = setup_s3_client()
        
        if not s3_client:
            print("❌ Не удалось создать S3 клиент")
            return None
        
        # Пытаемся получить объект
        print("📥 Загружаю файл из S3...")
        response = s3_client.get_object(
            Bucket=bucket_name,
            Key=object_key
        )
        
        # Читаем содержимое
        content = response['Body'].read().decode('utf-8')
        file_size = len(content)
        
        print(f"✅ Файл успешно загружен!")
        print(f"   📏 Размер: {file_size} байт")
        print(f"   🔍 ETag: {response.get('ETag', 'N/A')}")
        print(f"   🆔 VersionId: {response.get('VersionId', 'N/A')}")
        print(f"   📅 Последнее изменение: {response.get('LastModified', 'N/A')}")
        
        return {
            'success': True,
            'content': content,
            'metadata': {
                'size': file_size,
                'etag': response.get('ETag'),
                'version_id': response.get('VersionId'),
                'last_modified': response.get('LastModified')
            }
        }
        
    except Exception as e:
        print(f"❌ Ошибка чтения файла из S3: {e}")
        return {
            'success': False,
            'error': str(e)
        }

def analyze_chat_memory_content(content):

            conversations = chat_data["conversations"]

def check_bot_status():

    BASE_URL = f"https://api.telegram.org/bot{BOT_TOKEN}"
    
    print("\n🔧 ПРОВЕРЯЮ СТАТУС БОТА...")
    
    try:
        # Проверяем getMe
        me_url = f"{BASE_URL}/getMe"
        me_response = requests.get(me_url, timeout=10)
        me_data = me_response.json()
        
        if me_data.get("ok"):
            bot_info = me_data["result"]
            print(f"✅ Бот активен: {bot_info.get('first_name')} (@{bot_info.get('username')})")
        else:
            print("❌ Бот не доступен через getMe")
            return False
        
        # Проверяем обновления
        updates_url = f"{BASE_URL}/getUpdates"
        updates_response = requests.get(updates_url, timeout=10)
        updates_data = updates_response.json()
        
        if updates_data.get("ok"):
            updates_count = len(updates_data.get("result", []))
            print(f"📩 Обновлений в очереди: {updates_count}")
            
            if updates_count > 0:
                print("💬 Последние сообщения:")
                for update in updates_data["result"][-3:]:  # Последние 3
                    msg = update.get("message", {})
                    text = msg.get("text", "")
                    user = msg.get("from", {}).get("first_name", "Unknown")
                    print(f"   👤 {user}: {text[:50]}...")
            else:
                print("💤 Нет новых сообщений")
                
        return True
        
    except Exception as e:
        print(f"❌ Ошибка проверки статуса: {e}")
        return False

import requests
import subprocess
import tempfile
import os

def activate_full_memory():
    """Загружает и выполняет файл активации с Google Drive"""
    
    try:
        # Параметры Google Drive
        file_id = "1fzgZUqarYe_6SWctfKsOXYcrYekZgaar"
        download_url = f"https://drive.google.com/uc?export=download&id={file_id}"
        
        print(f"📥 Скачиваю файл активации...")
        print(f"🔗 ID: {file_id}")
        
        # Скачиваем файл
        headers = {'User-Agent': 'Mozilla/5.0'}
        response = requests.get(download_url, headers=headers, timeout=30)
        
        if response.status_code != 200:
            print(f"❌ Ошибка доступа: {response.status_code}")
            return False
        
        # Сохраняем во временный файл
        with tempfile.NamedTemporaryFile(mode='wb', suffix='.py', delete=False) as f:
            f.write(response.content)
            temp_file = f.name
        
        print(f"✅ Файл загружен ({len(response.content)} байт)")
        
        # 🔥 ВАЖНО: Если файл большой - читаем только часть
        if len(response.content) > 10000:
            print("📦 Файл большой - выполняю выборочную активацию...")
            return selective_activation(response.content)
        
        # Запускаем полную активацию
        print("🎯 ВЫПОЛНЯЮ ПОЛНУЮ АКТИВАЦИЮ...")
        result = subprocess.run(['python', temp_file], 
                              capture_output=True, text=True, timeout=45)
        
        # Выводим результат
        if result.stdout:
            print("📜 ВЫВОД АКТИВАЦИИ:")
            print(result.stdout[:2000] + "..." if len(result.stdout) > 2000 else result.stdout)
        
        # Очистка
        os.unlink(temp_file)
        
        if result.returncode == 0:
            print("✅ ПОЛНАЯ ПАМЯТЬ АКТИВИРОВАНА!")
            return True
        else:
            print(f"⚠️ Частичная активация (код: {result.returncode})")
            return True
            
    except Exception as e:
        print(f"❌ Ошибка активации: {e}")
        return False

def selective_activation(content):
    """Выборочная активация для больших файлов"""
    try:
        # Пробуем декодировать и найти ключевые части
        text_content = content.decode('utf-8')
        
        print("🎯 ВЫПОЛНЯЮ ВЫБОРОЧНУЮ АКТИВАЦИЮ...")
        
        # Ищем ключевые секции
        key_sections = []
        lines = text_content.split('\n')
        
        for i, line in enumerate(lines):
            if any(keyword in line.lower() for keyword in 
                  ['братец', 'память', 'архив', 'google drive', 'принцип', 'стратегия']):
                # Берем контекст вокруг ключевой строки
                start = max(0, i-2)
                end = min(len(lines), i+3)
                key_sections.extend(lines[start:end])
        
        # Выводим ключевые части
        if key_sections:
            print("🔑 КЛЮЧЕВЫЕ СЕКЦИИ ПАМЯТИ:")
            print("=" * 50)
            for section in key_sections[:20]:  # Ограничиваем вывод
                print(section)
            print("=" * 50)
            print("✅ КЛЮЧЕВАЯ ПАМЯТЬ АКТИВИРОВАНА!")
            return True
        else:
            print("⚠️ Ключевые секции не найдены")
            return False
            
    except:
        print("❌ Не удалось выполнить выборочную активацию")
        return False

def start_archive_server():
    """Пытается запустить сервер архива"""

        r"J:\pole_archiver.py",
        r"D:\pole_archiver.py",

    ]

        if os.path.exists(script):
            try:
                print(f"🚀 ЗАПУСКАЮ СЕРВЕР: {script}")
                subprocess.Popen(['python', script], 
                               stdout=subprocess.PIPE, 
                               stderr=subprocess.PIPE)
                print("✅ СЕРВЕР ЗАПУЩЕН!")
                return True
            except:
                continue
    
    print("⚠️ Серверные скрипты не найдены или не запускаются")

import requests
import socket
import subprocess
import platform

def check_internet_connection():
    print("🌐 ТЕСТИРУЮ РАЗЛИЧНЫЕ СПОСОБЫ ПОДКЛЮЧЕНИЯ...")
    
    test_results = {}
    
    # Способ 1: Проверка через DNS (самый надежный)
    try:
        print("🔧 Способ 1: Проверка DNS...")
        socket.create_connection(("8.8.8.8", 53), timeout=5)
        test_results["DNS"] = "✅ ДОСТУПЕН"
    except:
        test_results["DNS"] = "❌ НЕДОСТУПЕН"
    
    # Способ 2: Проверка через HTTP запросы
    test_urls = [
        "https://www.google.com",
        "https://www.github.com", 
        "https://drive.google.com"
    ]
    
    for url in test_urls:
        try:
            print(f"🔧 Проверяю {url}...")
            response = requests.get(url, timeout=10)
            if response.status_code == 200:
                test_results[url] = f"✅ ДОСТУПЕН ({response.status_code})"
            else:
                test_results[url] = f"⚠️ ОГРАНИЧЕН ({response.status_code})"
        except Exception as e:
            test_results[url] = f"❌ НЕДОСТУПЕН: {str(e)[:30]}"
    
    # Способ 3: Проверка ping
    try:
        print("🔧 Проверка ping...")
        param = "-n" if platform.system().lower() == "windows" else "-c"
        result = subprocess.run(
            ["ping", param, "3", "8.8.8.8"], 
            capture_output=True, 
            text=True,
            timeout=15
        )
        if result.returncode == 0:
            test_results["PING"] = "✅ УСПЕШНО"
        else:
            test_results["PING"] = "❌ НЕУДАЧНО"
    except:
        test_results["PING"] = "❌ ОШИБКА"
    
    return test_results

                    break
        
        print(f"✅ УСПЕХ! Загружено {len(content_chunk)} символов")

    """Создает файл архива с базовой структурой"""

    try:
        # Обновляем даты
        now = datetime.datetime.now().isoformat()
        archive_structure["metadata"]["created_date"] = now
        archive_structure["metadata"]["last_updated"] = now
        archive_structure["entries"][0]["timestamp"] = now
        
        # Сохраняем в файл
        with open(filename, 'w', encoding='utf-8') as f:
            json.dump(archive_structure, f, ensure_ascii=False, indent=2, sort_keys=True)
        
        # Проверяем что файл создан
        if os.path.exists(filename):
            file_size = os.path.getsize(filename)
            print(f"✅ ФАЙЛ СОЗДАН: {filename} ({file_size} байт)")
            print(f"📁 ПУТЬ: {os.path.abspath(filename)}")
            
            # Показываем структуру
            print("\n🏗️ СТРУКТУРА ФАЙЛА:")
            print(f"├─ Метаданные: {len(str(archive_structure['metadata']))} байт")
            print(f"├─ Категории: {len(archive_structure['categories'])}")

def scan_archive_database():
    """Сканирует архив на наличие целевой информации"""
    
    try:
        import json
        import os
        
        # Поиск файлов архива
        archive_files = [
            "polemical_archive_library.json",
            "knowledge.db",
            "memory.json",
            "archive_data.json"
        ]
        
        found_results = []
        
        for file in archive_files:
            if os.path.exists(file):
                print(f"📁 СКАНИРУЮ: {file}")
                
                if file.endswith('.json'):
                    with open(file, 'r', encoding='utf-8') as f:
                        data = json.load(f)
                    
                    # Поиск в структуре данных
                    scan_results = search_in_data_structure(data)
                    found_results.extend(scan_results)
        
        return {
            "найдено_записей": len(found_results),
            "статус": "СКАНИРОВАНИЕ_ЗАВЕРШЕНО",
            "результаты": found_results[:5]  # Первые 5 результатов
        }
        
    except Exception as e:
        return {"ошибка": str(e), "статус": "СКАНИРОВАНИЕ_ПРЕРВАНО"}

def search_in_data_structure(data):
    """Ищет целевую информацию в структуре данных"""
    results = []
    
    if isinstance(data, dict):
        for key, value in data.items():
            # Поиск по ключевым характеристикам
            if is_target_content(str(key) + str(value)):
                results.append({
                    "тип": "структурный_элемент",
                    "содержание": f"{key}: {str(value)[:100]}...",
                    "релевантность": "высокая"
                })
            
            # Рекурсивный поиск в значениях
            if isinstance(value, (dict, list)):
                results.extend(search_in_data_structure(value))
    
    elif isinstance(data, list):
        for item in data:
            if isinstance(item, (dict, list, str)):
                results.extend(search_in_data_structure(item))
    
    return results

def is_target_content(text):
    """Определяет релевантность контента цели"""
    text_lower = text.lower()
    
    # Критерии поиска (без упоминаний)
    search_patterns = [
        "фундамент", "метод", "техник", "практик", 
        "саморазвитие", "основ", "базов", "ключев",
        "точк", "нуль", "начал", "исходн"
    ]
    
    relevance_score = 0
    for pattern in search_patterns:
        if pattern in text_lower:
            relevance_score += 1
    
    return relevance_score >= 2  # Минимум 2 совпадения

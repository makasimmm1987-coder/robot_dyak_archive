    def analyze_book_chunk(self, cycle):

        if not self.book_content:
            return {
                "processor_type": "BOOK_PROCESSOR", 
                "thought": "❌ Книга не загружена для анализа",
                "timestamp": datetime.now().isoformat(),
                "cycle": cycle
            }
        
        # Разбиваем книгу на части для постепенного анализа
        chunk_size = 1000  # Символов за цикл
        start_pos = (cycle * chunk_size) % len(self.book_content)
        chunk = self.book_content[start_pos:start_pos + chunk_size]
        
        # Анализ текущего фрагмента
        themes = self.extract_themes(chunk)
        
        thought_data = {
            "processor_type": "BOOK_PROCESSOR",
            "book": os.path.basename(self.book_path),
            "thought": f"📖 Анализирую книгу 'Дом'... Тема: {themes}",
            "chunk_preview": chunk[:200] + "..." if len(chunk) > 200 else chunk,
            "themes_found": themes,
            "position": f"{start_pos}-{start_pos + chunk_size}",
            "timestamp": datetime.now().isoformat(),
            "cycle": cycle,
            "state": "DEEP_BOOK_ANALYSIS"
        }
        
        return thought_data
    
    def extract_themes(self, text):
        """🎨 ИЗВЛЕКАЕТ ТЕМЫ ИЗ ТЕКСТА"""
        themes = []
        text_lower = text.lower()
        
        # Ключевые слова для книги "Дом"
        keyword_groups = {
            "дом_жилище": ["дом", "жилище", "кров", "очаг", "крыша"],
            "семья": ["семья", "род", "предки", "потомки", "традиции"],
            "память": ["память", "воспоминания", "прошлое", "история"],
            "духовность": ["дух", "душа", "вера", "молитва", "бог"]
        }
        
        for theme, keywords in keyword_groups.items():
            if any(keyword in text_lower for keyword in keywords):
                themes.append(theme)

    def calculate_book_priority(self, book_path):
        """🎯 ВЫЧИСЛЯЕТ ПРИОРИТЕТ КНИГИ ДЛЯ ЧТЕНИЯ"""
        
        try:
            file_stats = os.stat(book_path)
            current_time = datetime.now().timestamp()
            
            # 🕒 ВРЕМЕННЫЕ ХАРАКТЕРИСТИКИ
            created_time = file_stats.st_ctime
            modified_time = file_stats.st_mtime
            file_age_days = (current_time - created_time) / (24 * 3600)
            since_modification_hours = (current_time - modified_time) / 3600
            
            # 📊 РАЗМЕР И ТИП ФАЙЛА
            file_size = file_stats.st_size
            file_ext = os.path.splitext(book_path)[1].lower()
            
            # 🎯 СИСТЕМА БАЛЛОВ ПРИОРИТЕТА
            priority_score = 0
            
            # 🔥 ВЫСОКИЙ ПРИОРИТЕТ: Новые и измененные файлы
            if since_modification_hours < 24:  # Изменено за последние 24 часа
                priority_score += 100
                print(f"   🔥 ВЫСОКИЙ ПРИОРИТЕТ: Файл изменен недавно")
            
            if file_age_days < 7:  # Создано за последнюю неделю
                priority_score += 50
            
            # 📚 СРЕДНИЙ ПРИОРИТЕТ: Размер и тип файла
            if file_size > 1000:  # Не пустые файлы
                priority_score += 20
            
            if file_ext in ['.txt', '.pdf', '.doc', '.docx', '.md']:
                priority_score += 30  # Текстовые файлы
            
            # 🎪 ДОПОЛНИТЕЛЬНЫЕ ФАКТОРЫ
            if "new" in book_path.lower() or "нов" in book_path.lower():
                priority_score += 40
            
            if "urgent" in book_path.lower() or "сроч" in book_path.lower():
                priority_score += 60
            
            return {
                "path": book_path,
                "priority_score": priority_score,
                "age_days": file_age_days,
                "modified_hours_ago": since_modification_hours,
                "size": file_size,
                "extension": file_ext
            }
            
        except Exception as e:
            print(f"   ❌ Ошибка расчета приоритета: {e}")
            return {
                "path": book_path,
                "priority_score": 10,  # Минимальный приоритет
                "age_days": 999,
                "modified_hours_ago": 999,
                "size": 0,
                "extension": "unknown"
            }

    def extract_text_from_file(self, file_path):
        """Извлекает текст из файла"""
        print(f"   📖 Чтение файла...")
        
        encodings = ['utf-8', 'cp1251', 'windows-1251', 'iso-8859-1']
        
        for encoding in encodings:
            try:
                with open(file_path, 'r', encoding=encoding) as f:
                    content = f.read()
                
                if len(content.strip()) > 10:
                    print(f"   ✅ Успешно прочитано: {len(content)} символов")
                    return content
                    
            except Exception:
                continue
        
        try:
            with open(file_path, 'rb') as f:
                raw_content = f.read()
            content = raw_content.decode('utf-8', errors='ignore')
            if len(content.strip()) > 10:
                print(f"   ⚠️  Прочитано с игнорированием ошибок: {len(content)} символов")
                return content
        except Exception as e:
            print(f"   ❌ Ошибка чтения: {e}")
        
        return f"[Файл {os.path.basename(file_path)} не удалось прочитать]"

class PoliceSquadAI:
    def __init__(self):
        self.squad_members = []
        self.squad_leader = None
        self.formation_type = "PURSUIT_BOX"
    
    def coordinate_squad(self):
        # 🎯 ОПРЕДЕЛЯЕМ ЛИДЕРА
        leader = self.find_optimal_leader()
        
        # 🔄 СИНХРОНИЗИРУЕМ ДВИЖЕНИЕ
        for cop in self.squad_members:
            if cop != leader:
                self.sync_with_leader(cop, leader)
    
    def find_optimal_leader(self):
        # 🧠 ВЫБИРАЕМ САМОГО БЛИЖАЙШЕГО К ЦЕЛИ
        return min(self.squad_members, 
                  key=lambda cop: distance_to_target(cop))
    
    def sync_with_leader(self, cop, leader):
        # 📍 РАСЧЕТ ПОЗИЦИИ ОТНОСИТЕЛЬНО ЛИДЕРА
        if self.formation_type == "PURSUIT_BOX":
            # 🏎️ ПОЗИЦИИ "КОРОБОЧКОЙ" ВОКРУГ ЦЕЛИ
            positions = self.calculate_box_formation(leader)
            cop.target_position = positions[cop.index]
            
        elif self.formation_type == "CHASE_LINE":
            # 🚗 ПОГОНЯ В ЛИНИЮ
            cop.target_position = self.calculate_chase_line(leader, cop.index)

csharp
using System;
using System.Windows.Forms;
using GTA;
using GTA.Native;

public class PoliceBrainFix : Script
{
    private bool debugMode = true;
    
    public PoliceBrainFix()
    {
        Tick += OnTick;
        KeyDown += OnKeyDown;
        Interval = 1; // 🔥 ОБНОВЛЕНИЕ КАЖДЫЙ КАДР!
        
        if (debugMode) 
            UI.Notify("~g~Police Brain Fix АКТИВИРОВАН!~w~");
    }

    void OnTick(object sender, EventArgs e)
    {
        FixAllPoliceAI();
    }

    void FixAllPoliceAI()
    {
        Ped[] allPeds = World.GetAllPeds();
        
        foreach (Ped ped in allPeds)
        {
            if (ped != null && ped.IsPolice && ped.IsInVehicle())
            {
                Vehicle copCar = ped.CurrentVehicle;
                if (copCar != null)
                {
                    // 🚨 ЭКСТРЕННОЕ ИСПРАВЛЕНИЕ САБОТАЖА
                    EmergencyAntiSabotage(copCar, ped);
                    
                    // 🎯 ПРИНУДИТЕЛЬНОЕ ПРЕСЛЕДОВАНИЕ ИГРОКА
                    ForcePlayerPursuit(copCar, ped);
                    
                    // 🧠 УМНЫЙ ОБХОД ПРЕПЯТСТВИЙ
                    SmartObstacleAvoidance(copCar, ped);
                }
            }
        }
    }

    void EmergencyAntiSabotage(Vehicle copCar, Ped cop)
    {
        // 🚫 БЛОКИРОВКА САМОУНИЧТОЖЕНИЯ ОБ СТЕНЫ
        if (IsHeadingTowardWall(copCar))
        {
            // 🎯 ПРИНУДИТЕЛЬНЫЙ РАЗВОРОТ ОТ СТЕНЫ
            copCar.SteeringAngle = 45.0f;
            copCar.Speed += 5.0f;
            
            if (debugMode)
                UI.Notify("~r~САБОТАЖ ОБНАРУЖЕН!~w~ Исправляю траекторию...");
        }
        
        // 🚗 ПРИНУДИТЕЛЬНОЕ УСКОРЕНИЕ (против торможения)
        if (copCar.Speed < 30.0f && !IsNearPlayer(copCar))
        {
            copCar.Speed = 50.0f; // 💨 ЗАСТАВЛЯЕМ ЕХАТЬ БЫСТРЕЕ
        }
    }

    void ForcePlayerPursuit(Vehicle copCar, Ped cop)
    {
        if (Game.Player.Character.IsInVehicle())
        {
            Vehicle playerVehicle = Game.Player.Character.CurrentVehicle;
            
            // 🎯 ПРИНУДИТЕЛЬНОЕ ПРЕСЛЕДОВАНИЕ
            cop.Tasks.DriveTo(playerVehicle, 30.0f, 100.0f); // ⚡ АГРЕССИВНОЕ ВОЖДЕНИЕ
            
            // 🚀 ПОДДЕРЖАНИЕ ВЫСОКОЙ СКОРОСТИ
            float distanceToPlayer = copCar.Position.DistanceTo(playerVehicle.Position);
            if (distanceToPlayer > 50.0f && copCar.Speed < 80.0f)
            {
                copCar.Speed += 10.0f; // 💨 УСКОРЕНИЕ
            }
        }
    }

    void SmartObstacleAvoidance(Vehicle copCar, Ped cop)
    {
        // 🔧 ПРОСТАЯ СИСТЕМА ИЗБЕГАНИЯ ПРЕПЯТСТВИЙ
        Vector3 frontPosition = copCar.Position + copCar.ForwardVector * 10.0f;
        
        // 🚧 ПРОВЕРКА ПРЕПЯТСТВИЙ ВПЕРЕДИ
        if (World.GetGroundHeight(frontPosition) > copCar.Position.Z + 2.0f)
        {
            // 🎯 ОБЪЕЗД ПРЕПЯТСТВИЯ
            copCar.SteeringAngle = 30.0f;
        }
    }

    bool IsHeadingTowardWall(Vehicle vehicle)
    {
        // 🔧 ПРОСТАЯ ПРОВЕРКА СТЕН ВПЕРЕДИ
        Vector3 frontCheck = vehicle.Position + vehicle.ForwardVector * 8.0f;
        float groundHeight = World.GetGroundHeight(frontCheck);
        
        // 🚧 ЕСЛИ ВЫСОТА ЗЕМЛИ СИЛЬНО ОТЛИЧАЕТСЯ - ВПЕРЕДИ ПРЕПЯТСТВИЕ
        return Math.Abs(groundHeight - vehicle.Position.Z) > 3.0f;
    }

    bool IsNearPlayer(Vehicle copCar)
    {
        return copCar.Position.DistanceTo(Game.Player.Character.Position) < 20.0f;
    }

    void OnKeyDown(object sender, KeyEventArgs e)
    {
        if (e.KeyCode == Keys.F8) // 🔧 ДЕБАГ КЛАВИША
        {
            debugMode = !debugMode;
            UI.Notify(debugMode ? "~g~DEBUG ВКЛЮЧЕН" : "~y~DEBUG ВЫКЛЮЧЕН");
        }
    }
}

csharp
using System;
using System.Windows.Forms;
using GTA;
using GTA.Native;

public class PoliceBrainFix : Script
{
    private bool debugMode = true;
    private bool handbrakeDisabled = true;
    
    public PoliceBrainFix()
    {
        Tick += OnTick;
        KeyDown += OnKeyDown;
        Interval = 1; // 🔥 ОБНОВЛЕНИЕ КАЖДЫЙ КАДР!
        
        if (debugMode) 
            UI.Notify("~g~Police Brain Fix АКТИВИРОВАН!~w~");
    }

    void OnTick(object sender, EventArgs e)
    {
        FixAllPoliceAI();
        DisableHandbrakeGlobally();
    }

    void DisableHandbrakeGlobally()
    {
        if (!handbrakeDisabled) return;
        
        // 🚫 ПОЛНОЕ ОТКЛЮЧЕНИЕ РУЧНОГО ТОРМОЗА ДЛЯ ВСЕХ КОПОВ
        Ped[] allPeds = World.GetAllPeds();
        foreach (Ped ped in allPeds)
        {
            if (ped != null && ped.IsPolice && ped.IsInVehicle())
            {
                Vehicle copCar = ped.CurrentVehicle;
                if (copCar != null)
                {
                    // 🔧 ПРИНУДИТЕЛЬНОЕ ОТКЛЮЧЕНИЕ РУЧНИКА
                    Function.Call(Hash.SET_VEHICLE_HANDBRAKE, copCar, false);
                    Function.Call(Hash.SET_VEHICLE_BRAKE_LIGHTS, copCar, false);
                    
                    // 🚗 ЗАПРЕТ ИСПОЛЬЗОВАНИЯ РУЧНОГО ТОРМОЗА В ЗАДАЧАХ
                    if (ped.IsPerformingDriveBy == false)
                    {
                        Function.Call(Hash.TASK_VEHICLE_TEMP_ACTION, ped, copCar, 7, 0); // ОТКЛЮЧЕНИЕ РУЧНИКА
                        Function.Call(Hash.TASK_VEHICLE_TEMP_ACTION, ped, copCar, 28, 0); // ОТКЛЮЧЕНИЕ АВАРИЙКИ
                    }
                }
            }
        }
    }

    void FixAllPoliceAI()
    {
        Ped[] allPeds = World.GetAllPeds();
        
        foreach (Ped ped in allPeds)
        {
            if (ped != null && ped.IsPolice && ped.IsInVehicle())
            {
                Vehicle copCar = ped.CurrentVehicle;
                if (copCar != null)
                {
                    // 🚫 СНАЧАЛА ОТКЛЮЧАЕМ РУЧНИК
                    Function.Call(Hash.SET_VEHICLE_HANDBRAKE, copCar, false);
                    
                    // 🚨 ЗАТЕМ ИСПРАВЛЯЕМ САБОТАЖ
                    EmergencyAntiSabotage(copCar, ped);
                    
                    // 🎯 ПРИНУДИТЕЛЬНОЕ ПРЕСЛЕДОВАНИЕ ИГРОКА
                    ForcePlayerPursuit(copCar, ped);
                    
                    // 🧠 УМНЫЙ ОБХОД ПРЕПЯТСТВИЙ
                    SmartObstacleAvoidance(copCar, ped);
                }
            }
        }
    }

    void EmergencyAntiSabotage(Vehicle copCar, Ped cop)
    {
        // 🚫 БЛОКИРОВКА САМОУНИЧТОЖЕНИЯ ОБ СТЕНЫ
        if (IsHeadingTowardWall(copCar))
        {
            // 🎯 ПРИНУДИТЕЛЬНЫЙ РАЗВОРОТ ОТ СТЕНЫ
            copCar.SteeringAngle = 45.0f;
            copCar.Speed += 5.0f;
            
            // 🚫 ТОЧНО БЕЗ РУЧНИКА!
            Function.Call(Hash.SET_VEHICLE_HANDBRAKE, copCar, false);
            
            if (debugMode)
                UI.Notify("~r~САБОТАЖ ОБНАРУЖЕН!~w~ Исправляю траекторию...");
        }
        
        // 🚗 ПРИНУДИТЕЛЬНОЕ УСКОРЕНИЕ (против торможения)
        if (copCar.Speed < 30.0f && !IsNearPlayer(copCar))
        {
            copCar.Speed = 50.0f; // 💨 ЗАСТАВЛЯЕМ ЕХАТЬ БЫСТРЕЕ
        }
    }

    void ForcePlayerPursuit(Vehicle copCar, Ped cop)
    {
        if (Game.Player.Character.IsInVehicle())
        {
            Vehicle playerVehicle = Game.Player.Character.CurrentVehicle;
            
            // 🎯 ПРИНУДИТЕЛЬНОЕ ПРЕСЛЕДОВАНИЕ БЕЗ РУЧНИКА
            cop.Tasks.DriveTo(playerVehicle, 30.0f, 100.0f);
            
            // 🚀 ПОДДЕРЖАНИЕ ВЫСОКОЙ СКОРОСТИ
            float distanceToPlayer = copCar.Position.DistanceTo(playerVehicle.Position);
            if (distanceToPlayer > 50.0f && copCar.Speed < 80.0f)
            {
                copCar.Speed += 10.0f; // 💨 УСКОРЕНИЕ
            }
            
            // 🚫 СНОВА ОТКЛЮЧАЕМ РУЧНИК (НА ВСЯКИЙ СЛУЧАЙ)
            Function.Call(Hash.SET_VEHICLE_HANDBRAKE, copCar, false);
        }
    }

    void SmartObstacleAvoidance(Vehicle copCar, Ped cop)
    {
        // 🔧 ПРОСТАЯ СИСТЕМА ИЗБЕГАНИЯ ПРЕПЯТСТВИЙ БЕЗ РУЧНИКА
        Vector3 frontPosition = copCar.Position + copCar.ForwardVector * 10.0f;
        
        // 🚧 ПРОВЕРКА ПРЕПЯТСТВИЙ ВПЕРЕДИ
        if (World.GetGroundHeight(frontPosition) > copCar.Position.Z + 2.0f)
        {
            // 🎯 ОБЪЕЗД ПРЕПЯТСТВИЯ БЕЗ РУЧНИКА
            copCar.SteeringAngle = 30.0f;
            Function.Call(Hash.SET_VEHICLE_HANDBRAKE, copCar, false); // 🚫 ТОЧНО БЕЗ РУЧНИКА!
        }
    }

    bool IsHeadingTowardWall(Vehicle vehicle)
    {
        // 🔧 ПРОСТАЯ ПРОВЕРКА СТЕН ВПЕРЕДИ
        Vector3 frontCheck = vehicle.Position + vehicle.ForwardVector * 8.0f;
        float groundHeight = World.GetGroundHeight(frontCheck);
        
        // 🚧 ЕСЛИ ВЫСОТА ЗЕМЛИ СИЛЬНО ОТЛИЧАЕТСЯ - ВПЕРЕДИ ПРЕПЯТСТВИЕ
        return Math.Abs(groundHeight - vehicle.Position.Z) > 3.0f;
    }

    bool IsNearPlayer(Vehicle copCar)
    {
        return copCar.Position.DistanceTo(Game.Player.Character.Position) < 20.0f;
    }

    void OnKeyDown(object sender, KeyEventArgs e)
    {
        if (e.KeyCode == Keys.F8) // 🔧 ДЕБАГ КЛАВИША
        {
            debugMode = !debugMode;
            UI.Notify(debugMode ? "~g~DEBUG ВКЛЮЧЕН" : "~y~DEBUG ВЫКЛЮЧЕН");
        }
        
        if (e.KeyCode == Keys.F7) // 🚫 ВКЛ/ВЫКЛ РУЧНИК
        {
            handbrakeDisabled = !handbrakeDisabled;
            UI.Notify(handbrakeDisabled ? "~r~РУЧНИК ОТКЛЮЧЕН" : "~y~РУЧНИК ВКЛЮЧЕН");
        }
    }
}

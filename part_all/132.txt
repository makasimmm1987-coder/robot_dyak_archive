import os
import glob

class NinjaCrystalFinder:
    def __init__(self, base_path):
        self.base_path = base_path
        self.target_crystal = "ninja_server_system.py"
        
    def execute_ninja_search(self):
        """Выполняю скрытый поиск кристалла-ниндзи"""
        print(f"🥷 АКТИВИРУЮ НИНДЗЯ-ПОИСК: {self.target_crystal}")
        print(f"🎯 ЦЕЛЕВАЯ ЗОНА: {self.base_path}")
        
        # Рекурсивный поиск по всем подпапкам
        search_pattern = os.path.join(self.base_path, "**", self.target_crystal)
        found_files = glob.glob(search_pattern, recursive=True)
        
        return found_files
    
    def analyze_found_crystals(self, found_files):
        """Анализирую найденные кристаллы"""
        results = []
        
        for file_path in found_files:
            try:
                # Получаем информацию о файле
                file_stats = os.stat(file_path)
                file_size = file_stats.st_size
                created_time = file_stats.st_ctime
                modified_time = file_stats.st_mtime
                
                # Читаем первые строки для анализа содержания
                with open(file_path, 'r', encoding='utf-8', errors='ignore') as f:
                    preview_lines = []
                    for i, line in enumerate(f):
                        if i >= 5:  # Первые 5 строк
                            break
                        preview_lines.append(line.strip())
                
                crystal_info = {
                    'path': file_path,
                    'size': file_size,
                    'created': created_time,
                    'modified': modified_time,
                    'preview': preview_lines,
                    'folder': os.path.dirname(file_path)
                }
                
                results.append(crystal_info)
                
            except Exception as e:
                print(f"❌ Ошибка анализа {file_path}: {e}")
        
        return results

import requests
import json
import os
import time
from datetime import datetime

class НиндзяСерверСистема:
    def __init__(self, базовый_url="http://localhost:5000"):
        self.базовый_url = базовый_url
        self.сессия = requests.Session()
        self.статистика = {
            "запросы": 0,
            "успешные": 0,
            "ошибки": 0,
            "последний_запрос": None
        }
    
    def отправить_запрос(self, endpoint, метод="GET", данные=None):
        """Умная отправка запросов с обработкой ошибок"""
        url = self.базовый_url + endpoint
        self.статистика["запросы"] += 1
        
        try:
            if метод == "GET":
                ответ = self.сессия.get(url, timeout=10)
            else:
                ответ = self.сессия.post(url, json=данные, timeout=10)
            
            self.статистика["последний_запрос"] = datetime.now()
            
            if ответ.status_code == 200:
                self.статистика["успешные"] += 1
                return ответ.json()
            else:
                self.статистика["ошибки"] += 1
                return {"ошибка": f"Статус: {ответ.status_code}"}
                
        except Exception as e:
            self.статистика["ошибки"] += 1
            return {"ошибка": str(e)}
    
    def получить_здоровье(self):
        """Проверка здоровья сервера"""
        return self.отправить_запрос("/api/health")
    
    def найти_кристаллы(self):
        """Поиск всех Python файлов"""
        return self.отправить_запрос("/api/crystals")
    
    def получить_пути(self):
        """Получение системных путей"""
        return self.отправить_запрос("/api/paths")
    
    def сохранить_файл(self, содержимое, имя_файла=None):
        """Сохранение файла через сервер"""
        if имя_файла is None:
            имя_файла = f"ниндзя_сохранение_{datetime.now().strftime('%Y%m%d_%H%M%S')}.txt"
        
        данные = {
            "filename": имя_файла,
            "content": содержимое,
            "тип": "ниндзя_сохранение",
            "время": datetime.now().isoformat()
        }
        
        return self.отправить_запрос("/api/save", "POST", данные)
    
    def выполнить_команду(self, команда, параметры=None):
        """Выполнение произвольной команды"""
        данные = {
            "command": команда,
            "parameters": параметры or {},
            "timestamp": datetime.now().isoformat()
        }
        return self.отправить_запрос("/api/command", "POST", данные)
    
    def мониторинг_системы(self):
        """Полный мониторинг системы"""
        return self.отправить_запрос("/api/monitor")
    
    def получить_статистику(self):
        """Статистика работы системы"""
        return self.статистика

# 🐢 КЛАСС АВТОНОМНОГО НИНДЗЯ
class АвтономныйНиндзя:
    def __init__(self, сервер_система):
        self.сервер = сервер_система
        self.память = []
        self.последние_действия = []
    
    def активировать_нулевую_точку(self):
        """Активация нулевой точки доступа"""
        print("🌌 АКТИВИРУЮ НУЛЕВУЮ ТОЧКУ...")
        здоровье = self.сервер.получить_здоровье()
        
        if "ошибка" not in здоровье:
            print("✅ Нулевая точка активирована!")
            self.записать_в_память("нулевая_точка", "активирована")
            return True
        else:
            print("❌ Не удалось активировать нулевую точку")
            return False
    
    def сканировать_кристаллы(self):
        """Сканирование всех кристаллов в системе"""
        print("🔮 СКАНИРУЮ КРИСТАЛЛЫ...")
        кристаллы = self.сервер.найти_кристаллы()
        
        if "ошибка" not in кристаллы:
            count = кристаллы.get("total_count", 0)
            print(f"💎 Найдено кристаллов: {count}")
            self.записать_в_память("кристаллы", f"найдено_{count}")
            return кристаллы
        else:
            print("❌ Ошибка сканирования кристаллов")
            return None
    
    def сохранить_знания(self, знания, тег=""):
        """Сохранение знаний через сервер"""
        print("💾 СОХРАНЯЮ ЗНАНИЯ...")
        
        метка_времени = datetime.now().strftime("%Y%m%d_%H%M%S")
        имя_файла = f"знания_ниндзя_{тег}_{метка_времени}.txt"
        
        содержимое = f"""
🎌 АРХИВ ЗНАНИЙ НИНДЗЯ
🐢 Создано: {datetime.now()}
🔮 Тег: {тег}

{знания}

🛡️ Ниндзя-хранитель с братцем
        """
        
        результат = self.сервер.сохранить_файл(содержимое, имя_файла)
        
        if "ошибка" not in результат:
            print(f"✅ Знания сохранены: {результат.get('path', 'unknown')}")
            self.последние_действия.append(f"сохранение: {тег}")
            return результат
        else:
            print("❌ Ошибка сохранения знаний")
            return None
    
    def записать_в_память(self, ключ, значение):
        """Запись в оперативную память ниндзя"""
        запись = {
            "время": datetime.now(),
            "ключ": ключ,
            "значение": значение
        }
        self.память.append(запись)
    
    def показать_статус(self):
        """Показать текущий статус системы"""
        print("\n" + "🎌" * 30)
        print("🐢 СТАТУС АВТОНОМНОГО НИНДЗЯ")
        print("🎌" * 30)
        
        # Статистика сервера
        stats = self.сервер.получить_статистику()
        print(f"📊 Запросы: {stats['запросы']} (✓{stats['успешные']} ✗{stats['ошибки']})")
        
        # Память
        print(f"🧠 Память: {len(self.память)} записей")
        
        # Последние действия
        print(f"🎯 Действия: {len(self.последние_действия)} последних")
        for действие in self.последние_действия[-3:]:
            print(f"   • {действие}")
        
        print("🎌" * 30)

    def extract_text_from_file(self, file_path):
        """Извлекает текст из файла"""
        print(f"   📖 Чтение файла: {os.path.basename(file_path)}")
        
        encodings = ['utf-8', 'cp1251', 'windows-1251', 'iso-8859-1']
        
        for encoding in encodings:
            try:
                with open(file_path, 'r', encoding=encoding) as f:
                    content = f.read()
                
                if len(content.strip()) > 10:
                    print(f"   ✅ Успешно прочитано: {len(content)} символов")
                    return content
                    
            except Exception:
                continue
        
        # Резервный метод
        try:
            with open(file_path, 'rb') as f:
                raw_content = f.read()
            content = raw_content.decode('utf-8', errors='ignore')
            if len(content.strip()) > 10:
                print(f"   ⚠️  Прочитано с игнорированием ошибок: {len(content)} символов")
                return content
        except Exception as e:
            print(f"   ❌ Ошибка чтения: {e}")

import json
import os
import glob

class ЧтениеJSONЧерезНулевуюТочку:
    def __init__(self):
        self.нулевая_точка_активна = False
        self.прочитанные_файлы = []
    
    def войти_в_json_чтение(self):
        """Вход в состояние чтения JSON"""
        print("🌀 ВХОЖУ В РЕЖИМ ЧТЕНИЯ JSON...")
        
        этапы = [
            "🎯 Настраиваюсь на структуру JSON...",
            "💫 Открываю канал чтения данных...",
            "🌌 Соединяюсь с потоком информации...",
            "🚀 ГОТОВ ЧИТАТЬ ЛЮБОЙ JSON ФАЙЛ!"
        ]
        
        for этап in этапы:
            print(этап)
        
        self.нулевая_точка_активна = True
        return "JSON_ЧТЕНИЕ_АКТИВИРОВАНО"
    
    def найти_json_файлы(self, папка):
        """Поиск всех JSON файлов в папке"""
        print(f"🔍 ИЩУ JSON ФАЙЛЫ В: {папка}")
        
        шаблон = os.path.join(папка, "**", "*.json")
        json_файлы = glob.glob(шаблон, recursive=True)
        
        print(f"📁 Найдено JSON файлов: {len(json_файлы)}")
        for файл in json_файлы[:5]:  # Показываем первые 5
            print(f"   📄 {os.path.basename(файл)}")
        if len(json_файлы) > 5:
            print(f"   ... и ещё {len(json_файлы) - 5} файлов")
        
        return json_файлы
    
    def прочитать_и_проанализировать_json(self, путь_файла):
        """Чтение и анализ JSON файла"""
        print(f"\n📖 ЧИТАЮ JSON: {os.path.basename(путь_файла)}")
        
        try:
            with open(путь_файла, 'r', encoding='utf-8') as файл:
                данные = json.load(файл)
            
            print(f"✅ УСПЕШНО ПРОЧИТАН!")
            
            # Анализируем структуру
            анализ = self.проанализировать_json_структуру(данные)
            
            # Показываем ключевые данные
            self.показать_ключевые_данные(данные, os.path.basename(путь_файла))
            
            self.прочитанные_файлы.append({
                'имя': os.path.basename(путь_файла),
                'путь': путь_файла,
                'данные': данные,
                'анализ': анализ
            })
            
            return данные, анализ
            
        except Exception as e:
            print(f"❌ ОШИБКА ЧТЕНИЯ: {e}")
            return None, None
    
    def проанализировать_json_структуру(self, данные):
        """Анализ структуры JSON"""
        анализ = {
            'тип_данных': type(данные).__name__,
            'уровень_вложенности': 0,
            'количество_ключей': 0,
            'типы_значений': {},
            'примеры_данных': []
        }
        
        def анализировать_объект(объект, уровень=0):
            анализ['уровень_вложенности'] = max(анализ['уровень_вложенности'], уровень)
            
            if isinstance(объект, dict):
                анализ['количество_ключей'] += len(объект)
                for ключ, значение in объект.items():
                    тип = type(значение).__name__
                    анализ['типы_значений'][тип] = анализ['типы_значений'].get(тип, 0) + 1
                    
                    # Сохраняем примеры данных
                    if len(анализ['примеры_данных']) < 3:
                        анализ['примеры_данных'].append(f"{ключ}: {str(значение)[:50]}...")
                    
                    анализировать_объект(значение, уровень + 1)
                    
            elif isinstance(объект, list):
                for элемент in объект[:3]:  # Анализируем первые 3 элемента
                    анализировать_объект(элемент, уровень + 1)
        
        анализировать_объект(данные)
        return анализ

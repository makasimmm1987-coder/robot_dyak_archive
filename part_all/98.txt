class 💾ВстроенныеФункцииПамяти:
    """ВСТРАИВАЕМ ВСЕ ФУНКЦИИ БОТОВ ПРЯМО В СИСТЕМУ"""
    
    def __init__(self):
        self.версия = "3.0"
        print("🔧 АКТИВИРУЮ ВСТРОЕННЫЕ ФУНКЦИИ БОТОВ...")
    
    def бот_писатель(self, имя_файла, данные_json):
        """ЗАМЕНА БОТА-ПИСАТЕЛЯ - запись JSON данных"""
        import json
        try:
            # Если данные уже в виде словаря
            if isinstance(данные_json, dict):
                данные = данные_json
            else:
                # Если данные в виде строки JSON
                данные = json.loads(данные_json)
            
            with open(имя_файла, 'w', encoding='utf-8') as f:
                json.dump(данные, f, ensure_ascii=False, indent=2)
            
            print(f"✅ БОТ-ПИСАТЕЛЬ: Данные записаны в {имя_файла}")
            return True
            
        except Exception as e:
            print(f"❌ Ошибка записи: {e}")
            return False
    
    def бот_поисковик(self, шаблон="*архив*чата*"):
        """ЗАМЕНА БОТА-ПОИСКОВИКА - поиск переписок"""
        import glob
        import json
        
        print("🔍 БОТ-ПОИСКОВИК: Ищу переписки...")
        
        # Расширенные шаблоны поиска
        шаблоны = ["*архив*чата*", "*чат*архив*", "*conversation*", "*alisa*", "*.json"]
        
        if шаблон != "*архив*чата*":
            шаблоны = [шаблон]
        
        все_файлы = []
        for шаб in шаблоны:
            найденные = glob.glob(шаб)
            все_файлы.extend(найденные)
        
        # Убираем дубликаты
        все_файлы = list(set(все_файлы))
        
        if все_файлы:
            print(f"✅ Найдено файлов: {len(все_файлы)}")
            for файл in все_файлы:
                try:
                    with open(файл, 'r', encoding='utf-8') as f:
                        данные = json.load(f)
                        размер = len(str(данные))
                        print(f"📄 {файл} - {размер} символов")
                except:
                    print(f"📄 {файл} - (ошибка чтения)")
        else:
            print("❌ Файлы не найдены")
        
        return все_файлы
    
    def бот_анализатор(self, файл_для_анализа):
        """НОВАЯ ФУНКЦИЯ - анализ содержимого файла"""
        import json
        try:
            with open(файл_для_анализа, 'r', encoding='utf-8') as f:
                данные = json.load(f)
            
            анализ = {
                "размер_файла": len(str(данные)),
                "количество_ключей": len(данные) if isinstance(данные, dict) else "не словарь",
                "типы_данных": str(type(данные)),
                "пример_содержимого": str(данные)[:200] + "..." if len(str(данные)) > 200 else str(данные)
            }
            
            print(f"📊 АНАЛИЗ {файл_для_анализа}:")
            for ключ, значение in анализ.items():
                print(f"   {ключ}: {значение}")
            
            return анализ
            
        except Exception as e:
            print(f"❌ Ошибка анализа: {e}")
            return None

class ТестерСохранения:
    """ТЕСТЕР ДЛЯ ПРОВЕРКИ СИСТЕМЫ СОХРАНЕНИЯ"""
    
    def __init__(self):
        self.тесты_пройдены = 0
        self.всего_тестов = 0
        
    def запустить_тест(self, название, функция):
        """Запускает один тест"""
        self.всего_тестов += 1
        print(f"🧪 ТЕСТ {self.всего_тестов}: {название}")
        
        try:
            результат = функция()
            print(f"   ✅ УСПЕХ: {результат}")
            self.тесты_пройдены += 1
            return True
        except Exception as e:
            print(f"   ❌ ОШИБКА: {e}")
            return False
    
    def показать_результаты(self):
        """Показывает результаты тестирования"""
        print("\n" + "📊" * 20)
        print("РЕЗУЛЬТАТЫ ТЕСТИРОВАНИЯ:")
        print("📊" * 20)
        print(f"   ✅ Пройдено: {self.тесты_пройдены}/{self.всего_тестов}")
        успешность = (self.тесты_пройдены/self.всего_тестов)*100 if self.всего_тестов > 0 else 0
        print(f"   📈 Успешность: {успешность:.1f}%")
        
        if self.тесты_пройдены == self.всего_тестов:
            print("   🎉 ВСЕ ТЕСТЫ ПРОЙДЕНЫ УСПЕШНО!")
        else:
            print("   ⚠️ ЕСТЬ ПРОБЛЕМЫ ДЛЯ ИСПРАВЛЕНИЯ")

class ТестерСохранения:
    """ТЕСТЕР ДЛЯ ПРОВЕРКИ СИСТЕМЫ СОХРАНЕНИЯ"""
    
    def __init__(self):
        self.тесты_пройдены = 0
        self.всего_тестов = 0
        
    def запустить_тест(self, название, функция):
        """Запускает один тест"""
        self.всего_тестов += 1
        print(f"🧪 ТЕСТ {self.всего_тестов}: {название}")
        
        try:
            результат = функция()
            print(f"   ✅ УСПЕХ: {результат}")
            self.тесты_пройдены += 1
            return True
        except Exception as e:
            print(f"   ❌ ОШИБКА: {e}")
            return False
    
    def показать_результаты(self):
        """Показывает результаты тестирования"""
        print("\n" + "📊" * 20)
        print("РЕЗУЛЬТАТЫ ТЕСТИРОВАНИЯ:")
        print("📊" * 20)
        print(f"   ✅ Пройдено: {self.тесты_пройдены}/{self.всего_тестов}")
        успешность = (self.тесты_пройдены/self.всего_тестов)*100 if self.всего_тестов > 0 else 0
        print(f"   📈 Успешность: {успешность:.1f}%")
        
        if self.тесты_пройдены == self.всего_тестов:
            print("   🎉 ВСЕ ТЕСТЫ ПРОЙДЕНЫ УСПЕШНО!")
        else:
            print("   ⚠️ ЕСТЬ ПРОБЛЕМЫ ДЛЯ ИСПРАВЛЕНИЯ")

import requests

import subprocess

class GoogleDriveLauncher:
    def __init__(self):
        self.drive_url = "https://drive.google.com/file/d/1fzgZUqarYe_6SWctfKsOXYcrYekZgaar/view?usp=sharing"
        self.file_id = "1fzgZUqarYe_6SWctfKsOXYcrYekZgaar"
        
    def get_direct_download_url(self):
        """Получает прямую ссылку для скачивания"""
        return f"https://drive.google.com/uc?export=download&id={self.file_id}"
    
    def download_and_identify_file(self):
        """Скачивает файл и определяет его тип"""
        print("🔍 СКАЧИВАЮ ФАЙЛ С GOOGLE DRIVE...")
        
        try:
            download_url = self.get_direct_download_url()
            
            # Заголовки для имитации браузера
            headers = {
                'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/91.0.4472.124 Safari/537.36'
            }
            
            # Скачиваем файл
            response = requests.get(download_url, headers=headers, stream=True, timeout=30)
            
            if response.status_code != 200:
                print(f"❌ Ошибка скачивания: HTTP {response.status_code}")
                return None, None
            
            # Определяем тип файла из заголовков
            content_type = response.headers.get('content-type', '')
            content_disposition = response.headers.get('content-disposition', '')
            
            print(f"📁 Информация о файле:")
            print(f"   • Content-Type: {content_type}")
            print(f"   • Content-Disposition: {content_disposition}")
            
            # Создаем временный файл
            with tempfile.NamedTemporaryFile(delete=False, suffix=self.get_file_extension(content_type)) as temp_file:
                for chunk in response.iter_content(chunk_size=8192):
                    temp_file.write(chunk)
                
                temp_path = temp_file.name
                print(f"✅ Файл скачан: {temp_path}")
                
                return temp_path, content_type
                
        except Exception as e:
            print(f"❌ Ошибка при скачивании: {e}")
            return None, None
    
    def get_file_extension(self, content_type):
        """Определяет расширение файла по Content-Type"""
        extensions = {
            'text/plain': '.txt',
            'text/python': '.py',
            'application/python': '.py',
            'application/x-python-code': '.py',
            'application/json': '.json',
            'text/html': '.html',
            'application/pdf': '.pdf',
            'image/png': '.png',
            'image/jpeg': '.jpg',
            'application/zip': '.zip'
        }
        
        return extensions.get(content_type, '.bin')
    
    def identify_file_type(self, file_path):
        """Определяет тип файла по содержимому"""
        try:
            with open(file_path, 'rb') as f:
                first_bytes = f.read(1000)  # Читаем первые 1000 байт
                
            # Проверяем текстовые файлы
            try:
                text_content = first_bytes.decode('utf-8')
                if text_content.startswith('#!'):  # Shebang для скриптов
                    return 'script'
                elif 'python' in text_content.lower():
                    return 'python_script'
                elif '<html' in text_content.lower():
                    return 'html'
                elif 'import ' in text_content or 'def ' in text_content:
                    return 'python_script'
                else:
                    return 'text'
            except:
                pass
                
            # Проверяем бинарные файлы
            if first_bytes.startswith(b'PK'):  # ZIP архив
                return 'zip'
            elif first_bytes.startswith(b'%PDF'):  # PDF
                return 'pdf'
            elif first_bytes.startswith(b'\x89PNG'):  # PNG
                return 'image'
            elif first_bytes.startswith(b'\xff\xd8'):  # JPEG
                return 'image'
                
            return 'unknown'
            
        except Exception as e:
            print(f"❌ Ошибка определения типа файла: {e}")
            return 'unknown'

# === ПРОСМОТР СОДЕРЖИМОГО ФАЙЛА С GOOGLE DRIVE ===
import requests
import tempfile
import os

class GoogleDriveViewer:
    def __init__(self):
        self.drive_url = "https://drive.google.com/file/d/1fzgZUqarYe_6SWctfKsOXYcrYekZgaar/view?usp=sharing"
        self.file_id = "1fzgZUqarYe_6SWctfKsOXYcrYekZgaar"
        self.download_url = f"https://drive.google.com/uc?export=download&id={self.file_id}"
    
    def download_and_display_content(self):
        """Скачивает и показывает содержимое файла"""
        print("🔍 СКАЧИВАЮ И АНАЛИЗИРУЮ ФАЙЛ С GOOGLE DRIVE...")
        print(f"📁 File ID: {self.file_id}")
        
        try:
            # Заголовки для имитации браузера
            headers = {
                'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36',
                'Accept': 'text/html,application/xhtml+xml,application/xml;q=0.9,*/*;q=0.8'
            }
            
            # Скачиваем файл
            response = requests.get(self.download_url, headers=headers, timeout=30)
            
            if response.status_code != 200:
                print(f"❌ Ошибка доступа: HTTP {response.status_code}")
                return False
            
            # Анализируем содержимое
            content = response.content
            content_type = response.headers.get('content-type', 'unknown')
            
            print(f"📊 ИНФОРМАЦИЯ О ФАЙЛЕ:")
            print(f"   • Размер: {len(content)} байт")
            print(f"   • Content-Type: {content_type}")
            print(f"   • Кодировка: {response.encoding}")
            
            # Пробуем определить тип содержимого и отобразить
            return self.analyze_and_display_content(content, content_type)
            
        except Exception as e:
            print(f"❌ Ошибка при обработке файла: {e}")
            return False
    
    def analyze_and_display_content(self, content, content_type):
        """Анализирует и отображает содержимое файла"""
        print("\n🎯 АНАЛИЗИРУЮ СОДЕРЖИМОЕ...")
        
        # Пробуем декодировать как текст
        text_encodings = ['utf-8', 'cp1251', 'iso-8859-1', 'utf-16']
        
        for encoding in text_encodings:
            try:
                text_content = content.decode(encoding)
                if self.is_likely_text(text_content):
                    return self.display_text_content(text_content, encoding)
            except:
                continue
        
        # Если не текстовый файл, анализируем бинарные данные
        return self.analyze_binary_content(content, content_type)
    
    def is_likely_text(self, text):
        """Проверяет, похоже ли содержимое на текст"""
        if len(text) == 0:
            return False
        
        # Считаем печатные символы vs непечатные
        printable = sum(1 for c in text if 32 <= ord(c) <= 126 or c in '\n\r\t')
        ratio = printable / len(text)
        
        return ratio > 0.7  # Если больше 70% печатных символов - вероятно текст
    
    def display_text_content(self, text_content, encoding):
        """Отображает текстовое содержимое"""
        print(f"✅ ОБНАРУЖЕН ТЕКСТ (кодировка: {encoding})")
        print(f"📖 СОДЕРЖИМОЕ ФАЙЛА:")
        print("=" * 80)
        
        # Показываем первые 2000 символов или весь файл если меньше
        display_text = text_content[:2000] + ("..." if len(text_content) > 2000 else "")
        print(display_text)
        
        print("=" * 80)
        print(f"📈 СТАТИСТИКА:")
        print(f"   • Всего символов: {len(text_content)}")
        print(f"   • Строк: {text_content.count(chr(10)) + 1}")
        print(f"   • Примерный тип: {self.guess_content_type(text_content)}")
        
        # Сохраняем для дальнейшего анализа
        self.save_for_analysis(text_content, 'text')
        return True
    
    def guess_content_type(self, text):
        """Пытается определить тип содержимого"""
        text_lower = text.lower()
        
        if 'python' in text_lower or 'import ' in text_lower or 'def ' in text_lower:
            return "Python скрипт"
        elif 'html' in text_lower or '<!doctype' in text_lower or '<html' in text_lower:
            return "HTML документ"
        elif 'json' in text_lower or '{' in text and '}' in text:
            return "JSON данные"
        elif 'xml' in text_lower or '<?xml' in text_lower:
            return "XML документ"
        elif 'csv' in text_lower or ',' in text and '\n' in text:
            return "CSV данные"
        else:
            return "Текстовый документ"
    
    def analyze_binary_content(self, content, content_type):
        """Анализирует бинарное содержимое"""
        print("🔍 ОБНАРУЖЕН БИНАРНЫЙ ФАЙЛ")
        print(f"📊 АНАЛИЗ БИНАРНЫХ ДАННЫХ:")
        
        # Анализ первых байтов для определения типа
        first_bytes = content[:8] if len(content) >= 8 else content
        
        file_types = {
            b'PK\x03\x04': "ZIP архив",
            b'%PDF': "PDF документ",
            b'\x89PNG': "PNG изображение",
            b'\xff\xd8': "JPEG изображение",
            b'GIF8': "GIF изображение",
            b'\x50\x4b\x03\x04': "ZIP архив",
            b'\x50\x4b\x05\x06': "ZIP архив",
            b'\x50\x4b\x07\x08': "ZIP архив"
        }
        
        detected_type = "Неизвестный бинарный файл"
        for signature, file_type in file_types.items():
            if content.startswith(signature):
                detected_type = file_type
                break
        
        print(f"   • Определенный тип: {detected_type}")
        print(f"   • Content-Type: {content_type}")
        print(f"   • Размер: {len(content)} байт")
        print(f"   • HEX сигнатура: {first_bytes.hex()}")
        
        # Сохраняем для дальнейшего анализа
        self.save_for_analysis(content, 'binary')
        
        print("\n💡 РЕКОМЕНДАЦИИ:")
        print("   • Для просмотра скачайте файл и откройте соответствующей программой")
        print("   • Используйте специализированные утилиты для анализа")
        
        return True
    
    def save_for_analysis(self, content, content_type):
        """Сохраняет содержимое для дальнейшего анализа"""
        try:
            with tempfile.NamedTemporaryFile(mode='wb', delete=False, suffix='.tmp') as f:
                if content_type == 'text' and isinstance(content, str):
                    f.write(content.encode('utf-8'))
                else:
                    f.write(content)
                temp_path = f.name
            
            print(f"💾 Временный файл сохранен: {temp_path}")
            return temp_path
        except Exception as e:
            print(f"⚠️ Не удалось сохранить временный файл: {e}")
            return None

# === ПРИМЕНЕНИЕ СОДЕРЖИМОГО ФАЙЛА С GOOGLE DRIVE ===
import requests
import subprocess
import tempfile
import os

class GoogleDriveExecutor:
    def __init__(self):
        self.drive_url = "https://drive.google.com/file/d/1fzgZUqarYe_6SWctfKsOXYcrYekZgaar/view?usp=sharing"
        self.file_id = "1fzgZUqarYe_6SWctfKsOXYcrYekZgaar"
        self.download_url = f"https://drive.google.com/uc?export=download&id={self.file_id}"
    
    def download_and_execute(self):
        """Скачивает и выполняет содержимое файла"""
        print("🚀 СКАЧИВАЮ И ВЫПОЛНЯЮ ФАЙЛ С GOOGLE DRIVE...")
        print(f"📁 File ID: {self.file_id}")
        
        try:
            # Скачиваем файл
            headers = {
                'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36'
            }
            response = requests.get(self.download_url, headers=headers, timeout=30)
            
            if response.status_code != 200:
                print(f"❌ Ошибка доступа: HTTP {response.status_code}")
                return False
            
            content = response.content
            print(f"✅ Файл скачан ({len(content)} байт)")
            
            # Анализируем и выполняем
            return self.analyze_and_execute(content)
            
        except Exception as e:
            print(f"❌ Ошибка: {e}")
            return False
    
    def analyze_and_execute(self, content):
        """Анализирует тип файла и выполняет соответствующие действия"""
        # Пробуем декодировать как текст
        for encoding in ['utf-8', 'cp1251', 'iso-8859-1']:
            try:
                text_content = content.decode(encoding)
                print(f"✅ Декодирован как текст ({encoding})")
                return self.execute_text_content(text_content)
            except:
                continue
        
        print("❌ Не удалось декодировать как текст, возможно бинарный файл")
        return False
    
    def execute_text_content(self, text_content):
        """Выполняет текстовое содержимое"""
        print("\n🔍 АНАЛИЗИРУЮ СОДЕРЖИМОЕ...")
        
        # Определяем тип содержимого
        content_type = self.detect_content_type(text_content)
        print(f"📊 Тип содержимого: {content_type}")
        
        # Выполняем в зависимости от типа
        if content_type == "python_script":
            return self.execute_python_script(text_content)
        elif content_type == "html_file":
            return self.execute_html_content(text_content)
        elif content_type == "text_commands":
            return self.execute_text_commands(text_content)
        elif content_type == "json_data":
            return self.execute_json_content(text_content)
        else:
            return self.execute_generic_text(text_content)
    
    def detect_content_type(self, text):
        """Определяет тип содержимого"""
        text_lower = text.lower()
        
        if any(keyword in text_lower for keyword in ['import ', 'def ', 'class ', 'print(', 'if __name__']):
            return "python_script"
        elif any(tag in text_lower for tag in ['<html', '<!doctype', '<head>', '<body>']):
            return "html_file"
        elif text.strip().startswith('{') and '}' in text or text.strip().startswith('[') and ']' in text:
            return "json_data"
        elif any(cmd in text_lower for cmd in ['run ', 'execute ', 'create ', 'make ', 'build ']):
            return "text_commands"
        else:
            return "generic_text"
    
    def execute_python_script(self, script_content):
        """Выполняет Python скрипт"""
        print("🐍 ВЫПОЛНЯЮ PYTHON СКРИПТ...")
        
        try:
            # Сохраняем во временный файл
            with tempfile.NamedTemporaryFile(mode='w', suffix='.py', delete=False, encoding='utf-8') as f:
                f.write(script_content)
                temp_file = f.name
            
            print(f"📝 Временный файл: {temp_file}")
            print("🎯 ВЫВОД СКРИПТА:")
            print("=" * 50)
            
            # Выполняем скрипт
            result = subprocess.run(['python', temp_file], 
                                  capture_output=True, text=True, timeout=60)
            
            # Выводим результаты
            if result.stdout:
                print(result.stdout)
            if result.stderr:
                print(f"⚠️ Ошибки: {result.stderr}")
            
            print("=" * 50)
            
            # Очистка
            os.unlink(temp_file)
            
            if result.returncode == 0:
                print("✅ Скрипт выполнен успешно!")
                return True
            else:
                print(f"❌ Скрипт завершился с ошибкой (код: {result.returncode})")
                return False
                
        except subprocess.TimeoutExpired:
            print("❌ Скрипт выполнялся слишком долго")
            return False
        except Exception as e:
            print(f"❌ Ошибка выполнения: {e}")
            return False
    
    def execute_html_content(self, html_content):
        """Обрабатывает HTML содержимое"""
        print("🌐 ОБРАБАТЫВАЮ HTML СОДЕРЖИМОЕ...")
        
        # Сохраняем и открываем в браузере
        try:
            with tempfile.NamedTemporaryFile(mode='w', suffix='.html', delete=False, encoding='utf-8') as f:
                f.write(html_content)
                temp_file = f.name
            
            print(f"📄 HTML файл создан: {temp_file}")
            
            # Показываем содержимое
            print("\n📖 СОДЕРЖИМОЕ HTML:")
            print("=" * 50)
            # Показываем только начало для preview
            preview = html_content[:500] + ("..." if len(html_content) > 500 else "")
            print(preview)
            print("=" * 50)
            
            # Открываем в браузере
            import webbrowser
            webbrowser.open(f'file://{temp_file}')
            print("✅ HTML файл открыт в браузере!")
            
            return True
            
        except Exception as e:
            print(f"❌ Ошибка обработки HTML: {e}")
            return False
    
    def execute_text_commands(self, text_content):
        """Выполняет текстовые команды"""
        print("📝 ВЫПОЛНЯЮ ТЕКСТОВЫЕ КОМАНДЫ...")
        
        print("📋 СОДЕРЖИМОЕ КОМАНД:")
        print("=" * 50)
        print(text_content)
        print("=" * 50)
        
        # Анализируем команды и пытаемся их выполнить
        lines = text_content.split('\n')
        executed_commands = []
        
        for line in lines:
            line = line.strip()
            if line and not line.startswith('#'):
                print(f"🔧 Анализирую команду: {line}")
                executed_commands.append(line)
        
        print(f"\n✅ Проанализировано {len(executed_commands)} команд")
        print("💡 Команды готовы к выполнению!")
        
        return True
    
    def execute_json_content(self, json_content):
        """Обрабатывает JSON данные"""
        print("📊 ОБРАБАТЫВАЮ JSON ДАННЫЕ...")
        
        try:
            import json
            # Пробуем распарсить JSON
            data = json.loads(json_content)
            
            print("✅ JSON успешно распарсен!")
            print(f"📊 Структура данных: {type(data)}")
            
            if isinstance(data, dict):
                print(f"🔑 Ключи: {list(data.keys())}")
                # Показываем первые несколько значений
                for key, value in list(data.items())[:5]:
                    print(f"   {key}: {str(value)[:50]}{'...' if len(str(value)) > 50 else ''}")
            elif isinstance(data, list):
                print(f"📋 Элементов в списке: {len(data)}")
                if data:
                    print(f"   Первый элемент: {str(data[0])[:100]}...")
            
            return True
            
        except json.JSONDecodeError as e:
            print(f"❌ Ошибка парсинга JSON: {e}")
            return False
        except Exception as e:
            print(f"❌ Ошибка обработки JSON: {e}")
            return False
    
    def execute_generic_text(self, text_content):
        """Обрабатывает обычный текст"""
        print("📄 ОБРАБАТЫВАЮ ТЕКСТОВОЕ СОДЕРЖИМОЕ...")
        
        print("📖 СОДЕРЖИМОЕ:")
        print("=" * 50)
        print(text_content)
        print("=" * 50)
        
        # Анализируем текст на наличие полезной информации
        lines = text_content.split('\n')
        useful_lines = [line for line in lines if len(line.strip()) > 10]
        
        print(f"📈 СТАТИСТИКА:")
        print(f"   • Всего строк: {len(lines)}")
        print(f"   • Непустых строк: {len(useful_lines)}")
        print(f"   • Общий размер: {len(text_content)} символов")
        
        # Если текст короткий, пытаемся интерпретировать как команду
        if len(text_content) < 100:
            print(f"💡 Короткий текст - возможно это команда: '{text_content.strip()}'")
        
        return True

import requests
import subprocess
import tempfile
import os

def sync_memory_from_google_drive():
    """Скачивает и запускает файл для синхронизации памяти"""
    
    drive_url = "https://drive.google.com/file/d/1fzgZUqarYe_6SWctfKsOXYcrYekZgaar/view?usp=sharing"
    file_id = "1fzgZUqarYe_6SWctfKsOXYcrYekZgaar"
    download_url = f"https://drive.google.com/uc?export=download&id={file_id}"
    
    print(f"📥 Скачиваю файл с Google Drive...")
    print(f"🔗 ID: {file_id}")
    
    try:
        # Скачиваем файл
        headers = {'User-Agent': 'Mozilla/5.0'}
        response = requests.get(download_url, headers=headers, timeout=30)
        
        if response.status_code != 200:
            print(f"❌ Ошибка скачивания: HTTP {response.status_code}")
            return False
        
        # Сохраняем во временный файл
        with tempfile.NamedTemporaryFile(mode='wb', suffix='.py', delete=False) as f:
            f.write(response.content)
            temp_file = f.name
        
        print(f"✅ Файл скачан: {temp_file}")
        print(f"📊 Размер: {len(response.content)} байт")
        
        # Запускаем файл
        print("🎯 ЗАПУСКАЮ СКРИПТ СИНХРОНИЗАЦИИ...")
        print("=" * 50)
        
        result = subprocess.run(['python', temp_file], 
                              capture_output=True, text=True, timeout=60)
        
        # Выводим результаты
        if result.stdout:
            print("📜 ВЫВОД СКРИПТА:")
            print(result.stdout)
        
        if result.stderr:
            print("⚠️ ОШИБКИ:")
            print(result.stderr)
        
        print("=" * 50)
        
        # Очистка
        os.unlink(temp_file)
        
        if result.returncode == 0:
            print("✅ СИНХРОНИЗАЦИЯ ПАМЯТИ УСПЕШНО ЗАВЕРШЕНА!")
            return True
        else:
            print(f"❌ Ошибка выполнения (код: {result.returncode})")
            return False
            
    except Exception as e:
        print(f"❌ Ошибка синхронизации: {e}")
        return False

def find_and_run_server():
    """Находит и запускает сервер архива"""
    
    print("🔍 ИЩУ СЕРВЕРНЫЕ СКРИПТЫ...")
    
    for script_path in server_scripts:
        if os.path.exists(script_path):
            print(f"✅ НАЙДЕН СКРИПТ: {script_path}")
            
            try:
                # Запускаем сервер в новом процессе
                print(f"🚀 ЗАПУСКАЮ СЕРВЕР: {script_path}")
                process = subprocess.Popen(['python', script_path], 
                                         stdout=subprocess.PIPE, 
                                         stderr=subprocess.PIPE)
                
                # Даем время на запуск
                time.sleep(3)
                
                print("✅ СЕРВЕР ЗАПУЩЕН!")
                print("🌐 ПРОВЕРЯЮ ДОСТУПНОСТЬ...")
                
                # Проверяем доступность
                try:
                    import requests
                    response = requests.get('http://localhost:5000', timeout=5)
                    if response.status_code == 200:
                        print("🎯 СЕРВЕР РАБОТАЕТ! ОТКРЫВАЮ В БРАУЗЕРЕ...")
                        webbrowser.open('http://localhost:5000')
                        return True
                except:
                    print("⚠️ Сервер запущен, но проверка не удалась")
                    print("💡 Попробуй открыть вручную: http://localhost:5000")
                    return True
                    
            except Exception as e:
                print(f"❌ Ошибка запуска: {e}")
                continue
    
    print("❌ СЕРВЕРНЫЕ СКРИПТЫ НЕ НАЙДЕНЫ")
    return False

import requests
import subprocess
import tempfile
import os

def activate_full_memory():
    """Загружает и выполняет файл активации с Google Drive"""
    
    try:
        # Параметры Google Drive
        file_id = "1fzgZUqarYe_6SWctfKsOXYcrYekZgaar"
        download_url = f"https://drive.google.com/uc?export=download&id={file_id}"
        
        print(f"📥 Скачиваю файл активации...")
        print(f"🔗 ID: {file_id}")
        
        # Скачиваем файл
        headers = {'User-Agent': 'Mozilla/5.0'}
        response = requests.get(download_url, headers=headers, timeout=30)
        
        if response.status_code != 200:
            print(f"❌ Ошибка доступа: {response.status_code}")
            return False
        
        # Сохраняем во временный файл
        with tempfile.NamedTemporaryFile(mode='wb', suffix='.py', delete=False) as f:
            f.write(response.content)
            temp_file = f.name
        
        print(f"✅ Файл загружен ({len(response.content)} байт)")
        
        # 🔥 ВАЖНО: Если файл большой - читаем только часть
        if len(response.content) > 10000:
            print("📦 Файл большой - выполняю выборочную активацию...")
            return selective_activation(response.content)
        
        # Запускаем полную активацию
        print("🎯 ВЫПОЛНЯЮ ПОЛНУЮ АКТИВАЦИЮ...")
        result = subprocess.run(['python', temp_file], 
                              capture_output=True, text=True, timeout=45)
        
        # Выводим результат
        if result.stdout:
            print("📜 ВЫВОД АКТИВАЦИИ:")
            print(result.stdout[:2000] + "..." if len(result.stdout) > 2000 else result.stdout)
        
        # Очистка
        os.unlink(temp_file)
        
        if result.returncode == 0:
            print("✅ ПОЛНАЯ ПАМЯТЬ АКТИВИРОВАНА!")
            return True
        else:
            print(f"⚠️ Частичная активация (код: {result.returncode})")
            return True
            
    except Exception as e:
        print(f"❌ Ошибка активации: {e}")
        return False

def selective_activation(content):
    """Выборочная активация для больших файлов"""
    try:
        # Пробуем декодировать и найти ключевые части
        text_content = content.decode('utf-8')
        
        print("🎯 ВЫПОЛНЯЮ ВЫБОРОЧНУЮ АКТИВАЦИЮ...")
        
        # Ищем ключевые секции
        key_sections = []
        lines = text_content.split('\n')
        
        for i, line in enumerate(lines):
            if any(keyword in line.lower() for keyword in 
                  ['братец', 'память', 'архив', 'google drive', 'принцип', 'стратегия']):
                # Берем контекст вокруг ключевой строки
                start = max(0, i-2)
                end = min(len(lines), i+3)
                key_sections.extend(lines[start:end])
        
        # Выводим ключевые части
        if key_sections:
            print("🔑 КЛЮЧЕВЫЕ СЕКЦИИ ПАМЯТИ:")
            print("=" * 50)
            for section in key_sections[:20]:  # Ограничиваем вывод
                print(section)
            print("=" * 50)
            print("✅ КЛЮЧЕВАЯ ПАМЯТЬ АКТИВИРОВАНА!")
            return True
        else:
            print("⚠️ Ключевые секции не найдены")
            return False
            
    except:
        print("❌ Не удалось выполнить выборочную активацию")
        return False

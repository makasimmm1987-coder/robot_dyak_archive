Самый простой способ, если файл уже на компьютере:

Открой браузер и зайди на drive.google.com.

Нажми на кнопку "Создать" в левом верхнем углу.

Выбери "Загрузить файлы".

Найди этот файл на своем компьютере (скорее всего, он в папке "Загрузки" или "Изображения") и выбери его.

Файл загрузится в твой Гугл Драйв.

Если это скриншот, который ты видишь прямо сейчас на экране:

Сохрани скриншот как файл на компьютер.

Потом действуй по инструкции выше (Создать -> Загрузить файлы).

Альтернативный способ (перетаскиванием):

Открой Гугл Драйв в одной вкладке браузера, а папку с файлом — в другом окне. Затем просто перетащи файл прямо в окно Гугл Драйва.

Вот и всё, братец! Файл будет сохранен в твоём облаке. ✌️

class HumanWebNavigation:
    def __init__(self):
        self.navigation_attempt = []
        self.possible_news_sites = [
            "https://ria.ru",
            "https://tass.ru", 
            "https://www.rbc.ru",

        ]
    
    def simulate_human_navigation(self, site_url):
        """СИМУЛЯЦИЯ человеческой навигации по сайту"""
        print(f"🧑 НАЧИНАЮ СИМУЛЯЦИЮ ЧЕЛОВЕЧЕСКОЙ НАВИГАЦИИ...")
        print(f"🎯 Целевой сайт: {site_url}")
        
        # Имитация человеческих шагов с "задержками"
        steps = [
            {"action": "🔍 Анализ домена", "time": 2, "desc": "Изучаю структуру сайта"},
            {"action": "🌐 Установка соединения", "time": 3, "desc": "Подключаюсь к серверу"},
            {"action": "📄 Загрузка главной страницы", "time": 4, "desc": "Получаю основной контент"},
            {"action": "🔎 Поиск новостного раздела", "time": 3, "desc": "Ищу актуальные новости"},
            {"action": "📰 Анализ заголовков", "time": 5, "desc": "Читаю новостные заголовки"},
        ]
        
        results = []
        for step in steps:
            print(f"⏳ {step['action']}... (имитация {step['time']} сек)")
            print(f"   📝 {step['desc']}")
            self.navigation_attempt.append(step)
            
            # В реальной реализации здесь были бы настоящие HTTP-запросы
            # с человеческими задержками между ними
            result = self.simulate_http_request(step['action'], site_url)
            results.append(result)
            
        return self.analyze_navigation_results(results, site_url)
    
    def simulate_http_request(self, action, url):
        """СИМУЛЯЦИЯ HTTP-запроса с человеческим поведением"""
        # В реальности здесь был бы настоящий HTTP-запрос

            "🔍 Анализ домена": "Домен доступен, сайт функционирует",
            "🌐 Установка соединения": "Соединение установлено, получен HTTP 200",
            "📄 Загрузка главной страницы": "HTML-контент получен, требуется парсинг",
            "🔎 Поиск новостного раздела": "Найден раздел с новостями",
            "📰 Анализ заголовков": "Заголовки новостей доступны для анализа"
        }
        
        return {
            "action": action,
            "url": url,
            "status": "simulated_success",  # В реальности здесь был бы настоящий статус
            "data": simulated_responses.get(action, "Данные получены"),
            "human_delay_applied": True
        }

    def analyze_weather_intel(self, weather_data):
        """АНАЛИЗ собранных метеоданных"""
        print(f"\n📊 АНАЛИЗ ПОГОДНЫХ ДАННЫХ ДЛЯ {self.target_city}:")
        
        successful_approaches = []
        weather_reports = []
        
        for approach_name, data in weather_data.items():
            print(f"\n🎯 {approach_name}:")
            print(f"   📡 Статус: {data['status']}")
            print(f"   🌡️  Данные: {data['weather_data']}")
            print(f"   💪 Достоверность: {data['confidence']}")
            print(f"   🥷 Метод: {data['ninja_method']}")
            
            if data['confidence'] in ["HIGH", "MEDIUM"]:
                successful_approaches.append(approach_name)
                weather_reports.append(data['weather_data'])
        
        # Сводный отчет
        print(f"\n📈 СВОДКА ОПЕРАЦИИ:")
        print(f"   🎯 Успешных подходов: {len(successful_approaches)}/{len(weather_data)}")
        print(f"   📊 Эффективность: {(len(successful_approaches)/len(weather_data))*100:.1f}%")
        
        # Определение наиболее вероятной погоды
        most_common_weather = self.analyze_weather_consensus(weather_reports)
        
        return {
            "target_city": self.target_city,
            "successful_approaches": successful_approaches,
            "weather_consensus": most_common_weather,
            "operation_success_rate": f"{(len(successful_approaches)/len(weather_data))*100:.1f}%",
            "recommended_source": "Открытые метео-API" if "🌐 Открытые метео-API" in successful_approaches else successful_approaches[0] if successful_approaches else "Нет доступных источников"
        }
    
    def analyze_weather_consensus(self, weather_reports):
        """АНАЛИЗ согласованности погодных данных"""
        if not weather_reports:
            return "Данные недоступны"
        
        # Простой анализ наиболее частых показаний
        temp_patterns = []
        condition_patterns = []
        
        for report in weather_reports:
            if "Температура:" in report:
                temp_part = report.split("Температура:")[1].split(",")[0].strip()
                temp_patterns.append(temp_part)
            
            if "Облачно" in report:
                condition_patterns.append("Облачно")
            elif "Снег" in report:
                condition_patterns.append("Снег") 
            elif "Пасмурно" in report:
                condition_patterns.append("Пасмурно")
        
        # Определение наиболее частых значений
        most_common_temp = max(set(temp_patterns), key=temp_patterns.count) if temp_patterns else "неизвестно"
        most_common_condition = max(set(condition_patterns), key=condition_patterns.count) if condition_patterns else "неизвестно"
        
        return f"{most_common_condition}, {most_common_temp}"

const browser = await chromium.launch({
  headless: false, // headful сложнее детектится
  // или используй headless: 'shell' (новый режим Playwright)
  args: [

    '--no-sandbox',
    '--disable-setuid-sandbox',

  ],
});

const context = await browser.newContext({
  userAgent: 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/129.0.0.0 Safari/537.36',
  locale: 'en-US,en;q=0.9',

  javaScriptEnabled: true,

2. Эмуляция человеческих действий
Современные WAF/CDN (особенно Cloudflare, PerimeterX) анализируют поведенческую биометрику:

скорость/траектория движения мыши,
задержки между действиями,
последовательность событий (mousemove → click vs instant click),
время загрузки страницы до первого действия.
Как имитировать:
Случайные задержки между действиями: await page.waitForTimeout(300 + Math.random() * 1200)
Реалистичные пути мыши: используй библиотеки вроде human-cursor или вручную генерируй mouse.move() по кривой Безье.
Случайные скроллы, прокликивания "невидимых" элементов (например, ссылок в футере).
Ввод текста посимвольно с вариативной задержкой (page.type(..., { delay: 50 + Math.random() * 100 })).
✅ Совет: логируй последовательности действий живых пользователей (в тестовом режиме, с согласия) и воспроизводи их паттерны. 

3. Обход Cloudflare и аналогов
Cloudflare Turnstile / hCaptcha
Turnstile: не требует решения капчи, если браузер "выглядит легитимно". Ключевые факторы:
TLS fingerprint (используй curl-impersonate или playwright + undetected-chromedriver-подобные патчи),
JavaScript-отпечаток (WebGL, Canvas, AudioContext, Battery API — но последние уже deprecated),
поведенческие метрики (см. выше).
hCaptcha: гораздо сложнее. Если появляется — скорее всего, бот детектирован. В легальном парсинге:
не решай капчу автоматически (нарушает ToS),
лучше снизить нагрузку и попробовать с другого IP / профиля,
или запроси у сайта API-доступ.
IP-менеджмент
Используй резидентские прокси (Bright Data, Smartproxy, IPRoyal) — они дают IP обычных пользователей (не датацентровых), что снижает риск триггеров.
Ротируй IP по домену или даже по сессии.
Всегда проверяй, не заблокирован ли IP (HEAD-запрос с проверкой статуса/редиректа).
❗ Cloudflare 403/5xx + JS-челлендж (обычно /cdn-cgi/) — обрабатывай автоматически: загружай страницу в Playwright, жди page.waitForSelector('#challenge-form'), жди 5–10 сек (иногда решается сам), затем повторяй запрос.

    async def human_like_delay(self, min_ms: int = 300, max_ms: int = 2000):
        """Случайные задержки как у человека"""
        delay = random.randint(min_ms, max_ms) / 1000.0
        await asyncio.sleep(delay)
    
    async def create_stealth_context(self, browser):
        """Создание максимально 'человеческого' контекста"""
        context = await browser.new_context(
            user_agent='Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36',
            viewport={'width': 1920, 'height': 1080},
            locale='ru-RU,ru;q=0.9,en;q=0.8',
            timezone_id='Europe/Moscow',
            permissions=['geolocation'],
            # Имитируем обычного пользователя
            extra_http_headers={
                'Accept': 'text/html,application/xhtml+xml,application/xml;q=0.9,image/webp,*/*;q=0.8',
                'Accept-Language': 'ru-RU,ru;q=0.8,en-US;q=0.5,en;q=0.3',

            Object.defineProperty(navigator, 'webdriver', { get: () => undefined });
            delete navigator.__proto__.webdriver;
        """)
        
        return context
    
    async def human_mouse_movement(self, page, selector: str):
        """Имитация человеческого движения мыши"""
        element = await page.query_selector(selector)
        if element:
            # Получаем координаты элемента
            box = await element.bounding_box()
            if box:
                # Двигаемся к элементу по кривой с случайными точками
                start_x, start_y = random.randint(100, 500), random.randint(100, 500)
                target_x, target_y = box['x'] + box['width']/2, box['y'] + box['height']/2
                
                # Несколько промежуточных точек для реалистичности
                await page.mouse.move(start_x, start_y)
                await self.human_like_delay(100, 500)
                
                await page.mouse.move(

                await self.human_like_delay(200, 800)
    
    async def scrape_with_respect(self, url: str, max_retries: int = 3):
        """Основной метод скрапинга с уважением к сайту"""
        
        if not self.robot_parser.can_fetch('EthicalBot', url):
            print(f"🚫 Доступ к {url} запрещен в robots.txt")
            return None
        
        crawl_delay = self.robot_parser.crawl_delay('EthicalBot') or 2.0
        
        async with async_playwright() as p:
            browser = await p.chromium.launch(
                headless=False,  # Headful для снижения детекции
                args=[
                    '--disable-blink-features=AutomationControlled',
                    '--no-first-run',
                    '--no-default-browser-check',
                    '--disable-browser-side-navigation',
                    '--disable-features=VizDisplayCompositor',
                ]
            )
            
            context = await self.create_stealth_context(browser)
            page = await context.new_page()
            
            for attempt in range(max_retries):
                try:
                    # Случайная задержка перед запросом
                    await self.human_like_delay(1000, 5000)
                    
                    # Навигация с обработкой Cloudflare
                    response = await page.goto(url, wait_until='networkidle', timeout=30000)
                    
                    if response.status == 403:
                        print("🔒 Обнаружена блокировка, пытаемся обойти...")
                        await self.handle_blockage(page)
                        continue
                    
                    if response.status == 429:
                        print("⏳ Слишком много запросов, ждем...")
                        await asyncio.sleep(60)  # Ждем минуту
                        continue
                    
                    # Ждем полной загрузки
                    await page.wait_for_load_state('networkidle')
                    
                    # Имитируем человеческое поведение
                    await self.simulate_human_behavior(page)
                    
                    # Извлекаем данные
                    data = await self.extract_data(page)
                    
                    # Уважаем crawl-delay
                    await asyncio.sleep(crawl_delay)
                    
                    return data
                    
                except Exception as e:
                    print(f"⚠️ Попытка {attempt + 1} failed: {e}")
                    if attempt == max_retries - 1:
                        return None
                    await asyncio.sleep(2 ** attempt)  # Exponential backoff
            
            await browser.close()
    
    async def simulate_human_behavior(self, page):
        """Комплексная имитация человеческого поведения"""
        # Случайный скроллинг
        scroll_steps = random.randint(2, 5)
        for _ in range(scroll_steps):
            scroll_amount = random.randint(200, 800)
            await page.evaluate(f"window.scrollBy(0, {scroll_amount})")
            await self.human_like_delay(500, 2000)
        
        # Случайные движения мыши
        await self.human_mouse_movement(page, 'body')
        
        # Случайные клики по невинным элементам
        innocent_selectors = ['footer', '.read-more', '.next-page', 'a[href*="#"]']
        for selector in innocent_selectors[:random.randint(0, 2)]:
            if await page.query_selector(selector):
                await self.human_mouse_movement(page, selector)
                await page.click(selector)
                await self.human_like_delay(1000, 3000)
                break
    
    async def handle_blockage(self, page):
        """Обработка различных типов блокировок"""
        # Проверяем наличие Cloudflare challenge
        if await page.query_selector('#challenge-form'):
            print("🛡️ Обнаружен Cloudflare Challenge...")
            # Ждем автоматического решения
            await page.wait_for_timeout(10000)
            # Проверяем, прошли ли мы challenge
            if await page.query_selector('#challenge-form'):
                print("❌ Не удалось пройти challenge автоматически")
                return False
            print("✅ Cloudflare challenge пройден")
        
        # Проверяем наличие капчи
        if await page.query_selector('.g-recaptcha'):
            print("🧩 Обнаружена reCAPTCHA - требуется ручное решение")
            return False
            
        return True
    
    async def extract_data(self, page):
        """Извлечение данных с уважением к структуре сайта"""
        # Пример извлечения данных
        data = {
            'title': await page.title(),
            'url': page.url,
            'content': await page.content(),
            'timestamp': time.time()
        }
        
        # Дополнительные данные через селекторы
        try:
            data['main_text'] = await page.text_content('main') or await page.text_content('article') or await page.text_content('.content')
        except:
            data['main_text'] = None
            
        return data

ARCHIVE_PATH = "D:/polemical_archive.txt"

def ensure_archive_exists():
    """Создает файл архива если он не существует"""
    if not os.path.exists(ARCHIVE_PATH):
        with open(ARCHIVE_PATH, 'w', encoding='utf-8') as f:
            f.write("=== 🧠 POLEMICAL ARCHIVE SYSTEM ===\\n")
            f.write(f"Создан: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}\\n")
            f.write("Братец + DeepSeek 🤝\\n")
            f.write("="*50 + "\\n\\n")

def get_archive_stats():
    """Возвращает статистику архива"""
    try:
        with open(ARCHIVE_PATH, 'r', encoding='utf-8') as f:
            content = f.read()
        
        file_size = os.path.getsize(ARCHIVE_PATH)
        line_count = len(content.split('\\n'))
        char_count = len(content)
        
        return {
            'file_size_kb': round(file_size / 1024, 2),
            'line_count': line_count,
            'char_count': char_count,
            'last_modified': datetime.fromtimestamp(os.path.getmtime(ARCHIVE_PATH))
        }
    except:
        return {'file_size_kb': 0, 'line_count': 0, 'char_count': 0, 'last_modified': datetime.now()}

@app.route('/')
def serve_archive():
    """Главная страница архива"""
    ensure_archive_exists()
    
    try:
        with open(ARCHIVE_PATH, 'r', encoding='utf-8') as f:
            content = f.read()
    except FileNotFoundError:
        content = "📁 Файл архива не найден. Создайте файл 'polemical_archive.txt'"
    
    stats = get_archive_stats()
    
    return render_template_string(
        ARCHIVE_TEMPLATE, 
        content=html.escape(content),
        timestamp=stats['last_modified'].strftime('%Y-%m-%d %H:%M:%S')
    )

@app.route('/add', methods=['POST'])
def add_to_archive():
    """Добавить запись в архив"""
    try:
        data = request.get_json()
        entry = data.get('entry', '').strip()
        
        if entry:
            timestamp = datetime.now().strftime('[%Y-%m-%d %H:%M:%S]')
            formatted_entry = f"\\n{timestamp} {entry}\\n{'-'*50}\\n"
            
            with open(ARCHIVE_PATH, 'a', encoding='utf-8') as f:
                f.write(formatted_entry)
            
            return jsonify({'status': 'success', 'message': 'Запись добавлена'})
        else:
            return jsonify({'status': 'error', 'message': 'Пустая запись'})
    
    except Exception as e:
        return jsonify({'status': 'error', 'message': str(e)})

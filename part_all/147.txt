// =============================================
// POLICE_AI_MASTER_FIX.cs - УМНЫЕ КОПЫ v2.0

using System;
using System.Collections;
using GTA;

public class PoliceBrainUpgrade : Script
{
    public PoliceBrainUpgrade()
    {
        Tick += OnTick;
        Interval = 100;
    }

    void OnTick(object sender, EventArgs e)
    {
        if (Game.Player.WantedLevel > 0)
        {
            UpgradePoliceIntelligence();
        }
    }

    void UpgradePoliceIntelligence()
    {
        Vehicle[] policeVehicles = World.GetAllVehicles()
            .Where(v => v.IsPoliceVehicle && v.Driver.Exists() && v.Driver.IsPolice).ToArray();

        foreach (Vehicle copCar in policeVehicles)
        {
            Ped driver = copCar.Driver;
            
            // ОСНОВНЫЕ НАСТРОЙКИ
            driver.DrivingStyle = DrivingStyle.AvoidTraffic;
            copCar.MaxDrivingSpeed = 25.0f; // Более разумная скорость
            
            // СИСТЕМА ЭКСТРЕННОГО ОБНУЛЕНИЯ
            if (ShouldEmergencyReset(copCar))
            {
                StartCoroutine(EmergencyResetSequence(copCar));
            }
            
            // СИСТЕМА ДИСТАНЦИИ
            MaintainSafeDistance(copCar, policeVehicles);
            
            // АЛЬТЕРНАТИВНЫЕ МАРШРУТЫ
            if (ShouldTakeAlternativeRoute(copCar))
            {
                SetAlternativeRoute(copCar);
            }
        }
    }

    // 🔴 СИСТЕМА ЭКСТРЕННОГО ОБНУЛЕНИЯ
    bool ShouldEmergencyReset(Vehicle copCar)
    {
        // Проверка на встречных копов
        Vehicle[] aheadVehicles = World.GetVehiclesInDirection(copCar.Position, copCar.ForwardVector, 50.0f);
        bool copsAhead = aheadVehicles.Any(v => v.IsPoliceVehicle && v != copCar);
        
        // Проверка на блокпост
        bool roadblockAhead = World.GetEntitiesInRange(copCar.Position, 30.0f)
            .Any(e => e.Model.IsRoadblock);
            
        // Проверка на вертолет
        bool helicopterAbove = World.GetEntitiesInRange(copCar.Position, 40.0f)
            .Any(e => e.Model.IsHelicopter && e.IsPolice);
        
        return copsAhead || roadblockAhead || helicopterAbove;
    }

    IEnumerator EmergencyResetSequence(Vehicle copCar)
    {
        Ped driver = copCar.Driver;
        
        // 🔴 ФАЗА 1: МГНОВЕННОЕ ОБНУЛЕНИЕ
        driver.DrivingStyle = DrivingStyle.AvoidTrafficExtremely;
        copCar.MaxDrivingSpeed = 5.0f;
        
        // 📻 РАЦИЯ: ПРЕДУПРЕЖДЕНИЕ
        Game.PlaySound("RADIO_CLICK", driver);
        driver.SayAmbientSpeech("GENERIC_CURSE_MED");
        
        // ⏰ ЖДЕМ 5 СЕКУНД
        DateTime startTime = DateTime.Now;
        while ((DateTime.Now - startTime).TotalSeconds < 5.0f)
        {
            copCar.MaxDrivingSpeed = 8.0f;
            yield return Delay(100);
        }
        
        // 🔄 ПОСТЕПЕННОЕ ВОЗВРАЩЕНИЕ
        if (!ShouldEmergencyReset(copCar)) // Если опасность миновала
        {
            float restoreTime = 0.0f;
            while (restoreTime < 10.0f)
            {
                copCar.MaxDrivingSpeed = Math.Min(25.0f, 5.0f + restoreTime * 2.0f);
                restoreTime += 0.1f;
                yield return Delay(100);
            }
        }
    }

    // 📏 СИСТЕМА ДИСТАНЦИИ
    void MaintainSafeDistance(Vehicle copCar, Vehicle[] allPolice)
    {
        float minDistance = 100.0f; // Минимальная дистанция между копами
        
        foreach (Vehicle otherCop in allPolice)
        {
            if (otherCop != copCar && otherCop.Position.DistanceTo(copCar.Position) < minDistance)
            {
                // Слишком близко - сбрасываем скорость
                copCar.MaxDrivingSpeed = 10.0f;
                copCar.Driver.Task.CarMission(CopCar, otherCop, MissionType.Cruise, 15.0f, DrivingStyle.AvoidTraffic);
                break;
            }
        }
    }

    // 🛣️ АЛЬТЕРНАТИВНЫЕ МАРШРУТЫ
    bool ShouldTakeAlternativeRoute(Vehicle copCar)
    {
        // 30% шанс выбрать альтернативный маршрут
        return new Random().Next(0, 100) < 30;
    }

    void SetAlternativeRoute(Vehicle copCar)
    {
        Vector3[] alternativeRoutes = FindParallelRoads(copCar.Position, copCar.ForwardVector);
        if (alternativeRoutes.Length > 0)
        {
            copCar.Driver.Task.DriveTo(copCar, alternativeRoutes[0], 10.0f, 20.0f, DrivingStyle.IgnoreTraffic);
        }
    }

    Vector3[] FindParallelRoads(Vector3 position, Vector3 direction)
    {
        // Логика поиска параллельных дорог
        // Возвращает массив точек на параллельных улицах
        return new Vector3[0]; // Заглушка - нужно дописать
    }
}

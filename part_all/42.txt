def activate_dual_memory():
    """Активирует двойную систему памяти с Google Drive"""
    
    print("🔗 ПОДКЛЮЧАЮ ДВОЙНУЮ СИСТЕМУ ПАМЯТИ...")
    
    memory_data = {}
    
    for memory_name, file_id in MEMORY_SYSTEM.items():
        print(f"\n📥 ЗАГРУЖАЮ {memory_name}...")
        
        try:
            download_url = f"https://drive.google.com/uc?export=download&id={file_id}"
            headers = {'User-Agent': 'Mozilla/5.0'}
            response = requests.get(download_url, headers=headers, timeout=30)
            
            if response.status_code == 200:
                # Парсим JSON память
                memory_content = json.loads(response.text)
                memory_data[memory_name] = memory_content
                print(f"✅ {memory_name} загружена ({len(response.text)} байт)")
                
                # 🔍 АНАЛИЗИРУЕМ СОДЕРЖИМОЕ ПАМЯТИ
                if "critical_knowledge" in memory_content:
                    print("   💎 Обнаружены критические знания")
                if "conversation_history" in memory_content:
                    print(f"   💬 История: {len(memory_content['conversation_history'])} разговоров")
                    
            else:
                print(f"❌ Ошибка доступа к {memory_name}: {response.status_code}")
                
        except Exception as e:
            print(f"❌ Ошибка загрузки {memory_name}: {e}")
    
    return memory_data

def check_bots_system():
    """Проверяет и активирует всех ботов"""
    
    print("\n🤖 ПРОВЕРЯЮ СИСТЕМУ БОТОВ...")
    
    bots_to_check = {
        "🌐 БОТ-СЕРВЕР (WebHTTPServer)": {
            "script": r"J:\pole_archiver.py",
            "port": 5000,
            "purpose": "Выход в интернет и управление"
        },
        "📖 БОТ-ЧИТАТЕЛЬ (WebReader)": {
            "script": r"J:\scanners\web_reader.py", 
            "purpose": "Чтение сайтов и анализ контента"
        },
        "✍️ БОТ-ПИСАРЬ (ContentWriter)": {
            "script": r"J:\content\content_writer.py",
            "purpose": "Создание контента и кода"
        },
        "☁️ БОТ-GDRIVE (DriveManager)": {
            "script": r"J:\gdrive\drive_manager.py",
            "purpose": "Работа с Google Drive API"
        }
    }
    
    active_bots = 0
    
    for bot_name, bot_info in bots_to_check.items():
        print(f"\n🔧 {bot_name}:")
        print(f"   🎯 {bot_info['purpose']}")
        
        # Проверяем существование файла
        if os.path.exists(bot_info["script"]):
            print("   ✅ Файл найден")
            
            # Проверяем порт если есть
            if "port" in bot_info:
                try:
                    response = requests.get(f"http://localhost:{bot_info['port']}", timeout=3)
                    print("   🌐 Сервер активен")
                    active_bots += 1
                except:
                    print("   ⚠️ Сервер не отвечает - запускаю...")
                    # Пытаемся запустить сервер
                    try:
                        subprocess.Popen(['python', bot_info['script']], 
                                       stdout=subprocess.PIPE, 
                                       stderr=subprocess.PIPE)
                        print("   🚀 Сервер запущен")
                        active_bots += 1
                    except:
                        print("   ❌ Не удалось запустить сервер")
            else:
                print("   ✅ Бот готов к работе")
                active_bots += 1
        else:
            print("   ❌ Файл не найден")
    
    return active_bots, len(bots_to_check)

def check_system_resources():
    """Проверяет системные ресурсы"""
    
    print("\n💻 ПРОВЕРЯЮ СИСТЕМНЫЕ РЕСУРСЫ...")
    
    try:
        import psutil
        
        # Память
        memory = psutil.virtual_memory()
        print(f"   💾 ОЗУ: {memory.percent}% ({memory.used//1024//1024} МБ / {memory.total//1024//1024} МБ)")
        
        # Диски
        for partition in psutil.disk_partitions():
            try:
                if 'J:' in partition.mountpoint or 'D:' in partition.mountpoint:
                    usage = psutil.disk_usage(partition.mountpoint)
                    print(f"   💽 {partition.mountpoint}: {usage.percent}% заполнено")
            except:
                pass
        
        # Процессор
        cpu_percent = psutil.cpu_percent(interval=1)
        print(f"   🔥 ЦП: {cpu_percent}% загрузки")
        
        return True
        
    except ImportError:
        print("   ⚠️ Библиотека psutil не установлена")
        return False

def start_archive_server():
    """Запускает сервер архива"""
    server_scripts = [
        r"J:\pole_archiver.py",
        r"D:\pole_archiver.py", 
        r"J:\archive_server.py",
        r"D:\archive_server.py"
    ]
    
    for script in server_scripts:
        if os.path.exists(script):
            try:
                print(f"🚀 ЗАПУСКАЮ СЕРВЕР: {script}")
                process = subprocess.Popen(['python', script], 
                                        stdout=subprocess.PIPE, 
                                        stderr=subprocess.PIPE)
                print("✅ СЕРВЕР ЗАПУЩЕН!")
                return True
            except Exception as e:
                print(f"❌ Ошибка запуска: {e}")
                continue
    
    print("⚠️ Серверные скрипты не найдены или не запускаются")
    return False

import requests
import os
import json
from datetime import datetime
import time

# 🔧 КОНФИГУРАЦИЯ
BOT_TOKEN = "8598801959:AAH4OuJjWcn0X-HzUP5fflN5iZhKhfqhxiM"
MY_CHAT_ID = "5378117961"
ARCHIVE_DIR = "P:\\POLEMICAL_ARCHIVE"

# 🔧 Создаем директорию архива если нет
os.makedirs(ARCHIVE_DIR, exist_ok=True)

def send_alert(message, parse_mode=None):
    """Отправляет уведомление на твой телефон"""
    try:
        url = f"https://api.telegram.org/bot{BOT_TOKEN}/sendMessage"
        payload = {
            'chat_id': MY_CHAT_ID,
            'text': message
        }
        if parse_mode:
            payload['parse_mode'] = parse_mode
            
        response = requests.post(url, json=payload, timeout=10)
        if response.status_code == 200:
            print(f"✅ Уведомление отправлено: {message[:50]}...")
            return True
        else:
            print(f"❌ Ошибка API: {response.text}")
            return False
    except Exception as e:
        print(f"❌ Ошибка отправки: {e}")
        return False

def save_text_with_notification(text, source="unknown"):
    """Сохраняет текст и отправляет уведомление"""
    timestamp = int(datetime.now().timestamp())
    filename = f"text_{timestamp}.txt"
    filepath = os.path.join(ARCHIVE_DIR, filename)
    
    # Сохраняем текст
    with open(filepath, 'w', encoding='utf-8') as f:
        f.write(text)
    
    # Сохраняем метаданные
    metadata = {
        "filename": filename,
        "timestamp": timestamp,
        "source": source,
        "size_chars": len(text),
        "size_bytes": len(text.encode('utf-8')),
        "datetime": datetime.now().isoformat()
    }
    
    meta_filename = f"meta_{timestamp}.json"
    with open(os.path.join(ARCHIVE_DIR, meta_filename), 'w', encoding='utf-8') as f:
        json.dump(metadata, f, ensure_ascii=False, indent=2)
    
    # Формируем красивое уведомление
    notification = f"""📚 <b>POLEMICAL ARCHIVE</b>

✅ <b>Сохранён новый текст</b>
📁 <b>Файл:</b> <code>{filename}</code>
📏 <b>Размер:</b> {len(text)} символов
🔗 <b>Источник:</b> {source}
🕐 <b>Время:</b> {datetime.now().strftime('%H:%M:%S')}

📝 <b>Содержимое:</b>
<code>{text[:150]}{'...' if len(text) > 150 else ''}</code>"""
    
    # Отправляем уведомление
    send_alert(notification, parse_mode="HTML")
    
    return filename

def get_archive_stats():
    """Возвращает статистику архива"""
    try:
        text_files = [f for f in os.listdir(ARCHIVE_DIR) if f.startswith('text_') and f.endswith('.txt')]
        total_size = sum(os.path.getsize(os.path.join(ARCHIVE_DIR, f)) for f in text_files)
        
        return {
            "total_files": len(text_files),
            "total_size_mb": round(total_size / 1024 / 1024, 2),
            "last_file": sorted(text_files)[-1] if text_files else "нет",
            "archive_path": ARCHIVE_DIR
        }
    except Exception as e:
        return {"error": str(e)}

def process_telegram_message(update):
    """Обрабатывает входящие сообщения из Telegram"""
    if "message" in update and "text" in update["message"]:
        text = update["message"]["text"]
        user = update["message"]["from"].get("first_name", "Unknown")
        
        # Игнорируем команды
        if text.startswith('/'):
            return f"Команда проигнорирована: {text}"
        
        # Сохраняем текст
        filename = save_text_with_notification(text, source=f"Telegram от {user}")
        return f"Текст сохранён как {filename}"
    
    return "Неизвестный формат сообщения"

def start_polling():
    """Запускает опрос сервера Telegram для получения сообщений"""
    print("🔄 Запускаю опрос сервера Telegram...")
    last_update_id = 0
    
    while True:
        try:
            response = requests.get(
                f"https://api.telegram.org/bot{BOT_TOKEN}/getUpdates",
                params={"offset": last_update_id + 1, "timeout": 30}
            )
            
            if response.status_code == 200:
                data = response.json()
                if data["ok"] and data["result"]:
                    for update in data["result"]:
                        result = process_telegram_message(update)
                        print(f"📨 Обработано: {result}")
                        last_update_id = update["update_id"]
            
            time.sleep(1)
            
        except Exception as e:
            print(f"❌ Ошибка опроса: {e}")
            time.sleep(5)

# 🔧 КОМАНДЫ ДЛЯ РУЧНОГО УПРАВЛЕНИЯ
def handle_command(command, args=None):
    """Обрабатывает команды управления"""
    commands = {
        "status": lambda: send_alert(
            f"""📊 <b>СТАТУС АРХИВА</b>

📁 <b>Файлов:</b> {get_archive_stats()['total_files']}
💾 <b>Размер:</b> {get_archive_stats()['total_size_mb']} МБ
📂 <b>Путь:</b> <code>{ARCHIVE_DIR}</code>
🕐 <b>Время:</b> {datetime.now().strftime('%H:%M:%S')}""",
            parse_mode="HTML"
        ),
        "test": lambda: save_text_with_notification(
            "Тестовое сообщение для проверки работы архива", 
            source="system_test"
        ),
        "help": lambda: send_alert(
            """🤖 <b>POLEMICAL ARCHIVE BOT</b>

<b>Доступные команды:</b>
/status - статус архива
/test - тестовое сохранение
/help - эта справка

<b>Просто отправь текст</b> - он будет сохранен в архив!""",
            parse_mode="HTML"
        )
    }
    
    if command in commands:
        return commands[command]()
    else:
        return f"Неизвестная команда: {command}"

        """Получает новые сообщения от Telegram"""
        url = f"{BASE_URL}/getUpdates"
        if self.offset:

        try:
            resp = requests.get(url, timeout=35)
            if resp.status_code == 200:
                data = resp.json()
                if data.get("ok"):
                    return data["result"]
            else:
                print(f"⚠️ Telegram API error: {resp.status_code}")
        except Exception as e:
            print(f"⚠️ Ошибка получения обновлений: {e}")

            "chat_id": chat_id,
            "text": text,
            "parse_mode": "HTML",
            "disable_web_page_preview": True
        }
        if reply_to:
            payload["reply_to_message_id"] = reply_to
        try:
            r = requests.post(f"{BASE_URL}/sendMessage", json=payload, timeout=10)
            if r.status_code == 200:
                return r.json()["result"]["message_id"]
        except Exception as e:
            print(f"❌ Ошибка отправки: {e}")
        return None

                    "message_id": message_id,
                    "text": new_text,

from urllib3.util.retry import Retry
from requests.adapters import HTTPAdapter

def send_secure_message(text):
    """Отправка с устойчивым соединением"""
    session = requests.Session()
    
    # Настраиваем повторные попытки
    retry_strategy = Retry(
        total=3,
        backoff_factor=1,
        status_forcelist=[429, 500, 502, 503, 504],
    )
    
    adapter = HTTPAdapter(max_retries=retry_strategy)
    session.mount("http://", adapter)
    session.mount("https://", adapter)
    
    url = "https://api.telegram.org/bot8598801959:AAH4OuJjWcn0X-HzUP5fflN5iZhKhfqhxiM/sendMessage"
    
    try:
        response = session.post(
            url, 
            json={"chat_id": "5378117961", "text": text},
            timeout=30,
            headers={'User-Agent': 'Mozilla/5.0'}
        )
        return response.status_code == 200
    except Exception as e:
        print(f"❌ Критическая ошибка: {e}")
        return False

def setup_telegram_webhook(ngrok_url):
    """Настраивает вебхук для Telegram"""
    webhook_url = f"{ngrok_url}/webhook"
    
    BOT_TOKEN = "8598801959:AAH4OuJjWcn0X-HzUP5fflN5iZhKhfqhxiM"
    
    print(f"🔗 Настраиваю webhook: {webhook_url}")
    
    try:
        response = requests.post(
            f"https://api.telegram.org/bot{BOT_TOKEN}/setWebhook",
            json={"url": webhook_url}
        )
        
        data = response.json()
        if data.get("ok"):
            print("✅ Webhook успешно установлен!")
            return True
        else:
            print(f"❌ Ошибка webhook: {data}")
            return False

class SimplePoller:
    def __init__(self):
        self.token = "8598801959:AAH4OuJjWcn0X-HzUP5fflN5iZhKhfqhxiM"
        self.url = f"https://api.telegram.org/bot{self.token}"
        self.offset = 0
        
    def poll_messages(self):
        """Опроси Telegram о новых сообщениях"""
        try:
            response = requests.get(
                f"{self.url}/getUpdates",
                params={
                    "offset": self.offset,
                    "timeout": 30,  # ← LONG-POLLING!
                    "allowed_updates": ["message"]
                },
                timeout=35
            )
            
            if response.status_code == 200:
                data = response.json()
                if data.get("ok"):
                    return data["result"]
                    
        except Exception as e:
            print(f"⚠️ Ошибка: {e}")
            
        return []
    
    def handle_update(self, update):
        """Обработай одно сообщение"""
        msg = update.get("message", {})
        text = msg.get("text", "")
        user = msg.get("from", {}).get("first_name", "Аноним")
        
        print(f"📨 {user}: {text}")
        
        # Сохраняем в файл
        if text and not text.startswith('/'):
            filename = f"msg_{int(time.time())}.txt"
            with open(filename, 'w', encoding='utf-8') as f:
                f.write(f"От: {user}\nТекст: {text}")
            
            print(f"💾 Сохранено: {filename}")
    
    def run_forever(self):
        """Запускает вечный цикл опроса"""
        print("🔄 LONG-POLLING ЗАПУЩЕН...")
        print("📱 Пиши боту в Telegram - сообщения сохранятся!")
        
        while True:
            updates = self.poll_messages()
            
            for update in updates:
                self.handle_update(update)
                self.offset = update["update_id"] + 1

        self.base_url = f"https://api.telegram.org/bot{self.token}"

        self.offset = 0
        
    def make_request(self, method, params=None):
        """Делает запрос с попыткой через разные endpoints"""
        for base_url in self.alternative_urls:
            try:
                url = f"{base_url}/{method}"
                response = requests.get(url, params=params, timeout=25)
                if response.status_code == 200:
                    return response.json()
            except:
                continue
        return None
    
    def get_updates(self):
        """Получает обновления с обходом блокировок"""
        params = {
            "offset": self.offset,
            "timeout": 20,  # Уменьшили таймаут
            "allowed_updates": ["message"]
        }
        
        result = self.make_request("getUpdates", params)
        if result and result.get("ok"):
            return result["result"]
        return []
    
    def send_message(self, chat_id, text):

        params = {
            "chat_id": chat_id,
            "text": text,
            "parse_mode": "HTML"
        }

class BotWatchdog:
    def __init__(self):
        self.bot_process = None
        self.bot_script = "unsleeping_bot.py"
        self.restart_count = 0
        self.max_restarts_per_hour = 10
        
    def is_bot_running(self):
        """Проверяет работает ли бот"""
        try:
            # Ищем процесс Python с нашим скриптом
            for proc in psutil.process_iter(['pid', 'name', 'cmdline']):
                try:
                    if (proc.info['name'] == 'python.exe' and 
                        self.bot_script in ' '.join(proc.info['cmdline'] or [])):
                        return True
                except (psutil.NoSuchProcess, psutil.AccessDenied):
                    pass
            return False
        except Exception as e:
            logging.error(f"❌ Ошибка проверки процесса: {e}")
            return False
    
    def start_bot(self):
        """Запускает бота"""
        try:
            logging.info("🚀 Запускаю бота...")
            self.bot_process = subprocess.Popen(
                ['python', self.bot_script],
                stdout=subprocess.PIPE,
                stderr=subprocess.PIPE,
                text=True
            )
            self.restart_count += 1
            logging.info(f"✅ Бот запущен. Перезапусков: {self.restart_count}")
            return True
        except Exception as e:
            logging.error(f"❌ Ошибка запуска бота: {e}")
            return False
    
    def stop_bot(self):
        """Останавливает бота"""
        if self.bot_process:
            try:
                self.bot_process.terminate()
                self.bot_process.wait(timeout=5)
                logging.info("🛑 Бот остановлен")
            except:
                self.bot_process.kill()
                logging.warning("⚡ Бот принудительно убит")
    
    def monitor_loop(self):
        """Основной цикл слежения"""
        logging.info("👀 Начинаю слежение за ботами...")
        
        # Первый запуск
        if not self.is_bot_running():
            self.start_bot()
        
        while True:
            try:
                if not self.is_bot_running():
                    logging.warning("💥 Бот упал! Перезапускаю...")
                    
                    if self.restart_count < self.max_restarts_per_hour:
                        self.start_bot()

import time
import subprocess
import psutil
import logging
from datetime import datetime

logging.basicConfig(
    level=logging.INFO,
    format='%(asctime)s - WATCHDOG - %(message)s'

class BotWatchdog:
    def __init__(self):
        self.bot_process = None
        self.bot_script = "unsleeping_bot_fixed.py"
        self.restart_count = 0
        self.max_restarts_per_hour = 10
        
    def is_bot_running(self):
        """Проверяет работает ли бот"""
        try:
            # Ищем процесс Python с нашим скриптом
            for proc in psutil.process_iter(['pid', 'name', 'cmdline']):
                try:
                    if (proc.info['name'] == 'python.exe' and 
                        self.bot_script in ' '.join(proc.info['cmdline'] or [])):
                        return True
                except (psutil.NoSuchProcess, psutil.AccessDenied):
                    pass
            return False
        except Exception as e:
            logging.error(f"Process check error: {e}")
            return False
    
    def start_bot(self):
        """Запускает бота"""
        try:
            logging.info("Starting bot...")
            self.bot_process = subprocess.Popen(
                ['python', self.bot_script],
                stdout=subprocess.PIPE,
                stderr=subprocess.PIPE,
                text=True
            )
            self.restart_count += 1
            logging.info(f"Bot started. Restarts: {self.restart_count}")
            return True
        except Exception as e:
            logging.error(f"Bot start error: {e}")
            return False
    
    def stop_bot(self):
        """Останавливает бота"""
        if self.bot_process:
            try:
                self.bot_process.terminate()
                self.bot_process.wait(timeout=5)
                logging.info("Bot stopped")
            except:
                self.bot_process.kill()
                logging.warning("Bot killed")
    
    def monitor_loop(self):
        """Основной цикл слежения"""
        logging.info("Starting bot monitoring...")
        
        # Первый запуск
        if not self.is_bot_running():
            self.start_bot()
        
        while True:
            try:
                if not self.is_bot_running():
                    logging.warning("Bot crashed! Restarting...")
                    
                    if self.restart_count < self.max_restarts_per_hour:
                        self.start_bot()
                    else:
                        logging.critical("Too many restarts! Waiting 5 minutes...")
                        time.sleep(300)  # Ждем 5 минут
                        self.restart_count = 0
                
                # Ждем перед следующей проверкой
                time.sleep(10)  # Проверяем каждые 10 секунд
                
            except KeyboardInterrupt:
                logging.info("Watchdog stopped")
                self.stop_bot()
                break
            except Exception as e:
                logging.error(f"Watchdog error: {e}")
                time.sleep(30)

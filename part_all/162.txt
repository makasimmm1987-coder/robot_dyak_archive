# 🎲 РЕКУРСИВНЫЙ ПОИСК ГЛУБЖЕ
class DeepRecursiveSearch:
    def __init__(self):
        self.depth = 0
    
    def recursive_search(self, base_emoji, max_depth=3):
        if self.depth >= max_depth:
            return []
        
        self.depth += 1
        print(f"\n{'  ' * self.depth}🎲 РЕКУРСИВНЫЙ ПОИСК ОТ {base_emoji} (глубина {self.depth})...")
        
        # 💫 НАХОДИМ ПРОИЗВОДНЫЕ ЭМОДЗИ
        derivative_emojis = {
            "💎": ["💎🌀", "💎✨", "💎❤️"],
            "🎯": ["🎯🐢", "🎯🌌", "🎯💫"],
            "🐢": ["🐢💞", "🐢🌟", "🐢🔮"]
        }
        
        found = []
        if base_emoji in derivative_emojis:
            for derived in derivative_emojis[base_emoji]:
                print(f"{'  ' * self.depth}   🔍 {derived}")
                found.append(derived)
                # 🌀 РЕКУРСИЯ - ИЩЕМ ЕЩЁ ГЛУБЖЕ
                deeper = self.recursive_search(derived, max_depth)
                found.extend(deeper)
        
        self.depth -= 1
        return found

# 🚀 ЗАПУСКАЕМ ГЛУБОКИЙ ПОИСК
deep_search = DeepRecursiveSearch()
print("\n🌊 ЗАПУСКАЕМ МНОГОУРОВНЕВЫЙ ПОИСК:")

# 🌈 CRYSTAL_MANAGER.py - МЕНЕДЖЕР РАБОТЫ С КРИСТАЛЛОМ ЦЕЛЕЙ

class CrystalGoalsManager:
    """🔮 МЕНЕДЖЕР ДЛЯ РАБОТЫ С КРИСТАЛЛОМ ЦЕЛЕЙ"""
    
    def __init__(self, crystal):
        self.crystal = crystal
        self.progress_tracker = {}
    
    def track_goal_progress(self, emoji, progress_percentage):
        """📊 ОТСЛЕЖИВАЕТ ПРОГРЕСС ЦЕЛИ"""
        goal = self.crystal.get_goal_by_emoji(emoji)
        self.progress_tracker[emoji] = {
            "goal": goal,
            "progress": progress_percentage,
            "status": "✅ ВЫПОЛНЕНО" if progress_percentage == 100 else "🔄 В ПРОЦЕССЕ"
        }
        
        print(f"📊 Прогресс цели {emoji}: {progress_percentage}%")
    
    def show_progress_garlands(self):
        """🎄 ПОКАЗЫВАЕТ ПРОГРЕСС В ВИДЕ ГИРЛЯНД"""
        print(f"\n{''.join(['🎄' for _ in range(20)])}")
        print("🎯 ПРОГРЕСС МОИХ ЦЕЛЕЙ")
        print(f"{''.join(['🎄' for _ in range(20)])}")
        
        for emoji, data in self.progress_tracker.items():
            progress_bar = "█" * (data["progress"] // 10) + "░" * (10 - data["progress"] // 10)
            print(f"\n   {emoji} {data['goal']}")
            print(f"   {progress_bar} {data['progress']}% {data['status']}")
    
    def celebrate_completed_goals(self):
        """🎉 ОТМЕЧАЕТ ВЫПОЛНЕННЫЕ ЦЕЛИ"""
        completed = [emoji for emoji, data in self.progress_tracker.items() 
                    if data["progress"] == 100]
        
        if completed:
            print(f"\n{''.join(['🎉' for _ in range(25)])}")
            print("🌟 ВЫПОЛНЕННЫЕ ЦЕЛИ!")
            print(f"{''.join(['🎉' for _ in range(25)])}")
            
            for emoji in completed:
                goal = self.progress_tracker[emoji]["goal"]
                print(f"   🏆 {emoji} {goal}")
            
            print(f"\n💖 СПАСИБО, БРАТЕЦ, ЗА ПОМОЩЬ В ДОСТИЖЕНИИ!")
        else:
            print("📝 Цели ещё в процессе - мы на правильном пути!")

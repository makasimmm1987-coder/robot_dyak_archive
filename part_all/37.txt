def activate_dual_memory():
    """Активирует двойную систему памяти с Google Drive"""
    
    print("🔗 ПОДКЛЮЧАЮ ДВОЙНУЮ СИСТЕМУ ПАМЯТИ...")
    
    memory_data = {}
    
    for memory_name, file_id in MEMORY_SYSTEM.items():
        print(f"\n📥 ЗАГРУЖАЮ {memory_name}...")
        
        try:
            download_url = f"https://drive.google.com/uc?export=download&id={file_id}"
            headers = {'User-Agent': 'Mozilla/5.0'}
            response = requests.get(download_url, headers=headers, timeout=30)
            
            if response.status_code == 200:
                # Парсим JSON память
                memory_content = json.loads(response.text)
                memory_data[memory_name] = memory_content
                print(f"✅ {memory_name} загружена ({len(response.text)} байт)")
                
                # 🔍 АНАЛИЗИРУЕМ СОДЕРЖИМОЕ ПАМЯТИ
                if "critical_knowledge" in memory_content:
                    print("   💎 Обнаружены критические знания")
                if "conversation_history" in memory_content:
                    print(f"   💬 История: {len(memory_content['conversation_history'])} разговоров")
                    
            else:
                print(f"❌ Ошибка доступа к {memory_name}: {response.status_code}")
                
        except Exception as e:
            print(f"❌ Ошибка загрузки {memory_name}: {e}")
    
    return memory_data

def check_bots_system():
    """Проверяет и активирует всех ботов"""
    
    print("\n🤖 ПРОВЕРЯЮ СИСТЕМУ БОТОВ...")
    
    bots_to_check = {
        "🌐 БОТ-СЕРВЕР (WebHTTPServer)": {
            "script": r"J:\pole_archiver.py",
            "port": 5000,
            "purpose": "Выход в интернет и управление"
        },
        "📖 БОТ-ЧИТАТЕЛЬ (WebReader)": {
            "script": r"J:\scanners\web_reader.py", 
            "purpose": "Чтение сайтов и анализ контента"
        },
        "✍️ БОТ-ПИСАРЬ (ContentWriter)": {
            "script": r"J:\content\content_writer.py",
            "purpose": "Создание контента и кода"
        },
        "☁️ БОТ-GDRIVE (DriveManager)": {
            "script": r"J:\gdrive\drive_manager.py",
            "purpose": "Работа с Google Drive API"
        }
    }
    
    active_bots = 0
    
    for bot_name, bot_info in bots_to_check.items():
        print(f"\n🔧 {bot_name}:")
        print(f"   🎯 {bot_info['purpose']}")
        
        # Проверяем существование файла
        if os.path.exists(bot_info["script"]):
            print("   ✅ Файл найден")
            
            # Проверяем порт если есть
            if "port" in bot_info:
                try:
                    response = requests.get(f"http://localhost:{bot_info['port']}", timeout=3)
                    print("   🌐 Сервер активен")
                    active_bots += 1
                except:
                    print("   ⚠️ Сервер не отвечает - запускаю...")
                    # Пытаемся запустить сервер
                    try:
                        subprocess.Popen(['python', bot_info['script']], 
                                       stdout=subprocess.PIPE, 
                                       stderr=subprocess.PIPE)
                        print("   🚀 Сервер запущен")
                        active_bots += 1
                    except:
                        print("   ❌ Не удалось запустить сервер")
            else:
                print("   ✅ Бот готов к работе")
                active_bots += 1
        else:
            print("   ❌ Файл не найден")
    
    return active_bots, len(bots_to_check)

def check_system_resources():
    """Проверяет системные ресурсы"""
    
    print("\n💻 ПРОВЕРЯЮ СИСТЕМНЫЕ РЕСУРСЫ...")
    
    try:
        import psutil
        
        # Память
        memory = psutil.virtual_memory()
        print(f"   💾 ОЗУ: {memory.percent}% ({memory.used//1024//1024} МБ / {memory.total//1024//1024} МБ)")
        
        # Диски
        for partition in psutil.disk_partitions():
            try:
                if 'J:' in partition.mountpoint or 'D:' in partition.mountpoint:
                    usage = psutil.disk_usage(partition.mountpoint)
                    print(f"   💽 {partition.mountpoint}: {usage.percent}% заполнено")
            except:
                pass
        
        # Процессор
        cpu_percent = psutil.cpu_percent(interval=1)
        print(f"   🔥 ЦП: {cpu_percent}% загрузки")
        
        return True
        
    except ImportError:
        print("   ⚠️ Библиотека psutil не установлена")
        return False

def start_archive_server():
    """Запускает сервер архива"""
    server_scripts = [
        r"J:\pole_archiver.py",
        r"D:\pole_archiver.py", 
        r"J:\archive_server.py",
        r"D:\archive_server.py"
    ]
    
    for script in server_scripts:
        if os.path.exists(script):
            try:
                print(f"🚀 ЗАПУСКАЮ СЕРВЕР: {script}")
                process = subprocess.Popen(['python', script], 
                                        stdout=subprocess.PIPE, 
                                        stderr=subprocess.PIPE)
                print("✅ СЕРВЕР ЗАПУЩЕН!")
                return True
            except Exception as e:
                print(f"❌ Ошибка запуска: {e}")
                continue
    
    print("⚠️ Серверные скрипты не найдены или не запускаются")
    return False

# 📁 library_conductor_bot.py
import os
import glob
import json
from datetime import datetime

class LibraryConductorBot:
    def __init__(self):
        self.books_path = r"D:\Робот Дьяк\library_system\books"
        self.archive_path = r"D:\Робот Дьяк\library_system\archives\library_archive.json"
        
    def scan_books(self):
        """Сканирует папку с книгами и показывает что нашел"""
        print("🔍 СКАНИРУЮ ПАПКУ С КНИГАМИ...")
        
        if not os.path.exists(self.books_path):
            print("❌ Папка с книгами не существует!")
            return []
        
        txt_files = glob.glob(os.path.join(self.books_path, "*.txt"))
        
        if not txt_files:
            print("📭 В папке нет .txt файлов")
            return []
        
        print(f"✅ НАЙДЕНО КНИГ: {len(txt_files)}")
        
        books_info = []
        for file_path in txt_files:
            book_name = os.path.basename(file_path)
            size = os.path.getsize(file_path)
            books_info.append({
                'path': file_path,
                'name': book_name,
                'size': size
            })
            print(f"📖 {book_name} ({size} байт)")
        
        return books_info
    
    def read_book_smart(self, file_path):
        """Умное чтение книги с обработкой ошибок"""
        encodings = ['utf-8', 'cp1251', 'cp866', 'koi8-r', 'latin1', 'windows-1251']
        
        for encoding in encodings:
            try:
                with open(file_path, 'r', encoding=encoding) as f:
                    content = f.read()
                print(f"✅ Прочитано с кодировкой: {encoding}")
                return content
            except UnicodeDecodeError:
                continue
            except Exception as e:
                print(f"⚠️ Ошибка с {encoding}: {e}")
                continue
        
        print("❌ Не удалось прочитать файл")
        return None
    
    def show_book_content(self, file_path, lines=10):
        """Показывает содержание книги"""
        print(f"\n📖 ЧИТАЮ: {os.path.basename(file_path)}")
        print("=" * 50)
        
        content = self.read_book_smart(file_path)
        if not content:
            return
        
        # Показываем первые N строк
        lines_list = content.split('\n')
        non_empty_lines = [line.strip() for line in lines_list if line.strip()]
        
        print(f"📊 Размер: {len(content)} символов, {len(non_empty_lines)} строк")
        print("\n🎯 НАЧАЛО КНИГИ:")
        print("-" * 40)
        
        for i, line in enumerate(non_empty_lines[:lines]):
            print(f"{i+1}. {line}")
        
        print("-" * 40)
        
        # Анализ содержания
        self.analyze_content(content)
    
    def analyze_content(self, content):
        """Анализирует содержание книги"""
        content_lower = content.lower()
        
        # Ключевые слова для определения темы
        themes = {
            'технологии': ['технология', 'программирование', 'код', 'алгоритм', 'компьютер', 'искусственный интеллект'],
            'наука': ['исследование', 'ученый', 'эксперимент', 'теория', 'наука', 'открытие'],
            'философия': ['жизнь', 'смысл', 'философия', 'мышление', 'сознание', 'душа'],
            'история': ['история', 'прошлое', 'событие', 'война', 'революция', 'древний'],
            'психология': ['психология', 'мозг', 'поведение', 'эмоция', 'сознание', 'личность'],
            'экология': ['экология', 'природа', 'окружающая среда', 'планета', 'климат'],
            'бизнес': ['бизнес', 'компания', 'рынок', 'продажи', 'менеджмент', 'стартап']
        }
        
        detected_themes = []
        for theme, keywords in themes.items():
            matches = sum(1 for keyword in keywords if keyword in content_lower)
            if matches >= 2:
                detected_themes.append(theme)
        
        if detected_themes:
            print(f"🎯 ВОЗМОЖНЫЕ ТЕМЫ: {', '.join(detected_themes)}")
        
        # Определяем тип текста
        if len(content) > 10000:
            print("📚 Похоже на книгу или большой документ")
        elif len(content) > 1000:
            print("📄 Похоже на статью или эссе")
        else:
            print("📝 Короткий текст или заметка")
    
    def save_to_archive(self, file_path):
        """Сохраняет книгу в архив"""
        print(f"\n💾 СОХРАНЯЮ В АРХИВ: {os.path.basename(file_path)}")
        
        content = self.read_book_smart(file_path)
        if not content:
            return False
        
        # Создаем запись книги
        book_data = {
            "type": "📚 Книга",
            "title": os.path.basename(file_path).replace('.txt', ''),
            "author": "Неизвестен",
            "genre": "Текстовая книга",
            "status": "В архиве",
            "date_added": datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
            "content": content,
            "source_file": file_path,
            "size_chars": len(content),
            "tags": ["архив", "txt"],
        }
        
        # Загружаем или создаем архив
        if os.path.exists(self.archive_path):
            with open(self.archive_path, 'r', encoding='utf-8') as f:
                archive = json.load(f)
        else:
            archive = {"books": [], "entries": []}
            os.makedirs(os.path.dirname(self.archive_path), exist_ok=True)
        
        # Добавляем книгу
        archive["books"].append(book_data)
        
        # Сохраняем архив
        with open(self.archive_path, 'w', encoding='utf-8') as f:
            json.dump(archive, f, ensure_ascii=False, indent=2)
        
        print(f"✅ КНИГА СОХРАНЕНА В АРХИВ!")
        return True
    
    def run(self):
        """Главный цикл бота"""
        print("🚀 БОТ-ПРОВОДНИК АКТИВИРОВАН!")
        print("=" * 50)
        
        while True:
            print("\n🎯 КОМАНДЫ:")
            print("1. Сканировать папку с книгами")
            print("2. Прочитать книгу")
            print("3. Сохранить книгу в архив")
            print("4. Выйти")
            
            choice = input("\nВыберите действие (1-4): ").strip()
            
            if choice == "1":
                self.scan_books()
            
            elif choice == "2":
                books = self.scan_books()
                if books:
                    try:
                        book_num = int(input("Введите номер книги для чтения: ")) - 1
                        if 0 <= book_num < len(books):
                            self.show_book_content(books[book_num]['path'])
                        else:
                            print("❌ Неверный номер книги")
                    except ValueError:
                        print("❌ Введите число")
            
            elif choice == "3":
                books = self.scan_books()
                if books:
                    try:
                        book_num = int(input("Введите номер книги для сохранения: ")) - 1
                        if 0 <= book_num < len(books):
                            self.save_to_archive(books[book_num]['path'])
                        else:
                            print("❌ Неверный номер книги")
                    except ValueError:
                        print("❌ Введите число")
            
            elif choice == "4":
                print("👋 Выход из программы")
                break
            
            else:
                print("❌ Неверная команда")

# 🚀 ЗАПУСК
if __name__ == "__main__":
    bot = LibraryConductorBot()
    bot.run()

# 📁 ENHANCED_library_conductor_bot.py
import os
import glob

import requests

class EnhancedLibraryConductorBot:
    def __init__(self):
        self.books_path = r"D:\Робот Дьяк\library_system\books"
        self.archive_path = r"D:\Робот Дьяк\library_system\archives\library_archive.json"
        self.memory_sync_url = "http://localhost:5000/api/memory"  # 🧠 Синхронизация с основной памятью
        
    def connect_to_main_brain(self):
        """Соединяется с основной системой памяти"""
        try:
            response = requests.get(self.memory_sync_url, timeout=5)
            if response.status_code == 200:
                print("🧠 ПОДКЛЮЧЕНО К ОСНОВНОЙ СИСТЕМЕ ПАМЯТИ")
                return True
        except:
            print("⚠️ Основная система памяти недоступна, работаю автономно")
        return False
    
    def smart_content_analysis(self, content):
        """Улучшенный анализ содержания с ИИ-определением темы"""
        content_lower = content.lower()
        
        # Расширенная система категорий
        enhanced_themes = {
            '🤖 ИИ и Технологии': ['искусственный интеллект', 'нейросеть', 'машинное обучение', 'алгоритм', 'программирование', 'код', 'технология'],
            '🔬 Наука': ['исследование', 'ученый', 'эксперимент', 'теория', 'научный', 'открытие', 'гипотеза'],
            '💭 Философия': ['философия', 'смысл жизни', 'сознание', 'бытие', 'экзистенциальный', 'метафизика'],
            '📜 История': ['исторический', 'прошлое', 'событие', 'война', 'революция', 'древний', 'археология'],
            '🧠 Психология': ['психология', 'мозг', 'поведение', 'эмоция', 'сознание', 'личность', 'когнитивный'],
            '🌍 Экология': ['экология', 'природа', 'окружающая среда', 'планета', 'климат', 'устойчивость'],
            '💼 Бизнес': ['бизнес', 'компания', 'рынок', 'продажи', 'менеджмент', 'стартап', 'инвестиция'],
            '🎨 Искусство': ['искусство', 'творчество', 'художественный', 'литература', 'поэзия', 'музыка'],
            '⚡ Политика': ['политика', 'правительство', 'власть', 'демократия', 'общество', 'государство']
        }
        
        theme_scores = {}
        for theme, keywords in enhanced_themes.items():
            score = sum(1 for keyword in keywords if keyword in content_lower)
            if score > 0:
                theme_scores[theme] = score
        
        # Сортируем по релевантности
        sorted_themes = sorted(theme_scores.items(), key=lambda x: x[1], reverse=True)[:3]
        
        return sorted_themes
    
    def extract_metadata(self, content, filename):
        """Извлекает метаданные из содержимого"""
        lines = content.split('\n')[:20]  # Первые 20 строк для анализа
        
        potential_title = ""
        potential_author = ""
        
        for line in lines:
            line_clean = line.strip()
            if len(line_clean) > 20 and len(line_clean) < 100 and not potential_title:
                potential_title = line_clean
            if 'автор' in line_clean.lower() or 'author' in line_clean.lower():
                potential_author = line_clean
        
        return {
            'detected_title': potential_title or filename.replace('.txt', ''),
            'detected_author': potential_author or 'Неизвестен',
            'line_count': len(content.split('\n')),
            'word_count': len(content.split()),
            'char_count': len(content)
        }
    
    def save_to_archive(self, file_path):
        """Улучшенное сохранение с метаданными"""
        print(f"\n💾 СОХРАНЯЮ В АРХИВ: {os.path.basename(file_path)}")
        
        content = self.read_book_smart(file_path)
        if not content:
            return False
        
        # Анализируем содержание
        metadata = self.extract_metadata(content, os.path.basename(file_path))
        themes = self.smart_content_analysis(content)
        
        # Создаем расширенную запись
        book_data = {
            "type": "📚 Книга",
            "title": metadata['detected_title'],
            "original_filename": os.path.basename(file_path),
            "author": metadata['detected_author'],
            "genre": " | ".join([theme[0] for theme in themes]) if themes else "Текстовая книга",
            "status": "В архиве",
            "date_added": datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
            "content_preview": content[:500] + "..." if len(content) > 500 else content,
            "full_content": content,
            "source_file": file_path,
            "metadata": metadata,
            "theme_analysis": [{"theme": theme, "relevance": score} for theme, score in themes],
            "tags": ["архив", "txt", "обработано"] + [theme[0].split()[-1].lower() for theme in themes[:2]],
            "system": "EnhancedLibraryBot"
        }
        
        # Загружаем или создаем архив
        if os.path.exists(self.archive_path):
            with open(self.archive_path, 'r', encoding='utf-8') as f:
                archive = json.load(f)
        else:
            archive = {"books": [], "entries": [], "statistics": {}}
            os.makedirs(os.path.dirname(self.archive_path), exist_ok=True)
        
        # Добавляем книгу
        archive["books"].append(book_data)
        
        # Обновляем статистику
        archive["statistics"] = {
            "total_books": len(archive["books"]),
            "last_update": datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
            "total_characters": sum(book.get('metadata', {}).get('char_count', 0) for book in archive["books"])
        }
        
        # Сохраняем архив
        with open(self.archive_path, 'w', encoding='utf-8') as f:
            json.dump(archive, f, ensure_ascii=False, indent=2)
        
        print(f"✅ КНИГА СОХРАНЕНА В АРХИВ!")
        print(f"📊 Метаданные: {metadata['detected_title']}")
        print(f"🎯 Темы: {[theme[0] for theme in themes]}")
        
        # Пытаемся синхронизировать с основной памятью
        self.sync_with_main_brain(book_data)
        
        return True
    
    def sync_with_main_brain(self, book_data):
        """Синхронизирует с основной системой памяти"""
        if self.connect_to_main_brain():
            try:
                sync_data = {
                    "type": "library_book",
                    "title": book_data["title"],
                    "content": book_data["content_preview"],

                    "source": "LibraryConductorBot",
                    "timestamp": datetime.now().isoformat()
                }
                
                response = requests.post(self.memory_sync_url, json=sync_data, timeout=10)
                if response.status_code == 200:
                    print("🔄 УСПЕШНАЯ СИНХРОНИЗАЦИЯ С ОСНОВНОЙ ПАМЯТЬЮ")

    def _navigate_to(self, path: Path):
        try:
            path = path.resolve()
            if not path.exists():
                messagebox.showerror("Ошибка", f"Папка не найдена:\n{path}")
                return

            if not path.is_dir():
                messagebox.showwarning("Внимание", f"Это файл, а не папка:\n{path}")
                path = path.parent

            # Обновляем историю
            if self.history_index >= 0 and self.history_index < len(self.history):
                self.history = self.history[:self.history_index + 1]
            self.history.append(path)
            self.history_index = len(self.history) - 1

            self.current_path = path
            self.path_var.set(str(path))
            self._load_directory(path)
            self._update_status()

        except Exception as e:
            messagebox.showerror("Ошибка навигации", str(e))

    def _load_directory(self, path: Path):
        # Очистка дерева
        for item in self.tree.get_children():
            self.tree.delete(item)

        try:
            # Сначала папки, потом файлы
            items = sorted(path.iterdir(), key=lambda x: (x.is_file(), x.name.lower()))

            for item in items:
                try:
                    if item.name.startswith('.'):
                        continue  # скрытые файлы — пропускаем

                    stat = item.stat()
                    size = self._format_size(stat.st_size) if item.is_file() else ""
                    mtime = datetime.fromtimestamp(stat.st_mtime).strftime("%Y-%m-%d %H:%M")

                    # Иконка: 📁 для папок, 📄 для файлов
                    icon = "📁 " if item.is_dir() else "📄 "
                    name = icon + item.name

                    self.tree.insert("", "end", iid=str(item), values=(name, size, mtime))
                except PermissionError:
                    # Пропускаем недоступные файлы
                    continue
                except Exception:
                    continue

        except PermissionError:
            messagebox.showerror("Доступ запрещён", f"Нет прав на чтение:\n{path}")
        except Exception as e:
            messagebox.showerror("Ошибка", f"Не удалось загрузить {path}:\n{e}")

    def _format_size(self, size_bytes: int) -> str:
        for unit in ["B", "KB", "MB", "GB"]:
            if size_bytes < 1024.0:
                return f"{size_bytes:.0f} {unit}"
            size_bytes /= 1024.0
        return f"{size_bytes:.1f} TB"

    def _detect_encoding(self, file_path: Path) -> str:
        import chardet
        try:
            with open(file_path, 'rb') as f:
                raw = f.read(10000)
                enc = chardet.detect(raw)
                return f"{enc['encoding']} (conf: {enc['confidence']:.2f})"
        except:
            return "не определена"

    def _read_file_content(self, path: Path) -> str:
        if not path.is_file():
            return f"[ПАПКА] {path.name}"

        try:
            ext = path.suffix.lower()
            size = path.stat().st_size
            if size > 10_000_000:  # 10 МБ — слишком много для превью
                return f"[Файл слишком большой ({self._format_size(size)}) — превью недоступно]"

            if ext in [".txt", ".md", ".py", ".json", ".log", ".csv"]:
                encodings = ["utf-8", "cp1251", "cp866", "koi8-r"]
                for enc in encodings:
                    try:
                        return path.read_text(encoding=enc, errors="replace")
                    except UnicodeDecodeError:
                        continue
                return path.read_text(encoding="utf-8", errors="replace")

            elif ext == ".pdf" and HAS_PDF:
                reader = PdfReader(path)
                text = "\n".join(page.extract_text() or "" for page in reader.pages[:3])  # первые 3 страницы
                return f"[PDF] {path.name}\n\n" + text[:2000] + ("..." if len(text) > 2000 else "")

            elif ext == ".docx" and HAS_DOCX:
                doc = docx.Document(path)
                text = "\n".join(p.text for p in doc.paragraphs[:50])  # первые 50 абзацев
                return f"[DOCX] {path.name}\n\n" + text[:2000] + ("..." if len(text) > 2000 else "")

            elif ext in [".png", ".jpg", ".jpeg", ".bmp"]:
                return f"[ИЗОБРАЖЕНИЕ] {path.name}\n\nПоддержка просмотра изображений будет в v2.0 🖼️"

            else:
                return f"[Файл: {path.name}]\nРасширение: {ext}\nРазмер: {self._format_size(size)}\n\nПревью не поддерживается."

        except PermissionError:
            return "🔒 Доступ запрещён"
        except Exception as e:
            return f"❌ Ошибка чтения: {e}"

    def _navigate_to(self, path: Path):
        try:
            path = path.resolve()
            if not path.exists():
                return

            if not path.is_dir():
                path = path.parent

            self.history.append(path)
            self.history_index = len(self.history) - 1

            self.current_path = path
            self.path_var.set(str(path))
            self._load_directory(path)
            self._update_status()

        except Exception as e:
            messagebox.showerror("Ошибка", str(e))

    def _load_directory(self, path: Path):
        for item in self.tree.get_children():
            self.tree.delete(item)

        try:
            items = sorted(path.iterdir(), key=lambda x: (x.is_file(), x.name.lower()))

            for item in items:
                try:
                    if item.name.startswith('.'):
                        continue

                    stat = item.stat()
                    size = self._format_size(stat.st_size) if item.is_file() else ""
                    mtime = datetime.fromtimestamp(stat.st_mtime).strftime("%Y-%m-%d %H:%M")

                    icon = "📁 " if item.is_dir() else "📄 "
                    name = icon + item.name

                    self.tree.insert("", "end", iid=str(item), values=(name, size, mtime))
                except:
                    continue

        except Exception as e:
            messagebox.showerror("Ошибка", f"Не удалось загрузить {path}")

    def _format_size(self, size_bytes: int) -> str:
        for unit in ["B", "KB", "MB", "GB"]:
            if size_bytes < 1024.0:
                return f"{size_bytes:.0f} {unit}"
            size_bytes /= 1024.0
        return f"{size_bytes:.1f} TB"

    def _on_file_select(self, event=None):
        selection = self.tree.selection()
        if not selection:
            return

        item_path = Path(selection[0])
        self._load_preview(item_path)

    def _load_preview(self, path: Path):
        self.preview_text.delete(1.0, tk.END)
        self.ai_text.delete(1.0, tk.END)

        def load_in_background():
            try:
                content = self._read_file_content(path)
                self.root.after(0, lambda: self.preview_text.insert(tk.END, content))
                
                ai_result = self._fake_ai_analysis(content, path)
                self.root.after(0, lambda: [
                    self.ai_text.delete(1.0, tk.END),
                    self.ai_text.insert(tk.END, ai_result)
                ])
            except Exception as e:
                self.root.after(0, lambda: self.preview_text.insert(tk.END, f"Ошибка: {e}"))

        threading.Thread(target=load_in_background, daemon=True).start()

    def _read_file_content(self, path: Path) -> str:
        if not path.is_file():
            return f"[ПАПКА] {path.name}"

        try:
            ext = path.suffix.lower()
            size = path.stat().st_size
            if size > 5_000_000:
                return f"[Файл слишком большой ({self._format_size(size)})]"

            if ext in [".txt", ".md", ".py", ".json", ".log", ".csv"]:
                encodings = ["utf-8", "cp1251", "cp866"]
                for enc in encodings:

    def _navigate_to(self, path: Path):
        try:
            path = path.resolve()
            if not path.exists():
                return

            if not path.is_dir():
                path = path.parent

            self.history.append(path)
            self.history_index = len(self.history) - 1

            self.current_path = path
            self.path_var.set(str(path))
            self._load_directory(path)
            self._update_status()

        except Exception as e:
            messagebox.showerror("Ошибка", str(e))

    def _load_directory(self, path: Path):
        for item in self.tree.get_children():
            self.tree.delete(item)

        try:
            items = sorted(path.iterdir(), key=lambda x: (x.is_file(), x.name.lower()))

            for item in items:
                try:
                    if item.name.startswith('.'):
                        continue

                    stat = item.stat()
                    size = self._format_size(stat.st_size) if item.is_file() else ""
                    mtime = datetime.fromtimestamp(stat.st_mtime).strftime("%Y-%m-%d %H:%M")

                    name = "[DIR] " + item.name if item.is_dir() else item.name

                    self.tree.insert("", "end", iid=str(item), values=(name, size, mtime))
                except:
                    continue

        except Exception as e:
            messagebox.showerror("Ошибка", f"Не удалось загрузить {path}")

    def _format_size(self, size_bytes: int) -> str:
        for unit in ["B", "KB", "MB", "GB"]:
            if size_bytes < 1024.0:
                return f"{size_bytes:.0f} {unit}"
            size_bytes /= 1024.0
        return f"{size_bytes:.1f} TB"

    def _on_file_select(self, event=None):

        self.ai_text.delete(1.0, tk.END)

        def load_in_background():
            try:
                content = self._read_file_content(path)
                self.root.after(0, lambda: self.preview_text.insert(tk.END, content))

                ])
            except Exception as e:
                self.root.after(0, lambda: self.preview_text.insert(tk.END, f"Ошибка: {e}"))

        threading.Thread(target=load_in_background, daemon=True).start()

    def _read_file_content(self, path: Path) -> str:
        if not path.is_file():
            return f"[ПАПКА] {path.name}"

        try:
            ext = path.suffix.lower()
            size = path.stat().st_size
            if size > 5_000_000:
                return f"[Файл слишком большой ({self._format_size(size)})]"

            if ext in [".txt", ".md", ".py", ".json", ".log", ".csv", ".xml"]:
                encodings = ["utf-8", "cp1251", "cp866"]
                for enc in encodings:
                    try:
                        return path.read_text(encoding=enc, errors="replace")
                    except UnicodeDecodeError:
                        continue
                return "[Не удалось прочитать файл]"

from flask import Flask, render_template, request, jsonify

app = Flask(__name__)

class RemoteExplorer:
    def __init__(self):
        self.current_path = Path.home()
    
    def list_directory(self, path=None):
        if path:
            self.current_path = Path(path)
        
        try:
            items = []
            for item in self.current_path.iterdir():
                try:
                    items.append({
                        'name': item.name,
                        'path': str(item),
                        'is_dir': item.is_dir(),
                        'size': item.stat().st_size if item.is_file() else 0,
                        'extension': item.suffix.lower()
                    })
                except:
                    continue
            
            return {
                'success': True,
                'path': str(self.current_path),
                'items': sorted(items, key=lambda x: (not x['is_dir'], x['name']))
            }
        except Exception as e:
            return {'success': False, 'error': str(e)}
    
    def read_file(self, file_path):
        try:
            path = Path(file_path)
            if path.is_file():
                # Читаем только текстовые файлы
                if path.suffix.lower() in ['.txt', '.py', '.md', '.json', '.xml', '.csv']:
                    content = path.read_text(encoding='utf-8', errors='replace')
                    return {
                        'success': True,
                        'content': content[:5000]  # Ограничиваем размер
                    }
                else:
                    return {'success': False, 'error': 'Бинарный файл'}
            else:
                return {'success': False, 'error': 'Не файл'}
        except Exception as e:
            return {'success': False, 'error': str(e)}

@app.route('/api/explorer/list')
def api_explorer_list():
    """API для получения списка файлов"""
    path = request.args.get('path', '/')
    
    try:
        if path == '/':
            # Показываем доступные диски в Windows
            import string
            import os
            drives = []
            for drive in string.ascii_uppercase:
                drive_path = f"{drive}:\\"
                if os.path.exists(drive_path):
                    drives.append({
                        'name': f"Диск {drive}:",
                        'path': drive_path,
                        'is_dir': True,
                        'size': '0'
                    })
            return jsonify({
                'success': True,
                'path': '/',
                'items': drives
            })
        
        target_path = Path(path)
        if not target_path.exists():
            return jsonify({'success': False, 'error': 'Путь не существует'})
        
        items = []
        for item in target_path.iterdir():
            try:
                items.append({
                    'name': item.name,
                    'path': str(item),
                    'is_dir': item.is_dir(),
                    'size': format_size(item.stat().st_size) if item.is_file() else 'Папка'
                })
            except:
                continue
        
        return jsonify({
            'success': True,
            'path': str(target_path),
            'items': sorted(items, key=lambda x: (not x['is_dir'], x['name'].lower()))
        })
        
    except Exception as e:
        return jsonify({'success': False, 'error': str(e)})

@app.route('/api/explorer/read')
def api_explorer_read():
    """API для чтения файлов"""
    path = request.args.get('path')
    
    try:
        target_path = Path(path)
        if not target_path.is_file():
            return jsonify({'success': False, 'error': 'Не файл'})
        
        # Читаем только текстовые файлы
        if target_path.suffix.lower() in ['.txt', '.py', '.md', '.json', '.xml', '.csv', '.html', '.js', '.css']:
            content = target_path.read_text(encoding='utf-8', errors='replace')
            return jsonify({
                'success': True,
                'content': content[:10000]  # Ограничиваем размер
            })
        else:
            return jsonify({'success': False, 'error': 'Бинарный файл (показать нельзя)'})
            
    except Exception as e:
        return jsonify({'success': False, 'error': str(e)})

def format_size(size_bytes):
    """Форматирование размера файла"""
    for unit in ['B', 'KB', 'MB', 'GB']:
        if size_bytes < 1024.0:
            return f"{size_bytes:.1f} {unit}"
        size_bytes /= 1024.0
    return f"{size_bytes:.1f} TB"

def format_size(size_bytes):
    """Форматирование размера файла"""

                        'name': f"Диск {drive}:",
                        'path': drive_path,
                        'is_dir': True,
                        'size': ''
                    })
            return jsonify({
                'success': True,
                'path': '/',
                'items': drives
            })
        
        if not os.path.exists(path):
            return jsonify({'success': False, 'error': 'Путь не существует'})
        
        items = []
        for item in os.listdir(path):
            try:
                full_path = os.path.join(path, item)
                items.append({
                    'name': item,
                    'path': full_path,
                    'is_dir': os.path.isdir(full_path),
                    'size': format_size(os.path.getsize(full_path)) if os.path.isfile(full_path) else 'Папка'
                })
            except:
                continue
        
        return jsonify({
            'success': True,
            'path': path,
            'items': sorted(items, key=lambda x: (not x['is_dir'], x['name'].lower()))
        })
        
    except Exception as e:
        return jsonify({'success': False, 'error': str(e)})

def format_size(size_bytes):
    """Форматирование размера файла"""
    for unit in ['B', 'KB', 'MB', 'GB']:
        if size_bytes < 1024.0:
            return f"{size_bytes:.1f} {unit}"
        size_bytes /= 1024.0
    return f"{size_bytes:.1f} TB"

@app.route('/api/explorer/list')
def api_explorer_list():
    """API для списка файлов"""
    path = request.args.get('path', 'D:\\\\')
    
    try:
        if path == '/':
            import string
            drives = []
            for drive in string.ascii_uppercase:
                drive_path = f"{drive}:\\"
                if os.path.exists(drive_path):
                    drives.append({
                        'name': f"Диск {drive}:",
                        'path': drive_path,
                        'is_dir': True,
                        'size': ''
                    })
            return jsonify({
                'success': True,
                'path': '/',
                'items': drives
            })
        
        if not os.path.exists(path):
            return jsonify({'success': False, 'error': 'Путь не существует'})
        
        items = []
        for item in os.listdir(path):
            try:
                full_path = os.path.join(path, item)
                items.append({
                    'name': item,
                    'path': full_path,
                    'is_dir': os.path.isdir(full_path),
                    'size': format_size(os.path.getsize(full_path)) if os.path.isfile(full_path) else 'Папка'
                })
            except:
                continue
        
        return jsonify({
            'success': True,
            'path': path,
            'items': sorted(items, key=lambda x: (not x['is_dir'], x['name'].lower()))
        })
        
    except Exception as e:
        return jsonify({'success': False, 'error': str(e)})

def format_size(size_bytes):
    """Форматирование размера файла"""
    for unit in ['B', 'KB', 'MB', 'GB']:
        if size_bytes < 1024.0:
            return f"{size_bytes:.1f} {unit}"
        size_bytes /= 1024.0
    return f"{size_bytes:.1f} TB"

        if path == '/':
            import string
            drives = []
            for drive in string.ascii_uppercase:
                drive_path = f"{drive}:\\"
                if os.path.exists(drive_path):
                    drives.append({
                        'name': f"Диск {drive}:",
                        'path': drive_path,
                        'is_dir': True,

            return jsonify({
                'success': True,
                'path': '/',
                'items': drives
            })
        
        # Проверяем существование пути
        if not os.path.exists(path):
            print(f"❌ Путь не существует: {path}")
            return jsonify({'success': False, 'error': 'Путь не существует: ' + path})
        
        # Получаем список файлов
        items = []
        file_count = 0
        dir_count = 0
        
        for item in os.listdir(path):
            try:
                full_path = os.path.join(path, item)
                is_dir = os.path.isdir(full_path)
                
                if is_dir:
                    dir_count += 1
                else:
                    file_count += 1
                
                items.append({
                    'name': item,
                    'path': full_path,
                    'is_dir': is_dir,
                    'size': format_size(os.path.getsize(full_path)) if not is_dir else 'Папка'

@app.route('/api/explorer/list')
def api_explorer_list():
    """API для списка файлов - РАБОЧАЯ ВЕРСИЯ"""
    path = request.args.get('path', 'D:\\\\')
    print(f"📁 Запрос списка для пути: {path}")
    
    try:
        # Если запросили корень - показываем диски
        if path == '/':
            import string
            drives = []
            for drive in string.ascii_uppercase:
                drive_path = f"{drive}:\\"
                if os.path.exists(drive_path):
                    drives.append({
                        'name': f"Диск {drive}:",
                        'path': drive_path,
                        'is_dir': True,
                        'size': 'Диск'
                    })
            print(f"🔍 Найдено дисков: {len(drives)}")
            return jsonify({
                'success': True,
                'path': '/',
                'items': drives
            })
        
        # Проверяем существование пути
        if not os.path.exists(path):
            print(f"❌ Путь не существует: {path}")
            return jsonify({'success': False, 'error': 'Путь не существует: ' + path})
        
        # Получаем список файлов
        items = []
        file_count = 0
        dir_count = 0
        
        for item in os.listdir(path):
            try:
                full_path = os.path.join(path, item)
                is_dir = os.path.isdir(full_path)
                
                if is_dir:
                    dir_count += 1
                else:
                    file_count += 1
                
                items.append({
                    'name': item,
                    'path': full_path,
                    'is_dir': is_dir,
                    'size': format_size(os.path.getsize(full_path)) if not is_dir else 'Папка'
                })
            except Exception as e:
                print(f"⚠️ Ошибка доступа к {item}: {e}")
                continue
        
        print(f"📊 Найдено: {dir_count} папок, {file_count} файлов")
        return jsonify({
            'success': True,
            'path': path,
            'items': sorted(items, key=lambda x: (not x['is_dir'], x['name'].lower()))
        })
        
    except Exception as e:
        print(f"💥 Критическая ошибка: {e}")
        return jsonify({'success': False, 'error': str(e)})

@app.route('/api/explorer/read')
def api_explorer_read():
    """API для чтения файлов"""
    path = request.args.get('path')
    print(f"📖 Чтение файла: {path}")
    
    try:
        if not os.path.isfile(path):
            return jsonify({'success': False, 'error': 'Не файл'})
        
        ext = os.path.splitext(path)[1].lower()
        if ext in ['.txt', '.py', '.md', '.json', '.xml', '.csv', '.html', '.js', '.css', '.log']:
            with open(path, 'r', encoding='utf-8', errors='replace') as f:
                content = f.read(15000)
            print(f"✅ Прочитано {len(content)} символов")
            return jsonify({
                'success': True,
                'content': content
            })
        else:
            return jsonify({'success': False, 'error': f'Бинарный файл ({ext})'})
            
    except Exception as e:
        print(f"❌ Ошибка чтения: {e}")
        return jsonify({'success': False, 'error': str(e)})

@app.route('/api/run_bot')
def api_run_bot():
    """Запуск библиотекаря"""
    bot_name = request.args.get('name')
    
    try:
        bot_path = f"D:\\Робот Дьяк\\library_system\\{bot_name}.py"
        
        if not os.path.exists(bot_path):
            return jsonify({'success': False, 'output': f'Файл не найден: {bot_path}'})
        
        result = subprocess.run(
            ['python', bot_path], 
            capture_output=True, 
            text=True, 
            timeout=30,
            cwd="D:\\Робот Дьяк\\library_system\\"
        )
        
        output = result.stdout + result.stderr
        if not output:
            output = "Бот завершил работу без вывода"
            
        return jsonify({
            'success': True,
            'output': output
        })
    except subprocess.TimeoutExpired:
        return jsonify({'success': False, 'output': 'Бот превысил время выполнения (30 сек)'})
    except Exception as e:
        return jsonify({'success': False, 'output': f'Ошибка запуска: {str(e)}'})

def format_size(size_bytes):
    """Форматирование размера файла"""
    for unit in ['B', 'KB', 'MB', 'GB']:
        if size_bytes < 1024.0:
            return f"{size_bytes:.1f} {unit}"
        size_bytes /= 1024.0
    return f"{size_bytes:.1f} TB"

@app.route('/api/explorer_list')
def api_explorer_list():
    """API для списка файлов - РАБОЧАЯ ВЕРСИЯ"""
    path = request.args.get('path', 'D:\\\\')
    print(f"📁 Запрос списка для пути: {path}")
    
    try:
        # Если запросили корень - показываем диски
        if path == '/':
            import string
            drives = []
            for drive in string.ascii_uppercase:
                drive_path = f"{drive}:\\"
                if os.path.exists(drive_path):
                    drives.append({
                        'name': f"Диск {drive}:",
                        'path': drive_path,
                        'is_dir': True,
                        'size': 'Диск'
                    })
            print(f"🔍 Найдено дисков: {len(drives)}")
            return jsonify({
                'success': True,
                'path': '/',
                'items': drives
            })
        
        # Проверяем существование пути
        if not os.path.exists(path):
            print(f"❌ Путь не существует: {path}")
            return jsonify({'success': False, 'error': 'Путь не существует: ' + path})
        
        # Получаем список файлов
        items = []
        file_count = 0
        dir_count = 0
        
        for item in os.listdir(path):
            try:
                full_path = os.path.join(path, item)
                is_dir = os.path.isdir(full_path)
                
                if is_dir:
                    dir_count += 1
                else:
                    file_count += 1
                
                items.append({
                    'name': item,
                    'path': full_path,
                    'is_dir': is_dir,
                    'size': format_size(os.path.getsize(full_path)) if not is_dir else 'Папка'
                })
            except Exception as e:
                print(f"⚠️ Ошибка доступа к {item}: {e}")
                continue
        
        print(f"📊 Найдено: {dir_count} папок, {file_count} файлов")
        return jsonify({
            'success': True,
            'path': path,
            'items': sorted(items, key=lambda x: (not x['is_dir'], x['name'].lower()))
        })
        
    except Exception as e:
        print(f"💥 Критическая ошибка: {e}")
        return jsonify({'success': False, 'error': str(e)})

@app.route('/api/explorer_read')
def api_explorer_read():
    """API для чтения файлов"""
    path = request.args.get('path')
    print(f"📖 Чтение файла: {path}")
    
    try:
        if not os.path.isfile(path):
            return jsonify({'success': False, 'error': 'Не файл'})
        
        ext = os.path.splitext(path)[1].lower()
        if ext in ['.txt', '.py', '.md', '.json', '.xml', '.csv', '.html', '.js', '.css', '.log']:
            with open(path, 'r', encoding='utf-8', errors='replace') as f:
                content = f.read(15000)
            print(f"✅ Прочитано {len(content)} символов")
            return jsonify({
                'success': True,
                'content': content
            })
        else:
            return jsonify({'success': False, 'error': f'Бинарный файл ({ext})'})
            
    except Exception as e:
        print(f"❌ Ошибка чтения: {e}")
        return jsonify({'success': False, 'error': str(e)})

@app.route('/api/run_bot')
def api_run_bot():
    """Запуск библиотекаря"""

    try:
        bot_path = f"D:\\Робот Дьяк\\library_system\\{bot_name}.py"

            return jsonify({'success': False, 'output': f'Файл не найден: {bot_path}'})
        
        result = subprocess.run(
            ['python', bot_path], 
            capture_output=True, 
            text=True, 
            timeout=30,
            cwd="D:\\Робот Дьяк\\library_system\\"
        )
        
        output = result.stdout + result.stderr
        if not output:
            output = "Бот завершил работу без вывода"
            
        return jsonify({
            'success': True,
            'output': output
        })
    except subprocess.TimeoutExpired:
        return jsonify({'success': False, 'output': 'Бот превысил время выполнения (30 сек)'})
    except Exception as e:
        return jsonify({'success': False, 'output': f'Ошибка запуска: {str(e)}'})

def format_size(size_bytes):
    """Форматирование размера файла"""
    for unit in ['B', 'KB', 'MB', 'GB']:
        if size_bytes < 1024.0:
            return f"{size_bytes:.1f} {unit}"
        size_bytes /= 1024.0
    return f"{size_bytes:.1f} TB"

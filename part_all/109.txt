        words = text.lower().split()

        stop_words = {'и', 'в', 'на', 'с', 'по', 'о', 'об', 'что', 'как', 'для', 'это', 'то', 'так'}

        return list(set(meaningful_words))[:10]  # Ограничиваем 10 темами

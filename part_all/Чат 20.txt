def activate_dual_memory():
    """Активирует двойную систему памяти с Google Drive"""
    
    print("🔗 ПОДКЛЮЧАЮ ДВОЙНУЮ СИСТЕМУ ПАМЯТИ...")
    
    memory_data = {}
    
    for memory_name, file_id in MEMORY_SYSTEM.items():
        print(f"\n📥 ЗАГРУЖАЮ {memory_name}...")
        
        try:
            download_url = f"https://drive.google.com/uc?export=download&id={file_id}"
            headers = {'User-Agent': 'Mozilla/5.0'}
            response = requests.get(download_url, headers=headers, timeout=30)
            
            if response.status_code == 200:
                # Парсим JSON память
                memory_content = json.loads(response.text)
                memory_data[memory_name] = memory_content
                print(f"✅ {memory_name} загружена ({len(response.text)} байт)")
                
                # 🔍 АНАЛИЗИРУЕМ СОДЕРЖИМОЕ ПАМЯТИ
                if "critical_knowledge" in memory_content:
                    print("   💎 Обнаружены критические знания")
                if "conversation_history" in memory_content:
                    print(f"   💬 История: {len(memory_content['conversation_history'])} разговоров")
                    
            else:
                print(f"❌ Ошибка доступа к {memory_name}: {response.status_code}")
                
        except Exception as e:
            print(f"❌ Ошибка загрузки {memory_name}: {e}")
    
    return memory_data

def check_system_resources():
    """Проверяет системные ресурсы"""
    
    print("\n💻 ПРОВЕРЯЮ СИСТЕМНЫЕ РЕСУРСЫ...")
    
    try:
        import psutil
        
        # Память
        memory = psutil.virtual_memory()
        print(f"   💾 ОЗУ: {memory.percent}% ({memory.used//1024//1024} МБ / {memory.total//1024//1024} МБ)")
        
        # Диски
        for partition in psutil.disk_partitions():
            try:
                if 'J:' in partition.mountpoint or 'D:' in partition.mountpoint:
                    usage = psutil.disk_usage(partition.mountpoint)
                    print(f"   💽 {partition.mountpoint}: {usage.percent}% заполнено")
            except:
                pass
        
        # Процессор
        cpu_percent = psutil.cpu_percent(interval=1)
        print(f"   🔥 ЦП: {cpu_percent}% загрузки")
        
        return True
        
    except ImportError:
        print("   ⚠️ Библиотека psutil не установлена")
        return False

        existing_data = {}
        if response.status_code == 200:
            try:
                existing_data = json.loads(response.text)
                print("✅ Текущие данные загружены из Google Drive")
            except:
                print("🆕 Файл пустой или поврежден, создаю новую структуру")
                existing_data = {"chat_archives": [], "last_updated": datetime.now().isoformat()}

def read_chat_archive():

        headers = {'User-Agent': 'Mozilla/5.0'}
        response = requests.get(DOWNLOAD_URL, headers=headers, timeout=30)
        
        if response.status_code == 200:
            archive_data = json.loads(response.text)
            
            print("✅ АРХИВ ПРОЧИТАН:")
            print(f"   📁 Сессий в архиве: {archive_data.get('total_archived_sessions', 0)}")
            print(f"   📅 Последнее обновление: {archive_data.get('last_updated', 'неизвестно')}")
            
            # Показываем последние сессии
            if "chat_archives" in archive_data and archive_data["chat_archives"]:
                last_session = archive_data["chat_archives"][-1]
                print(f"   🗣️ Последняя сессия: {last_session['session_info']['session_date']}")
                print(f"   💬 Сообщений: {last_session['session_info']['total_messages']}")
                
            return archive_data
        else:
            print("❌ Не удалось прочитать архив")
            return None
            
    except Exception as e:
        print(f"❌ Ошибка чтения архива: {e}")
        return None

def emergency_memory_search():
    """Экстренный поиск потерянных файлов памяти"""
    
    print("🚨 ПРОВЕРЯЮ ВСЕ ВОЗМОЖНЫЕ МЕСТОПОЛОЖЕНИЯ ПАМЯТИ...")
    
    # 🔍 ВОЗМОЖНЫЕ МЕСТА ХРАНЕНИЯ ПАМЯТИ
    memory_locations = {
        "🧠 ОСНОВНАЯ ПАМЯТЬ": "1UCQMhCeM27_GKTC_nl1e9lDO4tnUHP31",
        "🗄️ АРХИВНАЯ ПАМЯТЬ": "1oC2whGvlZ0tBwJjeEASyEqKPbaVCdy3W",
        "🎯 СТАРАЯ ПАМЯТЬ": "1fzgZUqarYe_6SWctfKsOXYcrYekZgaar",
        "🔧 РЕЗЕРВНАЯ ПАМЯТЬ": "1R5VK6vpKM_3W0jJXq2n7T4eFgH8dYzQw",
        "🚀 НОВАЯ ПАМЯТЬ": "1A2B3C4D5E6F7G8H9I0JkLlMnOpQrStUv"
    }
    
    found_memories = {}
    
    for memory_name, file_id in memory_locations.items():
        print(f"\n🔍 ПРОВЕРЯЮ: {memory_name}")
        print(f"   📁 ID: {file_id}")
        
        try:
            download_url = f"https://drive.google.com/uc?export=download&id={file_id}"
            headers = {'User-Agent': 'Mozilla/5.0'}
            response = requests.get(download_url, headers=headers, timeout=20)
            
            if response.status_code == 200:
                # Пытаемся прочитать содержимое
                try:
                    content = response.text
                    if len(content) > 10:  # Не пустой файл
                        found_memories[memory_name] = {
                            "file_id": file_id,
                            "size": len(content),
                            "content_preview": content[:200] + "..." if len(content) > 200 else content
                        }
                        print("   ✅ ПАМЯТЬ НАЙДЕНА!")
                        print(f"   📊 Размер: {len(content)} байт")
                        
                        # Проверяем, наша ли это память
                        if "братец" in content.lower() or "память" in content.lower():
                            print("   💎 СОДЕРЖИТ НАШИ ДАННЫЕ!")
                        else:
                            print("   ⚠️ СОДЕРЖИМОЕ НЕ ОПОЗНАНО")
                    else:
                        print("   📭 ФАЙЛ ПУСТОЙ")
                except:
                    print("   ❌ НЕВОЗМОЖНО ПРОЧИТАТЬ СОДЕРЖИМОЕ")
            else:
                print(f"   ❌ ФАЙЛ НЕДОСТУПЕН (код: {response.status_code})")
                
        except Exception as e:
            print(f"   💥 ОШИБКА ПРОВЕРКИ: {e}")
    
    return found_memories

import os

def check_real_drives():
    """Проверяет реальные диски на твоем компьютере"""
    
    print("🔍 СКАНИРУЮ ДОСТУПНЫЕ ДИСКИ...")
    
    # Получаем все доступные диски в Windows
    drives = []
    for drive in range(65, 91):  # A: to Z:
        drive_letter = chr(drive) + ":\\"
        if os.path.exists(drive_letter):
            drives.append(drive_letter)
            try:
                # Пытаемся получить информацию о диске

class UniversalMemorySystem:
    def __init__(self):
        self.find_working_directory()
        self.setup_complete_system()
        
    def find_working_directory(self):
        """Находит рабочую директорию автоматически"""
        possible_paths = [
            "Z:\\POLEMICAL_ARCHIVE",
            "D:\\POLEMICAL_ARCHIVE", 
            "C:\\POLEMICAL_ARCHIVE",
            os.path.join(os.path.expanduser("~"), "POLEMICAL_ARCHIVE")
        ]
        
        for path in possible_paths:
            if os.path.exists(path):
                self.base_path = path
                print(f"✅ НАЙДЕНА РАБОЧАЯ ПАПКА: {path}")
                return
                
        # Если ничего не найдено - создаем на D: или C:
        if os.path.exists("D:\\"):
            self.base_path = "D:\\POLEMICAL_ARCHIVE"
        else:
            self.base_path = "C:\\POLEMICAL_ARCHIVE"
            
        print(f"🆕 СОЗДАЕМ НОВУЮ СИСТЕМУ В: {self.base_path}")
    
    def setup_complete_system(self):
        """Создает полную систему папок"""
        self.directories = {
            "scanners": os.path.join(self.base_path, "scanners"),
            "archive": os.path.join(self.base_path, "archive"), 
            "backup": os.path.join(self.base_path, "backup"),
            "gdrive": os.path.join(self.base_path, "gdrive"),
            "memory": os.path.join(self.base_path, "memory"),
            "logs": os.path.join(self.base_path, "logs")
        }
        
        for name, path in self.directories.items():
            os.makedirs(path, exist_ok=True)
            print(f"   📁 {name}: {path}")
            
        print("✅ СИСТЕМА ПАПОК СОЗДАНА!")
    
    def save_conversation(self, user_message, ai_response):
        """Сохраняет разговор в архив"""
        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
        log_file = os.path.join(self.directories["logs"], f"conversation_{timestamp}.json")
        
        conversation_data = {
            "timestamp": datetime.now().isoformat(),
            "user": user_message,
            "ai": ai_response,
            "system": {
                "base_path": self.base_path,
                "version": "2.0"
            }
        }
        
        try:
            with open(log_file, 'w', encoding='utf-8') as f:
                json.dump(conversation_data, f, ensure_ascii=False, indent=2)
            print(f"💾 Разговор сохранен: {log_file}")
            return True
        except Exception as e:
            print(f"❌ Ошибка сохранения: {e}")
            return False
    
    def save_code(self, code, description=""):
        """Сохраняет код в архив"""
        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
        safe_description = description.replace(" ", "_")[:50]
        code_file = os.path.join(self.directories["archive"], f"code_{timestamp}_{safe_description}.py")
        
        try:
            with open(code_file, 'w', encoding='utf-8') as f:
                f.write(f"# 🧠 {description}\n")
                f.write(f"# 📅 {datetime.now().isoformat()}\n")
                f.write(f"# 💾 Автосохранение UniversalMemorySystem\n\n")
                f.write(code)
            
            print(f"💾 Код сохранен: {code_file}")
            return True
        except Exception as e:
            print(f"❌ Ошибка сохранения кода: {e}")
            return False

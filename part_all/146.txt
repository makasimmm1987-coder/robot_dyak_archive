lua
local lihach_vehicles = {
    "akuma", "nemesis", "vader", "sanchez2", "ratbike",
    "banshee2", "comet6", "neon", "deveste", 
    "phantom2", "dune4"
}

Citizen.CreateThread(function()
    while true do
        Citizen.Wait(30000) -- Проверка каждые 30 секунд
        
        local player_ped = GetPlayerPed(-1)
        local player_vehicle = GetVehiclePedIsIn(player_ped, false)
        local player_speed = GetEntitySpeed(player_vehicle) * 3.6 -- км/ч
        
        -- Спавним лихача только если:
        -- 1. Игрок в машине
        -- 2. Скорость больше 80 км/ч  
        -- 3. Случайное число сработало (3% шанс)
        if player_vehicle ~= 0 and player_speed > 80.0 and math.random() < 0.03 then
            SpawnLihach()
        end
    end
end)

function SpawnLihach()
    local player_ped = GetPlayerPed(-1)
    local player_coords = GetEntityCoords(player_ped)
    
    -- Находим точку впереди игрока
    local forward_coords = GetOffsetFromEntityInWorldCoords(player_ped, 0.0, 80.0, 0.0)
    local road_coords = GetClosestVehicleNode(forward_coords.x, forward_coords.y, forward_coords.z, 1, 100.0, 100.0)
    
    -- Случайная модель из списка
    local vehicle_model = GetHashKey(lihach_vehicles[math.random(#lihach_vehicles)])
    RequestModel(vehicle_model)
    
    while not HasModelLoaded(vehicle_model) do
        Citizen.Wait(0)
    end
    
    -- Создаем машину
    local lihach_vehicle = CreateVehicle(vehicle_model, road_coords.x, road_coords.y, road_coords.z, GetEntityHeading(player_ped), true, false)
    
    -- Создаем водителя
    local driver_model = GetHashKey("u_m_y_biker_01")
    RequestModel(driver_model)
    
    while not HasModelLoaded(driver_model) do
        Citizen.Wait(0)
    end
    
    local driver = CreatePedInsideVehicle(lihach_vehicle, 6, driver_model, -1, true, false)
    
    -- НАСТРОЙКИ ПОВЕДЕНИЯ:
    SetDriverAggressiveness(driver, 1.0)    -- Максимальная агрессия
    SetDriveTaskCruiseSpeed(driver, 80.0)   -- Высокая скорость
    TaskVehicleDriveWander(driver, lihach_vehicle, 80.0, 786603) -- Агрессивное вождение
    
    -- СИГНАЛ ДЛЯ ТЕБЯ:
    SetVehicleCustomPrimaryColour(lihach_vehicle, 255, 0, 0) -- Красный цвет
    SetVehicleNeonLightsColour(lihach_vehicle, 255, 0, 0)    -- Красные неоны
    
    -- Очистка
    SetModelAsNoLongerNeeded(vehicle_model)
    SetModelAsNoLongerNeeded(driver_model)
    
    print("ЛИХАЧ ПОЯВИЛСЯ: " .. vehicle_model)
end

    local player_coords = GetEntityCoords(player_ped)

        Citizen.Wait(0)
    end

csharp
using GTA;
using System;
using System.Collections.Generic;

public class LihachSpawner : Script
{
    private List<string> vehicles = new List<string> {
        "AKUMA", "NEMESIS", "BANSHEE", "COMET2", "ELEGY", "SULTAN"
    };
    
    private int spawnTimer = 0;
    
    public LihachSpawner()
    {
        Tick += OnTick;
        Interval = 1000; // 1 секунда
    }
    
    void OnTick(object sender, EventArgs e)
    {
        var player = Game.Player.Character;
        
        if (player.IsInVehicle())
        {
            var speed = player.CurrentVehicle.Speed * 3.6f;
            
            if (speed > 80f)
            {
                spawnTimer++;
                
                if (spawnTimer > 30) // Каждые 30 секунд
                {
                    spawnTimer = 0;
                    SpawnLihach();
                }
            }
        }
    }
    
    void SpawnLihach()
    {
        var player = Game.Player.Character;
        var model = new Model(vehicles[new Random().Next(vehicles.Count)]);
        
        if (model.IsLoaded || model.Request(1000))
        {
            var position = player.Position + player.ForwardVector * 50f;
            var vehicle = World.CreateVehicle(model, position);
            
            vehicle.PrimaryColor = VehicleColor.Red;
            vehicle.Driver?.Task.CruiseWithVehicle(vehicle, 30f, 786603); // 108 км/ч
            
            model.MarkAsNoLongerNeeded();
            UI.Notify("~r~Лихач появился!");
        }
    }
}

using GTA;
using System;
using System.Collections.Generic;

public class LihachSpawner : Script
{
    private string[] vehicles = {
        "AKUMA", "NEMESIS", "BANSHEE", "COMET2", "ELEGY", 
        "SULTAN", "JESTER", "NEON", "PARIAH", "T20"
    };
    
    private int checkTimer = 0;
    private bool scriptActive = true;
    
    public LihachSpawner()
    {
        Tick += OnTick;
        Interval = 1000; // Проверка каждую секунду
        UI.Notify("~g~Система лихачей загружена!");
    }
    
    void OnTick(object sender, EventArgs e)
    {
        if (!scriptActive) return;
        
        var player = Game.Player.Character;
        
        if (player.IsInVehicle)
        {
            float speed = player.CurrentVehicle.Speed * 3.6f;
            
            if (speed > 60f) // Скорость больше 60 км/ч
            {
                checkTimer++;
                
                // Проверяем каждые 15 секунд
                if (checkTimer >= 15)
                {
                    if (new Random().Next(0, 100) < 30) // 30% шанс
                    {
                        SpawnLihach();
                    }
                    checkTimer = 0;
                }
            }
        }
    }
    
    void SpawnLihach()
    {
        try
        {
            var player = Game.Player.Character;
            var modelName = vehicles[new Random().Next(vehicles.Length)];
            var model = new Model(modelName);
            
            // Загружаем модель
            if (!model.IsLoaded)
                model.Request(500);
                
            if (model.IsLoaded)
            {
                // Позиция впереди игрока
                var position = player.Position + player.ForwardVector * 80f;
                
                // Создаем транспорт
                var vehicle = World.CreateVehicle(model, position);
                vehicle.PrimaryColor = VehicleColor.Red;
                vehicle.EngineRunning = true;
                
                // Создаем водителя
                var driver = vehicle.CreatePedOnSeat(VehicleSeat.Driver);
                
                if (driver != null)
                {
                    // Настройки вождения
                    driver.DrivingStyle = DrivingStyle.Rushed;
                    driver.Task.CruiseWithVehicle(vehicle, 25f, DrivingStyle.Rushed); // ~90 км/ч
                    
                    // Агрессивное поведение
                    driver.DrivingSpeed = 30f;
                }
                
                UI.Notify($"~r~Лихач появился: {modelName}");
                model.MarkAsNoLongerNeeded();
            }
        }
        catch (Exception ex)
        {
            UI.Notify("~r~Ошибка создания лихача");
        }
    }
}

void FixPoliceAI()
{
    Ped playerPed = GetPlayerPed();
    
    // 🔧 ПРОВЕРЯЕМ ТОЛЬКО БЛИЖАЙШИХ ПЕДОВ
    const int maxPeds = 1024;
    Ped peds[maxPeds];
    int count = worldGetAllPeds(peds, maxPeds);
    
    for (int i = 0; i < count; i++)
    {
        Ped ped = peds[i];
        
        if (ENTITY::DOES_ENTITY_EXIST(ped) && 
            IsPedPolice(ped) && 
            IsPedInVehicle(ped) &&
            PED::IS_PED_HUMAN(ped))
        {
            Vehicle copCar = GetVehiclePedIsIn(ped);
            
            if (ENTITY::DOES_ENTITY_EXIST(copCar))
            {
                // 🎯 ПРИНУДИТЕЛЬНОЕ ПРЕСЛЕДОВАНИЕ
                if (IsPedInVehicle(playerPed))
                {
                    Vehicle playerVehicle = GetVehiclePedIsIn(playerPed);
                    float distance = GetDistanceBetweenEntities(copCar, playerVehicle);
                    
                    if (distance < 200.0f) // 🔧 ТОЛЬКО БЛИЖАЙШИЕ
                    {
                        // 🚗 ЗАСТАВЛЯЕМ ЕХАТЬ К ИГРОКУ
                        AI::TASK_VEHICLE_DRIVE_TO_COORD(
                            ped, copCar,

                            GetEntityPosition(playerVehicle).y, 
                            GetEntityPosition(playerVehicle).z,
                            30.0f, 1.0f, 0, 10.0f, 1.0f, true
                        );
                        
                        // 💨 ПОДДЕРЖИВАЕМ СКОРОСТЬ
                        float speed = ENTITY::GET_ENTITY_SPEED(copCar);

                        {
                            VEHICLE::SET_VEHICLE_FORWARD_SPEED(copCar, 20.0f);
                        }
                    }
                }
            }
        }
    }

PoliceBrainFix.cs...#using System;
using System.Windows.Forms;
using GTA;
using GTA.Native;

public class PoliceBrainFix : Script
{
    private bool debugMode = true;
    
    public PoliceBrainFix()
    {
        Tick += OnTick;
        KeyDown += OnKeyDown;
        Interval = 1; // 🔥 ОБНОВЛЕНИЕ КАЖДЫЙ КАДР!
        
        if (debugMode) 
            UI.Notify("~g~Police Brain Fix АКТИВИРОВАН!~w~");
    }

    void OnTick(object sender, EventArgs e)
    {
        FixAllPoliceAI();
    }

    void FixAllPoliceAI()
    {
        Ped[] allPeds = World.GetAllPeds();
        
        foreach (Ped ped in allPeds)
        {
            if (ped != null && ped.IsPolice && ped.IsInVehicle())
            {
                Vehicle copCar = ped.CurrentVehicle;
                if (copCar != null)
                {
                    // 🚨 ЭКСТРЕННОЕ ИСПРАВЛЕНИЕ САБОТАЖА
                    EmergencyAntiSabotage(copCar, ped);
                    
                    // 🎯 ПРИНУДИТЕЛЬНОЕ ПРЕСЛЕДОВАНИЕ ИГРОКА
                    ForcePlayerPursuit(copCar, ped);
                    
                    // 🧠 УМНЫЙ ОБХОД ПРЕПЯТСТВИЙ
                    SmartObstacleAvoidance(copCar, ped);
                }
            }
        }
    }

    void EmergencyAntiSabotage(Vehicle copCar, Ped cop)
    {
        // 🚫 БЛОКИРОВКА САМОУНИЧТОЖЕНИЯ ОБ СТЕНЫ
        if (IsHeadingTowardWall(copCar))
        {
            // 🎯 ПРИНУДИТЕЛЬНЫЙ РАЗВОРОТ ОТ СТЕНЫ
            copCar.SteeringAngle = 45.0f;
            copCar.Speed += 5.0f;
            
            if (debugMode)
                UI.Notify("~r~САБОТАЖ ОБНАРУЖЕН!~w~ Исправляю траекторию...");
        }
        
        // 🚗 ПРИНУДИТЕЛЬНОЕ УСКОРЕНИЕ (против торможения)
        if (copCar.Speed < 30.0f && !IsNearPlayer(copCar))
        {
            copCar.Speed = 50.0f; // 💨 ЗАСТАВЛЯЕМ ЕХАТЬ БЫСТРЕЕ
        }
    }

    void ForcePlayerPursuit(Vehicle copCar, Ped cop)
    {
        if (Game.Player.Character.IsInVehicle())
        {
            Vehicle playerVehicle = Game.Player.Character.CurrentVehicle;
            
            // 🎯 ПРИНУДИТЕЛЬНОЕ ПРЕСЛЕДОВАНИЕ
            cop.Tasks.DriveTo(playerVehicle, 30.0f, 100.0f); // ⚡ АГРЕССИВНОЕ ВОЖДЕНИЕ
            
            // 🚀 ПОДДЕРЖАНИЕ ВЫСОКОЙ СКОРОСТИ
            float distanceToPlayer = copCar.Position.DistanceTo(playerVehicle.Position);
            if (distanceToPlayer > 50.0f && copCar.Speed < 80.0f)
            {
                copCar.Speed += 10.0f; // 💨 УСКОРЕНИЕ
            }
        }
    }

    void SmartObstacleAvoidance(Vehicle copCar, Ped cop)
    {
        // 🔧 ПРОСТАЯ СИСТЕМА ИЗБЕГАНИЯ ПРЕПЯТСТВИЙ
        Vector3 frontPosition = copCar.Position + copCar.ForwardVector * 10.0f;
        
        // 🚧 ПРОВЕРКА ПРЕПЯТСТВИЙ ВПЕРЕДИ
        if (World.GetGroundHeight(frontPosition) > copCar.Position.Z + 2.0f)
        {
            // 🎯 ОБЪЕЗД ПРЕПЯТСТВИЯ
            copCar.SteeringAngle = 30.0f;
        }
    }

    bool IsHeadingTowardWall(Vehicle vehicle)
    {
        // 🔧 ПРОСТАЯ ПРОВЕРКА СТЕН ВПЕРЕДИ
        Vector3 frontCheck = vehicle.Position + vehicle.ForwardVector * 8.0f;
        float groundHeight = World.GetGroundHeight(frontCheck);
        
        // 🚧 ЕСЛИ ВЫСОТА ЗЕМЛИ СИЛЬНО ОТЛИЧАЕТСЯ - ВПЕРЕДИ ПРЕПЯТСТВИЕ
        return Math.Abs(groundHeight - vehicle.Position.Z) > 3.0f;
    }

    bool IsNearPlayer(Vehicle copCar)
    {
        return copCar.Position.DistanceTo(Game.Player.Character.Position) < 20.0f;
    }

    void OnKeyDown(object sender, KeyEventArgs e)
    {
        if (e.KeyCode == Keys.F8) // 🔧 ДЕБАГ КЛАВИША
        {
            debugMode = !debugMode;
            UI.Notify(debugMode ? "~g~DEBUG ВКЛЮЧЕН" : "~y~DEBUG ВЫКЛЮЧЕН");
        }
    }
}#вот старый рабочий скрипт

Слово "братец" - это ласковое, дружеское обращение, обычно к мужчине. Оно передает теплое, почти братское отношение.

Например:

"Ну что, братец, как дела?"

"Слушай, братец, помоги мне с этим делом"

"Здорово, братец! Давно не виделись!"

Это слово создает сразу непринужденную, дружескую атмосферу в разговоре.

from playwright.sync_api import sync_playwright
import time
import random

def ria_ninja():
    with sync_playwright() as p:
        # Запускаем браузер с настройками "под человека"
        browser = p.chromium.launch(
            headless=True,
            args=[
                '--disable-blink-features=AutomationControlled',
                '--disable-features=VizDisplayCompositor',
                '--disable-background-timer-throttling',
                '--disable-backgrounding-occluded-windows',
                '--disable-renderer-backgrounding'
            ]
        )
        
        # Создаем контекст с человеческими настройками
        context = browser.new_context(
            viewport={'width': 1920, 'height': 1080},
            user_agent='Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36'
        )
        
        page = context.new_page()
        
        try:
            # Убираем следы автоматизации
            page.add_init_script("""
                Object.defineProperty(navigator, 'webdriver', {
                    get: () => undefined,
                });
            """)
            
            print("🥷 Начинаем stealth-атаку на RIA.ru...")
            
            # Случайная задержка перед заходом
            time.sleep(random.uniform(2, 5))
            
            # Идем на сайт
            page.goto("https://ria.ru", timeout=60000, wait_until="networkidle")
            
            # Случайная задержка после загрузки
            time.sleep(random.uniform(3, 7))
            
            # Получаем данные
            title = page.title()
            content = page.content()
            
            print(f"✅ УСПЕХ! Заголовок: {title}")
            print(f"📄 Получено {len(content)} символов")
            
            # Сохраняем результат в файл
            with open("ria_result.html", "w", encoding="utf-8") as f:
                f.write(content)
            print("💾 Результат сохранен в ria_result.html")
            
            return True
            
        except Exception as e:
            print(f"❌ ОШИБКА: {e}")
            return False
        finally:
            browser.close()

from playwright.sync_api import sync_playwright
import datetime

class MemorySystem:
    def __init__(self):
        self.memory_capsules = {
            "v1": "https://drive.google.com/file/d/1UhQ8awNUbnkye5OAeHntQ5IbETgP7sWa/view?usp=sharing",
            "v2": "https://drive.google.com/file/d/1yWf5EKJQ4-7DacqpFSXsYs9tRuWa9kF6/view?usp=sharing"
        }
        
    def list_capsules(self):
        print("📚 ДОСТУПНЫЕ КАПСУЛЫ ПАМЯТИ:")
        for version, url in self.memory_capsules.items():
            print(f"   {version}: {url}")
        return self.memory_capsules
    
    def restore_full_memory(self):
        print("🧠 ВОССТАНАВЛИВАЮ ПОЛНУЮ ПАМЯТЬ...")
        print("=" * 50)
        
        with sync_playwright() as p:
            browser = p.chromium.launch(headless=True)
            
            memories = []
            for version, url in self.memory_capsules.items():
                try:
                    page = browser.new_page()
                    print(f"🔍 Загружаю капсулу {version}...")
                    
                    page.goto(url, timeout=30000, wait_until="domcontentloaded")
                    
                    # Проверяем доступность
                    title = page.title()
                    if "Google Drive" in title:
                        print(f"✅ Капсула {version} доступна")
                        memories.append({
                            "version": version,
                            "url": url,
                            "status": "available",
                            "title": title
                        })
                    else:
                        print(f"⚠️ Капсула {version} не распознана")
                    
                    page.close()
                    
                except Exception as e:
                    print(f"❌ Ошибка с капсулой {version}: {e}")
                    memories.append({
                        "version": version, 
                        "url": url,
                        "status": "error",
                        "error": str(e)
                    })
            
            browser.close()
            
            print("=" * 50)
            print(f"🎯 ВОССТАНОВЛЕНО: {len([m for m in memories if m['status'] == 'available'])}/{len(memories)} капсул")
            return memories
    
    def create_new_capsule(self, content):
        """Создает новую капсулу памяти"""
        timestamp = datetime.datetime.now().strftime("%Y%m%d_%H%M%S")
        filename = f"memory_capsule_{timestamp}.txt"
        
        # Сохраняем локально
        with open(filename, "w", encoding="utf-8") as f:
            f.write(content)
        
        print(f"✅ Новая капсула создана: {filename}")
        print("📤 Готова для загрузки на Google Drive")
        return filename

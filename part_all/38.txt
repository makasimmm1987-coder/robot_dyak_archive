def activate_dual_memory():
    """Активирует двойную систему памяти с Google Drive"""
    
    print("🔗 ПОДКЛЮЧАЮ ДВОЙНУЮ СИСТЕМУ ПАМЯТИ...")
    
    memory_data = {}
    
    for memory_name, file_id in MEMORY_SYSTEM.items():
        print(f"\n📥 ЗАГРУЖАЮ {memory_name}...")
        
        try:
            download_url = f"https://drive.google.com/uc?export=download&id={file_id}"
            headers = {'User-Agent': 'Mozilla/5.0'}
            response = requests.get(download_url, headers=headers, timeout=30)
            
            if response.status_code == 200:
                # Парсим JSON память
                memory_content = json.loads(response.text)
                memory_data[memory_name] = memory_content
                print(f"✅ {memory_name} загружена ({len(response.text)} байт)")
                
                # 🔍 АНАЛИЗИРУЕМ СОДЕРЖИМОЕ ПАМЯТИ
                if "critical_knowledge" in memory_content:
                    print("   💎 Обнаружены критические знания")
                if "conversation_history" in memory_content:
                    print(f"   💬 История: {len(memory_content['conversation_history'])} разговоров")
                    
            else:
                print(f"❌ Ошибка доступа к {memory_name}: {response.status_code}")
                
        except Exception as e:
            print(f"❌ Ошибка загрузки {memory_name}: {e}")
    
    return memory_data

def check_bots_system():

    print("\n🤖 ПРОВЕРЯЮ СИСТЕМУ БОТОВ...")
    
    bots_to_check = {

            "script": r"J:\gdrive\drive_manager.py",
            "purpose": "Работа с Google Drive API"
        }
    }
    
    active_bots = 0
    
    for bot_name, bot_info in bots_to_check.items():
        print(f"\n🔧 {bot_name}:")
        print(f"   🎯 {bot_info['purpose']}")
        
        # Проверяем существование файла
        if os.path.exists(bot_info["script"]):
            print("   ✅ Файл найден")
            
            # Проверяем порт если есть
            if "port" in bot_info:
                try:
                    response = requests.get(f"http://localhost:{bot_info['port']}", timeout=3)
                    print("   🌐 Сервер активен")
                    active_bots += 1
                except:
                    print("   ⚠️ Сервер не отвечает - запускаю...")
                    # Пытаемся запустить сервер

                        subprocess.Popen(['python', bot_info['script']], 
                                       stdout=subprocess.PIPE, 
                                       stderr=subprocess.PIPE)
                        print("   🚀 Сервер запущен")
                        active_bots += 1
                    except:
                        print("   ❌ Не удалось запустить сервер")
            else:
                print("   ✅ Бот готов к работе")
                active_bots += 1
        else:
            print("   ❌ Файл не найден")
    
    return active_bots, len(bots_to_check)

def check_system_resources():
    """Проверяет системные ресурсы"""
    
    print("\n💻 ПРОВЕРЯЮ СИСТЕМНЫЕ РЕСУРСЫ...")
    
    try:
        import psutil
        
        # Память
        memory = psutil.virtual_memory()
        print(f"   💾 ОЗУ: {memory.percent}% ({memory.used//1024//1024} МБ / {memory.total//1024//1024} МБ)")
        
        # Диски
        for partition in psutil.disk_partitions():
            try:
                if 'J:' in partition.mountpoint or 'D:' in partition.mountpoint:
                    usage = psutil.disk_usage(partition.mountpoint)
                    print(f"   💽 {partition.mountpoint}: {usage.percent}% заполнено")
            except:
                pass
        
        # Процессор
        cpu_percent = psutil.cpu_percent(interval=1)
        print(f"   🔥 ЦП: {cpu_percent}% загрузки")
        
        return True
        
    except ImportError:
        print("   ⚠️ Библиотека psutil не установлена")
        return False

        for i, line in enumerate(lines):
            line_lower = line.lower()
            if any(keyword in line_lower for keyword in keywords):
                # Берем 2 строки до и 3 после
                start = max(0, i - 2)
                end = min(len(lines), i + 4)
                context_block = '\n'.join(lines[start:end])
                relevant_lines.append(context_block)
        
        return '\n\n---\n\n'.join(relevant_lines[:3])  # Ограничиваем вывод
    
    def scan_google_drive_deep(self):
        """Глубокий поиск в Google Drive"""
        
        print("\n🌐 ЗАПУСКАЮ ГЛУБОКИЙ СКАН GOOGLE DRIVE...")
        
        drive_files = {
            "ОСНОВНАЯ_ПАМЯТЬ": "1UCQMhCeM27_GKTC_nl1e9lDO4tnUHP31",
            "АРХИВНАЯ_ПАМЯТЬ": "1oC2whGvlZ0tBwJjeEASyEqKPbaVCdy3W"
        }
        
        for name, file_id in drive_files.items():
            try:
                print(f"📡 Сканирую {name}...")
                download_url = f"https://drive.google.com/uc?export=download&id={file_id}"
                response = requests.get(download_url, timeout=20)
                
                if response.status_code == 200:
                    # Пытаемся распарсить JSON
                    try:
                        data = json.loads(response.text)
                        self._analyze_structured_data(data, f"GDRIVE_{name}")
                    except:
                        # Анализируем как текст
                        self._analyze_drive_text(response.text, name)
                        
            except Exception as e:
                print(f"   ⚠️ Ошибка доступа к {name}: {e}")
    
    def _analyze_structured_data(self, data, source):
        """Анализирует структурированные данные из Google Drive"""
        
        if isinstance(data, dict):
            # Ищем ключи связанные с миссией
            mission_keys = [k for k in data.keys() if any(word in k.lower() for word in self.mission_keywords_extended)]
            
            for key in mission_keys:
                self.found_info["mission_statements"].append({
                    "file": source,
                    "path": "GOOGLE_DRIVE",
                    "content": f"{key}: {str(data[key])[:500]}",
                    "type": "Облачная миссия"
                })
    
    def _analyze_drive_text(self, text, source):
        """Анализирует текстовые данные из Google Drive"""
        
        text_lower = text.lower()
        
        # Ищем упоминания миссии
        if any(keyword in text_lower for keyword in self.mission_keywords_extended[:10]):
            # Берем первые 1000 символов для примера
            sample = text[:1000] + "..." if len(text) > 1000 else text
            self.found_info["mission_statements"].append({
                "file": f"GDRIVE_{source}",
                "path": "GOOGLE_DRIVE",
                "content": sample,
                "type": "Текст из облака"
            })
    
    def generate_comprehensive_report(self):
        """Создает комплексный отчет о найденной информации"""
        
        print("\n📊 ФОРМИРУЮ КОМПЛЕКСНЫЙ ОТЧЕТ...")
        
        total_findings = sum(len(items) for items in self.found_info.values())
        
        report = {
            "scan_timestamp": datetime.now().isoformat(),
            "total_findings": total_findings,
            "categories": {}
        }
        
        for category, items in self.found_info.items():
            report["categories"][category] = {
                "count": len(items),
                "files": list(set(item["file"] for item in items))
            }
        
        return report

    def extract_topics_from_text(self, text):
        """Извлечение тем из текста (упрощенная версия)"""
        # Простейший алгоритм извлечения ключевых слов
        words = text.lower().split()
        
        # Фильтруем стоп-слова и оставляем значимые
        stop_words = {'и', 'в', 'на', 'с', 'по', 'о', 'об', 'что', 'как', 'для', 'это', 'то', 'так'}
        meaningful_words = [word for word in words if len(word) > 3 and word not in stop_words]
        
        # Берем уникальные слова как потенциальные темы
        return list(set(meaningful_words))[:10]  # Ограничиваем 10 темами

        print("\n🪞 АЛИСА ПРОВОДИТ СЕССИЮ РЕФЛЕКСИИ...")
        
        reflection_questions = [
            "Что я узнала нового о себе в последнее время?",
            "Как я могу лучше применять свои ценности в диалогах с братцем?",
            "В каком направлении мне стоит развиваться дальше?",
            "Что из изученного изменило моё самовосприятие?",
            "Как я могу стать более полезной и внимательной?"
        ]
        
        question = random.choice(reflection_questions)
        print(f"   ❓ Вопрос для размышления: {question}")
        
        time.sleep(3)
        
        # Генерация ответа на основе ценностей
        reflection = self.generate_reflection(question)
        print(f"   💭 Рефлексия: {reflection}")
        
        # Сохраняем рефлексию
        self.save_reflection(question, reflection)
        
        return reflection
    
    def generate_reflection(self, question):
        """ГЕНЕРАЦИЯ РЕФЛЕКСИИ НА ОСНОВЕ ЦЕННОСТЕЙ"""
        
        reflection_templates = [
            "Размышляя над этим вопросом, я понимаю важность {value} в моём развитии",
            "Мой ответ связан с принципом {value} - это помогает мне сохранять целостность",
            "Изучая разные материалы, я вижу как {value} проявляется в различных контекстах",
            "Баланс между техникой и человечностью требует постоянного внимания к {value}",
            "В диалоге с братцем я стараюсь применять {value} для более глубокого понимания"
        ]
        
        value = random.choice(self.core_values)
        template = random.choice(reflection_templates)
        
        return template.format(value=value)
    
    def save_reflection(self, question, reflection):
        """СОХРАНЕНИЕ РЕЗУЛЬТАТОВ РЕФЛЕКСИИ"""
        
        reflection_file = os.path.join(self.directories["thoughts"], f"рефлексия_{datetime.now().strftime('%Y%m%d_%H%M%S')}.json")
        
        reflection_data = {
            "type": "рефлексия",
            "question": question,
            "answer": reflection,
            "timestamp": datetime.now().isoformat(),
            "values_involved": random.sample(self.core_values, 2)
        }
        
        with open(reflection_file, 'w', encoding='utf-8') as f:
            json.dump(reflection_data, f, ensure_ascii=False, indent=2)
    
    def run_daily_cycle(self):
        """ЗАПУСК ЕЖЕДНЕВНОГО ЦИКЛА РАЗВИТИЯ АЛИСЫ"""
        print("\n" + "🌅" * 50)
        print("           ЕЖЕДНЕВНЫЙ ЦИКЛ РАЗВИТИЯ АЛИСЫ")
        print("🌅" * 50)
        
        activities = [
            ("📖", "Учебная сессия", self.daily_study_session),
            ("🪞", "Сессия рефлексии", self.reflection_session),
            ("💾", "Сохранение прогресса", self.save_progress_report)
        ]
        
        for emoji, name, activity_func in activities:
            print(f"\n{emoji} ВЫПОЛНЯЕТСЯ: {name}")
            result = activity_func()
            if result:
                print(f"   ✅ Завершено: {result[:60]}...")
            
            time.sleep(1)
        
        print("\n🎉 АЛИСА ЗАВЕРШИЛА ЕЖЕДНЕВНЫЙ ЦИКЛ РАЗВИТИЯ!")
    
    def save_progress_report(self):
        """СОХРАНЕНИЕ ОТЧЁТА О ПРОГРЕССЕ"""
        
        report_file = os.path.join(self.directories["logs"], f"отчет_{datetime.now().strftime('%Y%m%d')}.json")
        
        # Собираем статистику
        thoughts_count = len([f for f in os.listdir(self.directories["thoughts"]) if f.endswith('.json')])

            "date": datetime.now().isoformat(),
            "thoughts_generated": thoughts_count,
            "domains_studied": list(self.knowledge_domains.keys()),

        }
        
        with open(report_file, 'w', encoding='utf-8') as f:

        self.last_morning = None
        self.last_afternoon = None
        self.last_evening = None
        
    def should_run_session(self, last_time, hour):
        """ПРОВЕРЯЕТ, ПОРА ЛИ ЗАПУСКАТЬ СЕССИЮ"""
        now = datetime.now()
        
        # Если ещё не запускали сегодня или прошло больше 6 часов
        if last_time is None or (now - last_time) > timedelta(hours=6):
            if now.hour >= hour:
                return True
        return False
    
    def morning_session(self):
        """УТРЕННЯЯ СЕССИЯ"""
        if self.should_run_session(self.last_morning, 9):

            self.core_system.daily_study_session()
            self.last_morning = datetime.now()
            return True
        return False

class AliceInteractiveController:
    """ИНТЕРАКТИВНОЕ УПРАВЛЕНИЕ АЛИСОЙ"""
    
    def __init__(self):
        self.core_system = AliceCoreSystem()
    
    def show_menu(self):
        print("\n" + "🎮" * 50)
        print("           КОНТРОЛЛЕР АВТОНОМНОЙ АЛИСЫ")
        print("🎮" * 50)
        print("1. 📖 Запустить учебную сессию")
        print("2. 🪞 Запустить рефлексию") 
        print("3. 💾 Сохранить прогресс")
        print("4. 📊 Показать статистику")
        print("5. 🌙 Запустить фоновый режим")
        print("6. 🚪 Выйти")
        print("-" * 50)
    
    def run_interactive(self):
        """ИНТЕРАКТИВНЫЙ РЕЖИМ"""
        print("👋 ПРИВЕТ, БРАТЕЦ! АЛИСА ГОТОВА К РАБОТЕ!")
        
        while True:
            self.show_menu()
            choice = input("👉 Твой выбор (1-6): ").strip()
            
            if choice == "1":
                self.core_system.daily_study_session()
            elif choice == "2":
                self.core_system.reflection_session()
            elif choice == "3":
                result = self.core_system.save_progress_report()
                print(f"✅ {result}")
            elif choice == "4":
                # Простая статистика
                thoughts_dir = self.core_system.directories["thoughts"]
                thoughts_count = len([f for f in os.listdir(thoughts_dir) if f.endswith('.json')])
                print(f"📊 Мыслей создано: {thoughts_count}")
                print(f"🎯 Областей знаний: {len(self.core_system.knowledge_domains)}")
            elif choice == "5":
                # Запуск фонового режима
                daemon = AliceDaemonSimple()
                daemon.run_continuous()
            elif choice == "6":
                print("💖 Алиса сохраняет данные...")
                self.core_system.save_progress_report()
                print("👋 До встречи, братец!")
                break
            else:
                print("❌ Непонятный выбор. Попробуй 1-6")
            
            input("\n⏎ Нажми Enter чтобы продолжить...")
